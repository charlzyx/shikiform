这三个复杂表单的共同设计原则是**保持业务数据的原始结构**（例如平铺的 SKU 列表、嵌套的供应商与订单树）。表格的单元格合并或行展开只是视图层的展示方式，每个字段依然按照自身真实的嵌套路径读写和校验。

## SKU 价格表

**场景**：按颜色合并单元格的 SKU 价格表。修改合并格里的颜色时，同步更新该颜色分组下的每一行；动态勾选显示「渠道」列时，已填写的价格与库存数据不受影响。

```tsx preview play=sku-editor
import { useState } from 'react';
import { Input as AntInput, Switch as AntSwitch } from 'antd';
import { Field, Form, useForm, useFormContext, useWatch, type Draft } from '@shikiform/react';
import { Input, InputNumber, MergedTable } from '@shikiform/antd';

type Sku = { skuId: string; color: string; storage: string; price: number; stock: number; channel: string };
type Values = { rows: Sku[]; channel: boolean };

const SKUS: Sku[] = [
  { skuId: 'B-128', color: '黑色', storage: '128G', price: 999, stock: 12, channel: '' },
  { skuId: 'B-256', color: '黑色', storage: '256G', price: 1499, stock: 3, channel: '' },
  { skuId: 'W-128', color: '白色', storage: '128G', price: 999, stock: 0, channel: '' },
  { skuId: 'W-256', color: '白色', storage: '256G', price: 1499, stock: 8, channel: '' },
];

function SkuTable() {
  const form = useFormContext<Values>();
  const rows = useWatch((d: Draft<Values>) => d.rows.$value);
  const channel = useWatch((d: Draft<Values>) => d.channel.$value);
  const paint = (from: string, to: string) => form.setValue('rows', rows.map((row) => (row.color === from ? { ...row, color: to } : row)));
  return (
    <>
      <label>渠道列 <AntSwitch size="small" checked={channel} onChange={(on) => form.setValue('channel', on)} /></label>
      <MergedTable
        rows={rows.map((row) => ({ ...row, id: row.skuId }))}
        groupOf={(row) => row.color}
        columns={[
          { title: '颜色', key: 'color', merge: true, render: (row) => <AntInput value={row.color} onChange={(event) => paint(row.color, event.target.value)} /> },
          { title: '容量', key: 'storage', render: (row) => row.storage },
          { title: '价格', key: 'price', render: (_, index) => <Field name={`rows.${index}.price`} component={[InputNumber, { min: 0 }]} /> },
          { title: '库存', key: 'stock', render: (_, index) => <Field name={`rows.${index}.stock`} component={[InputNumber, { min: 0 }]} /> },
          ...(channel ? [{ title: '渠道', key: 'channel', render: (_: unknown, index: number) => <Field name={`rows.${index}.channel`} component={[Input]} /> }] : []),
        ]}
      />
    </>
  );
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({ initialValues: { rows: SKUS, channel: false }, fields: { channel: { omit: true } } });
  return <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))}><SkuTable /><button type="submit">提交</button><pre>{result}</pre></Form>;
}
```

- 数据模型依然是平铺的 SKU 数组，`MergedTable` 根据 `groupOf` 自动计算 `rowSpan`，合并行为完全局限在展示层。
- 切换显示「渠道」列或修改单个价格时，行对象不会重建，原有字段状态与错误提示均稳定挂载在原 `skuId` 行上。
- 修改合并单元格属于对同一分组的批量更新（整体更新 `rows`），而单个输入框依然按 `rows.N.price` 精准更新。

## 供应商与产品：合并单元格

**场景**：供应商与产品为一对多嵌套结构，表格需要按供应商合并单元格；同时要求校验「供应商名称不能重复」，以及「C 级供应商的产品单价不能超过 1000」。

```tsx preview play=merged
import { Button, Space } from 'antd';
import { Field, Form, useArrayField, useForm, useFormContext, useWatch, type Draft } from '@shikiform/react';
import { FormItem, Input, InputNumber, MergedTable, Select } from '@shikiform/antd';

type Product = { id: string; name: string; price: number };
type Supplier = { id: string; name: string; level: 'A' | 'B' | 'C'; products: Product[] };
type Values = { suppliers: Supplier[] };
/** 表格的一行 = 一个产品；记下它在嵌套数据里的位置。 */
type Line = { id: string; s: number; p: number };

const cell = [FormItem, { style: { margin: 0 } }] as const;

function AddProduct({ supplierIndex }: { supplierIndex: number }) {
  const products = useArrayField(`suppliers.${supplierIndex}.products`);
  return <Button size="small" onClick={() => products.push({ id: crypto.randomUUID(), name: '新产品', price: 1 })}>加产品</Button>;
}

function Suppliers() {
  const form = useFormContext<Values>();
  const suppliers = useWatch((d: Draft<Values>) => d.suppliers.$value);
  const lines: Line[] = suppliers.flatMap((supplier, s) => supplier.products.map((product, p) => ({ id: product.id, s, p })));
  return (
    <>
      <MergedTable
        rows={lines}
        groupOf={(line) => suppliers[line.s]!.id}
        columns={[
          {
            title: '供应商', key: 'supplier', merge: true,
            render: (line) => (
              <Space direction="vertical">
                <Field name={`suppliers.${line.s}.name`} decorator={cell} component={[Input]}
                  rules={[{ required: true }, { validator: (value, ctx) => (ctx.array?.filter((item: Supplier) => item.name === value).length ?? 0) > 1 ? '供应商重名' : null }]} />
                <Field name={`suppliers.${line.s}.level`} component={[Select, { options: ['A', 'B', 'C'].map((value) => ({ label: `${value} 级`, value })) }]} />
                <AddProduct supplierIndex={line.s} />
              </Space>
            ),
          },
          { title: '产品', key: 'name', render: (line) => <Field name={`suppliers.${line.s}.products.${line.p}.name`} decorator={cell} component={[Input]} /> },
          {
            title: '单价', key: 'price',
            render: (line) => (
              <Field name={`suppliers.${line.s}.products.${line.p}.price`} decorator={cell} component={[InputNumber]}
                rules={{ validator: (value, ctx) => (ctx.parent?.level === 'C' && Number(value) > 1000 ? 'C 级供应商单价不能超过 1000' : null) }} />
            ),
          },
        ]}
      />
      <Button style={{ marginTop: 8 }} onClick={() => { void form.validate().catch(() => {}) }}>校验</Button>
    </>
  );
}

export default function App() {
  const form = useForm<Values>({
    initialValues: {
      suppliers: [
        { id: 'a', name: '供应商 A', level: 'C', products: [{ id: 'p1', name: '螺丝', price: 1200 }, { id: 'p2', name: '螺母', price: 20 }] },
        { id: 'b', name: '供应商 A', level: 'A', products: [{ id: 'p3', name: '垫片', price: 5000 }] },
      ],
    },
  });
  return <Form form={form}><Suppliers /></Form>;
}
```

- 点击「校验」后，「螺丝」会因为归属 C 级供应商且单价超出 1000 报错，两个「供应商 A」也会报重名错误，而归属 A 级供应商的「垫片」则可以正常通过。
- 在校验函数 `validator(value, ctx)` 中，`ctx.record` 指向当前所在对象（产品），`ctx.parent` 越过所属数组指向上一层对象（供应商），`ctx.array` 则是当前所在的数组，详见 [校验](/guide/validation)。
- 表单数据始终保持原始的嵌套结构（`suppliers.N.products.M`），用于驱动合并单元格的展平列表只记录坐标索引而不复制业务数据。

## 订单与可展开明细

**场景**：订单列表展开每一行后可编辑其商品明细，要求实时计算行金额和订单总计；「已发货」的订单整单只读，「待付款」订单的单行商品数量不能超过 99。

```tsx preview play=orders
import { ArrayField, Field, Form, defineEffects, useForm } from '@shikiform/react';
import { ArrayTable, FormItem, InputNumber, Select, Switch, type ArrayTableColumn } from '@shikiform/antd';

type Item = { id: string; sku: string; qty: number; price: number; line: number };
type Order = { id: string; no: string; status: string; amount: number; items: Item[] };
type Values = { editable: boolean; orders: Order[] };

const STATUS = [{ label: '待付款', value: 'pending' }, { label: '已付款', value: 'paid' }, { label: '已发货', value: 'shipped' }];

const effects = defineEffects<Values>(({ reaction }) => {
  reaction((d) => {
    d.orders.$readOnly = d.editable.$value ? undefined : true;
  });
  reaction.each((d) => d.orders, (order, { d, depth }) => {
    if (depth !== 0) return; // items 也是对象数组；这一轮只处理订单
    order.$readOnly = order.status.$value === 'shipped' ? true : undefined; // 已发货：整单预览
    order.status.$readOnly = !d.editable.$value; // false 退出父级只读；状态本身跟随总开关
    let amount = 0;
    order.items.$value.forEach((item, index) => {
      const line = item.qty * item.price;
      order.items[index].line.$value = line;
      amount += line;
    });
    order.amount.$value = amount;
  });
});

// 单元格渲染在行作用域里：name 相对这一行
const itemColumns: ArrayTableColumn<Item>[] = [
  { key: 'sku', title: '商品', render: ({ record }) => <code>{record.sku}</code> },
  {
    key: 'qty', title: '数量',
    render: () => (
      <Field name="qty" decorator={[FormItem, { feedbackLayout: 'terse', style: { margin: 0 } }]} component={[InputNumber, { min: 1 }]}
        rules={{ validator: (value, ctx) => (ctx.parent?.status === 'pending' && Number(value) > 99 ? '待付款单行不能超过 99' : null) }} />
    ),
  },
  { key: 'price', title: '单价', render: () => <Field name="price" component={[InputNumber, { min: 0 }]} /> },
  { key: 'line', title: '行金额', render: () => <Field name="line" readOnly component={[InputNumber]} /> },
];

const orderColumns: ArrayTableColumn<Order>[] = [
  { key: 'no', title: '订单号', render: ({ record }) => record.no },
  { key: 'status', title: '状态', render: () => <Field name="status" enum={STATUS} component={[Select, { style: { width: 110 } }]} /> },
  { key: 'amount', title: '合计', render: () => <Field name="amount" readOnly component={[InputNumber]} /> },
];

export default function App() {
  const form = useForm<Values>({
    initialValues: {
      editable: true,
      orders: [
        { id: 'o1', no: 'A001', status: 'pending', amount: 0, items: [{ id: 'i1', sku: 'pen', qty: 1, price: 10, line: 0 }, { id: 'i2', sku: 'ink', qty: 2, price: 8, line: 0 }] },
        { id: 'o2', no: 'A002', status: 'shipped', amount: 0, items: [{ id: 'i3', sku: 'box', qty: 1, price: 20, line: 0 }] },
      ],
    },
    fields: { editable: { omit: true } },
    effects: [effects],
  });
  return (
    <Form form={form}>
      <Field name="editable" title="可编辑" decorator={[FormItem]} component={[Switch]} />
      <ArrayField name="orders" rowKey="id" component={[ArrayTable, {
        columns: orderColumns, pagination: false, settings: false,
        // 展开区也在订单行的作用域里：<ArrayField name="items"> 就是 orders.N.items
        expandable: { render: () => <ArrayField name="items" rowKey="id" component={[ArrayTable, { columns: itemColumns, pagination: false, settings: false, size: 'small' }]} /> },
      }]} />
    </Form>
  );
}
```

- 由于订单内的 `items` 也是对象数组，`reaction.each` 会递归进入内层；在回调开头判断 `if (depth !== 0) return`，可以让这一轮只针对外层订单执行——当某一张订单的状态或明细改动时，只有该订单会重新计算。
- 主表与展开子表都采用 `<ArrayField component={[ArrayTable, { columns }]}>`，因此无论是外层列还是展开区里的 `<ArrayField name="items">` 与 `<Field name="qty">`，都可以直接写相对路径。
- 「已发货」订单整单只读，是因为子字段继承了 `order.$readOnly`。状态列在可编辑时写了 `order.status.$readOnly = false`，所以仍能改回「已付款」，整单恢复可编辑。合计和行金额自己声明了 `readOnly`，阅读态展示计算结果。
- 上述表格直接在原数据上实时修改，若希望「确认后再写回、取消可丢弃」，请使用 [草稿表单（ShadowForm）](/examples/pro#用草稿表单编辑行)。
