## 查询表

**场景**：日志查询页面。选择「供应商」动态更新「模型」的可选项，支持筛选区展开/收起（收起时保留已输入的值）以及按状态过滤；发起新查询时重置到第 1 页，翻页时保留已勾选项。

```tsx preview play=query
import { useState } from 'react';
import { Field, Form, defineEffects, useField } from '@shikiform/react';
import { useQueryTable } from '@shikiform/hooks';

type Log = { id: string; keyword: string; provider: string; model: string; status: string };
type Query = { keyword: string; provider: string; model: string; user: string };
type Values = { query: Query; list: Log[] };

const LOGS: Log[] = Array.from({ length: 12 }, (_, i) => ({
  id: String(i + 1),
  keyword: `${i % 2 ? 'beta' : 'alpha'}-${i + 1}`,
  provider: i % 2 ? 'anthropic' : 'openai',
  model: i % 2 ? 'claude' : 'gpt',
  status: i % 3 ? 'ok' : 'error',
}));
const MODELS: Record<string, string[]> = { openai: ['gpt'], anthropic: ['claude'], '': ['gpt', 'claude'] };
const options = (values: string[]) => values.map((value) => ({ label: value, value }));

const effects = defineEffects<Values>(({ reaction, watch }) => {
  reaction((d) => {
    d.query.model.$enum = options(MODELS[d.query.provider.$value] ?? []);
  });
  watch((d) => d.query.provider.$value, (d) => {
    if (!(MODELS[d.query.provider.$value] ?? []).includes(d.query.model.$value)) d.query.model.$value = '';
  });
});

function TextFilter() {
  const f = useField<string>();
  return (
    <label htmlFor={f.id} style={{ display: 'inline-flex', alignItems: 'center', gap: 4 }}>
      {f.title}
      <input id={f.id} style={{ width: 100 }} value={f.value ?? ''} onChange={(e) => f.onChange(e.target.value)} />
    </label>
  );
}

function SelectFilter() {
  const f = useField<string>();
  return (
    <label htmlFor={f.id} style={{ display: 'inline-flex', alignItems: 'center', gap: 4 }}>
      {f.title}
      <select id={f.id} value={f.value ?? ''} onChange={(e) => f.onChange(e.target.value)}>
        <option value="">全部</option>
        {f.options.map((opt) => (
          <option key={String(opt.value)} value={String(opt.value)}>{opt.label}</option>
        ))}
      </select>
    </label>
  );
}

export default function App() {
  const [expanded, setExpanded] = useState(false);
  const table = useQueryTable({
    initialValues: { query: { keyword: '', provider: '', model: '', user: '' }, list: [] } as Values,
    fields: { 'query.provider': { enum: options(['openai', 'anthropic']) } },
    effects: [effects],
    defaultPageSize: 5,
    params: 'query',
    rows: 'list',
    request: async ({ params, page, pageSize, filters }) => {
      const rows = LOGS.filter((row) =>
        (!params.keyword || row.keyword.includes(params.keyword)) &&
        (!params.provider || row.provider === params.provider) &&
        (!params.model || row.model === params.model) &&
        (!params.user || row.id === params.user) &&
        (!filters.status || filters.status.includes(row.status)));
      return { data: rows.slice((page - 1) * pageSize, page * pageSize), total: rows.length };
    },
  });
  const pages = Math.max(1, Math.ceil(table.total / table.pageSize));
  const statusFilter = table.filteredValueOf('status')?.[0] ?? '';
  const toggleSelect = (id: string, checked: boolean) =>
    table.setSelectedKeys(checked ? [...table.selectedKeys, id] : table.selectedKeys.filter((k) => k !== id));
  return (
    <Form form={table.form} onSubmit={() => table.submit()}>
      <div style={{ display: 'flex', flexWrap: 'wrap', gap: 8, alignItems: 'center', marginBottom: 8 }}>
        <Field name="query.keyword" title="关键字" component={[TextFilter]} />
        <Field name="query.provider" title="供应商" component={[SelectFilter]} />
        <Field name="query.model" title="模型" component={[SelectFilter]} />
        <span style={{ display: expanded ? 'inline-flex' : 'none' }}>
          <Field name="query.user" title="用户 ID" component={[TextFilter]} />
        </span>
        <button type="submit">查询</button>
        <button type="button" onClick={table.reset}>重置</button>
        <button type="button" onClick={() => setExpanded((v) => !v)}>{expanded ? '收起' : '展开'}</button>
      </div>
      <div style={{ display: 'flex', gap: 8, marginBottom: 8 }}>
        <span>状态筛选：</span>
        {(['', 'ok', 'error'] as const).map((status) => (
          <button
            key={status || 'all'}
            type="button"
            style={{ fontWeight: statusFilter === status ? 700 : 400 }}
            onClick={() => table.setFilters(status ? { status: [status] } : {})}
          >
            {status === '' ? '全部' : status === 'ok' ? '成功' : '失败'}
          </button>
        ))}
      </div>
      <table>
        <thead>
          <tr>
            <th />
            <th>关键字</th>
            <th>供应商</th>
            <th>模型</th>
            <th>状态</th>
          </tr>
        </thead>
        <tbody>
          {table.data.map((row) => (
            <tr key={row.id}>
              <td>
                <input
                  type="checkbox"
                  aria-label={`选择 ${row.id}`}
                  checked={table.selectedKeys.includes(row.id)}
                  onChange={(e) => toggleSelect(row.id, e.target.checked)}
                />
              </td>
              <td>{row.keyword}</td>
              <td>{row.provider}</td>
              <td>{row.model}</td>
              <td>{row.status}</td>
            </tr>
          ))}
        </tbody>
      </table>
      <div style={{ marginTop: 8 }}>
        <button type="button" disabled={table.page <= 1} onClick={() => table.onPageChange(table.page - 1)}>上一页</button>
        <span> 第 {table.page} / {pages} 页 · 共 {table.total} 条 · 已选 {table.selectedKeys.length} </span>
        <button type="button" disabled={table.page >= pages} onClick={() => table.onPageChange(table.page + 1)}>下一页</button>
      </div>
    </Form>
  );
}
```

- `useQueryTable`（`@shikiform/hooks`）统一托管查询条件、分页、勾选、表头筛选和请求竞态；由于底层仍是标准 ShikiForm 引擎，条件之间可以照常编写 `reaction` 和 `watch`。
- 配置 `rows: 'list'` 后，接口返回的每一页数据会写入表单的 `list` 数组字段，因此表格单元格也可以直接放置 `<Field>` 进行行内编辑。
- 收起额外筛选条件时通过 `display: none` 隐藏而不卸载节点，已输入的值不会丢失。
- 调用 `table.setFilters` 切换筛选条件时，参数会传给 `request` 的 `filters`，同时回到第 1 页并清空勾选项；单纯翻页时勾选项会保留。详见 [查询 Hooks](/reference/query-hooks)。

## 极简查询表

**场景**：直接将同一个 `useQueryTable` Hook 接入最精简的原生 HTML 表单和表格。

```tsx preview play=query-native
import { Field, Form } from '@shikiform/react';
import { useQueryTable } from '@shikiform/hooks';

type Log = { id: string; keyword: string };
const LOGS: Log[] = Array.from({ length: 12 }, (_, i) => ({ id: String(i + 1), keyword: `${i % 2 ? 'beta' : 'alpha'}-${i + 1}` }));

export default function App() {
  const table = useQueryTable({
    initialValues: { keyword: '' },
    defaultPageSize: 5,
    params: (values: { keyword: string }) => values,
    request: async ({ params, page, pageSize }) => {
      const rows = LOGS.filter((row) => row.keyword.includes(params.keyword));
      return { data: rows.slice((page - 1) * pageSize, page * pageSize), total: rows.length };
    },
  });
  const pages = Math.max(1, Math.ceil(table.total / table.pageSize));
  const toggle = (id: string, on: boolean) =>
    table.setSelectedKeys(on ? [...table.selectedKeys, id] : table.selectedKeys.filter((key) => key !== id));
  return (
    <Form form={table.form} onSubmit={() => table.submit()}>
      <Field name="keyword">
        {(field) => <input aria-label="关键字" placeholder="alpha" value={field.value} onChange={(event) => field.onChange(event.target.value)} />}
      </Field>{' '}
      <button type="submit">查询</button> <button type="button" onClick={table.reset}>重置</button>{' '}
      <button type="button" onClick={table.refresh}>刷新</button>
      <table>
        <tbody>
          {table.data.map((row) => (
            <tr key={row.id}>
              <td><input type="checkbox" aria-label={`选择 ${row.id}`} checked={table.selectedKeys.includes(row.id)} onChange={(event) => toggle(row.id, event.target.checked)} /></td>
              <td>{row.keyword}</td>
            </tr>
          ))}
        </tbody>
      </table>
      <button type="button" disabled={table.page <= 1} onClick={() => table.onPageChange(table.page - 1)}>上一页</button>
      <span> {table.page} / {pages} · 已选 {table.selectedKeys.length} {table.loading ? '· 加载中' : ''} </span>
      <button type="button" disabled={table.page >= pages} onClick={() => table.onPageChange(table.page + 1)}>下一页</button>
    </Form>
  );
}
```

- `useQueryTable` 只返回纯状态与操作函数（`data`、`total`、`page`、`selectedKeys`、`submit`、`onPageChange` 等），对 UI 框架没有任何依赖。
- 通过 `<Form form={table.form} onSubmit={() => table.submit()}>` 包装后，用户在输入框内按回车也能直接触发查询。

## 面板里编辑一行

**场景**：表格本身只读展示，点击某行的「编辑」在编辑面板中修改数据；点击取消放弃修改，点击确定校验通过后按 `id` 写回原行。

```tsx preview play=row-edit
import { useState } from 'react';
import { Field, Form, ObjectField, Watch, useForm, type Draft } from '@shikiform/react';

type Row = { id: string; name: string; price: number };
type Values = { rows: Row[]; editor: Row & { open: boolean } };

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({
    initialValues: {
      rows: [{ id: 'a', name: '钢笔', price: 12 }, { id: 'b', name: '墨水', price: 8 }],
      editor: { open: false, id: '', name: '', price: 0 },
    },
    fields: { editor: { omit: true } }, // 编辑区不进提交结果
  });
  const open = (row: Row) => form.setValue('editor', { ...row, open: true });
  const close = () => form.setValue('editor.open', false);
  const save = async () => {
    try { await form.validate('editor'); } catch { return; }
    const { id, name, price } = form.getValue('editor');
    const index = form.getValue('rows').findIndex((row) => row.id === id);
    form.batch((d) => {
      d.rows[index].name.$value = name;
      d.rows[index].price.$value = price;
      d.editor.open.$value = false;
    });
  };
  return (
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))}>
      <Watch select={(d: Draft<Values>) => d.rows.$value}>
        {(rows) => (
          <table>
            <thead>
              <tr><th>名称</th><th>价格</th><th>操作</th></tr>
            </thead>
            <tbody>
              {rows.map((row) => (
                <tr key={row.id}>
                  <td>{row.name}</td>
                  <td>{row.price}</td>
                  <td><button type="button" onClick={() => open(row)}>编辑</button></td>
                </tr>
              ))}
            </tbody>
          </table>
        )}
      </Watch>
      <Watch select={(d: Draft<Values>) => d.editor.open.$value}>
        {(visible) => visible ? (
          <fieldset style={{ marginTop: 12 }}>
            <legend>编辑行</legend>
            <ObjectField name="editor">
              <Field<Row, 'name'> name="name" title="名称" required>
                {(f) => (
                  <p>
                    <label htmlFor={f.id}>{f.title}：</label>
                    <input id={f.id} value={f.value ?? ''} onChange={(e) => f.onChange(e.target.value)} onBlur={f.onBlur} />
                    {f.errors[0] && <small role="alert" style={{ color: 'crimson' }}> {f.errors[0]}</small>}
                  </p>
                )}
              </Field>
              <Field<Row, 'price'> name="price" title="价格">
                {(f) => (
                  <p>
                    <label htmlFor={f.id}>{f.title}：</label>
                    <input
                      id={f.id}
                      type="number"
                      min={0}
                      value={f.value ?? 0}
                      onChange={(e) => f.onChange(Number(e.target.value))}
                      onBlur={f.onBlur}
                    />
                  </p>
                )}
              </Field>
            </ObjectField>
            <div style={{ display: 'flex', gap: 8 }}>
              <button type="button" onClick={close}>取消</button>
              <button type="button" onClick={() => void save()}>确定</button>
            </div>
          </fieldset>
        ) : null}
      </Watch>
      <button type="submit" style={{ marginTop: 12 }}>提交（看 editor 不在结果里）</button>
      <pre>{result}</pre>
    </Form>
  );
}
```

- 编辑区使用同一个表单中声明为 `omit: true` 的 `editor` 子对象：可通过 `form.validate('editor')` 单独校验该子树，且不会进入提交结果。
- 写回时用 `form.batch` 同时更新原行字段并关闭编辑面板，合并为一轮联动与一次渲染通知。
- 需要弹窗或面板使用独立的表单实例时，见 [草稿表单（useShadowForm）](/examples/pro#草稿表单useShadowForm)。
