## 联动计算

金额由单价和数量派生时，使用 `reaction` 计算，不要放进 `watch`。

```tsx preview
import { useState } from 'react';
import { defineEffects, Form, Submit, typed, useField, useForm } from '@shikiform/react';

type Values = { price: number | null; qty: number | null; amount: number };

const effects = defineEffects<Values>(({ reaction }) => {
  reaction((d) => {
    const amount = (d.price.$value ?? 0) * (d.qty.$value ?? 0);
    d.amount.$value = amount;
  });
});

function NumberInput({ min }: { min?: number }) {
  const f = useField<number | null>();
  return (
    <p>
      <label htmlFor={f.id}>{f.title}：</label>
      <input
        id={f.id}
        type="number"
        min={min}
        value={f.value ?? ''}
        onChange={(e) => f.onChange(e.target.value === '' ? null : Number(e.target.value))}
        onBlur={f.onBlur}
      />
    </p>
  );
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({
    initialValues: { price: 10, qty: 2, amount: 0 },
    effects: [effects],
  });
  const F = typed(form);
  return (
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))}>
      <F.Field name="price" title="单价" component={[NumberInput, { min: 0 }]} />
      <F.Field name="qty" title="数量" component={[NumberInput, { min: 0 }]} />
      <F.Field name="amount" title="金额" component={[NumberInput]} />
      <Submit>提交</Submit>
      <pre aria-live="polite">{result}</pre>
    </Form>
  );
}
```

<Aside type="tip" title="计算字段的编辑与提交">
在 `reaction` 中写入 `$value` 的字段为计算字段：默认仍可编辑、仍随表单提交、不计入表单级 `dirty`。如需渲染为只读预览，加上 `readOnly`；仅展示不提交则声明 `omit: true`。
</Aside>

## 条件显隐

`$visible = false` 只负责不渲染、不校验，值默认仍会提交。下面同时写了 `$omit = !need`，所以关掉开关后税号不进提交 JSON。

```tsx preview
import { useState } from 'react';
import { Checkbox, CheckboxIndicator } from '@ui-sdk/react/checkbox';
import { defineEffects, Form, Submit, typed, useForm } from '@shikiform/react';

type Values = { invoice: boolean; taxId: string };

const effects = defineEffects<Values>(({ reaction }) => {
  reaction((d) => {
    const need = d.invoice.$value;
    d.taxId.$visible = need;
    d.taxId.$omit = !need;
    d.taxId.$required = need;
  });
});

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({
    initialValues: { invoice: false, taxId: '' },
    effects: [effects],
  });
  const F = typed(form);
  return (
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))}>
      <F.Field name="invoice" title="需要发票">
        {(f) => (
          <p>
            <label htmlFor={f.id} style={{ display: 'inline-flex', alignItems: 'center', gap: 6, cursor: 'pointer' }}>
              <Checkbox id={f.id} checked={Boolean(f.value)} onCheckedChange={(checked) => f.onChange(checked === true)}>
                <CheckboxIndicator>✓</CheckboxIndicator>
              </Checkbox>
              {f.title}
            </label>
          </p>
        )}
      </F.Field>
      <F.Field name="taxId" title="税号">
        {(f) => (
          <p>
            <label htmlFor={f.id}>{f.title}：</label>
            <input
              id={f.id}
              value={f.value ?? ''}
              onChange={(e) => f.onChange(e.target.value)}
              onBlur={f.onBlur}
            />
            {f.errors[0] && <small role="alert"> {f.errors[0]}</small>}
          </p>
        )}
      </F.Field>
      <Submit>提交</Submit>
      <pre aria-live="polite">{result}</pre>
    </Form>
  );
}
```

## SKU 选择器

**场景**：商品包含颜色、容量、版本三个规格维度，用户可以按任意顺序点选；需要根据库存自动禁用无货组合，并在切换时清理冲突选项，同时实时展示价格区间与可用库存。

```tsx preview play=sku-picker
import { Radio, RadioGroup, RadioIndicator } from '@ui-sdk/react/radio';
import { Field, Form, defineEffects, useField, useForm, type Draft } from '@shikiform/react';
import { comboMatch, comboOptions } from '@shikiform/core';

const SKUS = [
  { skuId: 'B-128-STD', color: '黑色', storage: '128G', edition: '标准版', price: 999, stock: 12 },
  { skuId: 'B-256-PRO', color: '黑色', storage: '256G', edition: 'Pro', price: 1499, stock: 3 },
  { skuId: 'W-256-PRO', color: '白色', storage: '256G', edition: 'Pro', price: 1499, stock: 8 },
  { skuId: 'W-128-STD', color: '白色', storage: '128G', edition: '标准版', price: 999, stock: 0 },
  { skuId: 'R-512-MAX', color: '红色', storage: '512G', edition: 'Max', price: 1999, stock: 5 },
];
const DIMS = ['color', 'storage', 'edition'] as const;
type Dim = (typeof DIMS)[number];
type Selection = Record<Dim, string>; // '' 表示未选
type Values = Selection & { qty: number; skuId: string; price: string; stock: string };

const selectionOf = (d: Draft<Values>): Selection => ({ color: d.color.$value, storage: d.storage.$value, edition: d.edition.$value });
const matching = (sel: Selection) => SKUS.filter((row) => DIMS.every((dim) => !sel[dim] || row[dim] === sel[dim]));
/** 选了 dim = value 之后：保留它，清掉和它组合不出来的其他维度。 */
function afterPick(sel: Selection, dim: Dim, value: string): Selection {
  const next: Selection = { ...sel, [dim]: value };
  for (const other of DIMS) {
    if (other !== dim && next[other] && !comboMatch(SKUS, { [dim]: value, [other]: next[other] })) next[other] = '';
  }
  return next;
}

const effects = defineEffects<Values>(({ reaction, watch }) => {
  // 选项：每个值都能点；点下去之后没有有货的组合，就禁用
  reaction((d) => {
    const sel = selectionOf(d);
    for (const dim of DIMS) {
      d[dim].$enum = comboOptions(SKUS, {}, dim).map((option) => {
        const value = String(option.value);
        return {
          label: option.label,
          value,
          disabled: !matching(afterPick(sel, dim, value)).some((row) => row.stock > 0),
        };
      });
    }
  });
  // 动作：改了一个维度，清掉与它冲突的其他维度
  for (const dim of DIMS) {
    watch((d) => d[dim].$value, (d) => {
      const sel = selectionOf(d);
      const next = afterPick(sel, dim, sel[dim]);
      for (const other of DIMS) if (next[other] !== sel[other]) d[other].$value = next[other];
    });
  }
  // 派生：价格区间、库存、数量上限
  reaction((d) => {
    const rows = matching(selectionOf(d));
    const sku = rows.length === 1 ? rows[0] : undefined;
    const prices = rows.map((row) => row.price);
    const [min, max] = [Math.min(...prices), Math.max(...prices)];
    d.skuId.$value = sku?.skuId ?? '—';
    d.price.$value = min === max ? `¥${min}` : `¥${min} ~ ¥${max}`;
    d.stock.$value = sku ? `${sku.stock} 件` : '—';
    d.qty.$disabled = !sku;
    d.qty.$rules = [{ max: sku?.stock ?? 1, message: `库存仅剩 ${sku?.stock ?? 0} 件` }];
  });
});

function RadioRow() {
  const f = useField<string>();
  return (
    <div style={{ display: 'flex', alignItems: 'center', gap: 10, marginBottom: 8 }}>
      <span style={{ width: 48 }}>{f.title}：</span>
      <RadioGroup value={f.value || null} onValueChange={(v) => f.onChange(v ?? '')}>
        {f.options.map((opt) => (
          <label
            key={String(opt.value)}
            style={{ display: 'inline-flex', alignItems: 'center', gap: 6, opacity: opt.disabled ? 0.45 : 1, cursor: opt.disabled ? 'not-allowed' : 'pointer' }}
          >
            <Radio value={String(opt.value)} disabled={opt.disabled}>
              <RadioIndicator />
            </Radio>
            {opt.label}
          </label>
        ))}
      </RadioGroup>
    </div>
  );
}

function ReadOnlyText() {
  const f = useField<string>();
  return (
    <p style={{ margin: '4px 0' }}>
      <span>{f.title}：</span>
      <strong>{f.value || '—'}</strong>
    </p>
  );
}

export default function App() {
  const form = useForm<Values>({
    initialValues: { color: '', storage: '', edition: '', qty: 1, skuId: '', price: '', stock: '' },
    fields: { skuId: { omit: true }, price: { omit: true }, stock: { omit: true } }, // 展示用，不进提交结果
    effects: [effects],
  });
  return (
    <Form form={form}>
      <Field name="color" title="颜色" component={[RadioRow]} />
      <Field name="storage" title="容量" component={[RadioRow]} />
      <Field name="edition" title="版本" component={[RadioRow]} />
      <Field name="price" title="价格" readOnly component={[ReadOnlyText]} />
      <Field name="stock" title="库存" readOnly component={[ReadOnlyText]} />
      <Field name="skuId" title="SKU" readOnly component={[ReadOnlyText]} />
      <Field<Values, 'qty'> name="qty" title="数量">
        {(f) => (
          <p style={{ margin: '8px 0 0' }}>
            <label htmlFor={f.id}>{f.title}：</label>
            <input
              id={f.id}
              type="number"
              min={1}
              style={{ width: 80 }}
              disabled={f.disabled}
              value={f.value ?? 1}
              onChange={(e) => f.onChange(Number(e.target.value))}
              onBlur={f.onBlur}
            />
            {f.errors[0] && <small role="alert" style={{ color: 'crimson' }}> {f.errors[0]}</small>}
          </p>
        )}
      </Field>
    </Form>
  );
}
```

- `reaction` 计算可选项与价格区间（[派生](/guide/linkage#reaction-与-watch)），`watch` 在某个维度变更后清理互斥维度（动作），`$rules` 让数量上限跟随库存更新。
- `comboOptions` 与 `comboMatch`（`@shikiform/core`）对组合表做过滤匹配，组合表数组由业务方传入。
- 价格、库存、SKU 编码在 `reaction` 里写入 `$value`，字段声明 `readOnly` 后以阅读态展示；同时声明 `omit: true`，不随表单提交。
- 先选「白色」时，库存为 0 的「128G」自动禁用；先选「128G」时「白色」同样禁用。

## 按勾选生成行

**场景**：勾选「颜色 × 容量」的规格组合后生成对应的价格明细行；取消勾选时移除该行，而已经填好价格的保留行不能在重新同步时被重置。

```tsx preview play=sku
import { useState } from 'react';
import { Checkbox, CheckboxGroup, CheckboxIndicator } from '@ui-sdk/react/checkbox';
import { ArrayField, Field, Form, defineEffects, useField, useForm } from '@shikiform/react';

const CATALOG = [
  { id: '白色/128G', price: 10, stock: 0 },
  { id: '白色/256G', price: 20, stock: 2 },
  { id: '黑色/128G', price: 12, stock: 3 },
  { id: '黑色/256G', price: 22, stock: 9 },
];
const PICKS = CATALOG.map((item) => ({ label: item.id, value: item.id, disabled: item.stock === 0 }));
type Row = { id: string; price: number; qty: number; total: number };
type Values = { picked: string[]; rows: Row[] };

const effects = defineEffects<Values>(({ reaction, watch, draft }) => {
  // 动作：勾选变了才同步行；已有的行原样保留（包括改过的价格）
  watch((d) => d.picked.$value, (d) => {
    const before = d.rows.$value;
    d.rows.$value = d.picked.$value.map((id) =>
      before.find((row) => row.id === id) ?? { id, price: CATALOG.find((item) => item.id === id)?.price ?? 0, qty: 1, total: 0 });
  });
  // 派生：每行一个实例，只重算改动的那一行
  reaction.each(draft.rows, (row) => {
    row.total.$value = row.price.$value * row.qty.$value;
  });
});

function NumberInline({ min }: { min?: number }) {
  const f = useField<number>();
  return (
    <label htmlFor={f.id} style={{ display: 'inline-flex', alignItems: 'center', gap: 4 }}>
      {f.title}
      {f.readOnly ? (
        <strong>{f.value}</strong>
      ) : (
        <input
          id={f.id}
          type="number"
          min={min}
          style={{ width: 64 }}
          value={f.value ?? 0}
          onChange={(e) => f.onChange(Number(e.target.value))}
          onBlur={f.onBlur}
        />
      )}
    </label>
  );
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({
    initialValues: { picked: [], rows: [] },
    effects: [effects],
  });
  return (
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values.rows))}>
      <Field<Values, 'picked'> name="picked" title="组合" enum={PICKS}>
        {(f) => (
          <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', gap: 12, marginBottom: 12 }}>
            <span>{f.title}：</span>
            <CheckboxGroup value={f.value ?? []} onValueChange={(next) => f.onChange(next)}>
              {f.options.map((opt) => {
                const val = String(opt.value);
                return (
                  <label
                    key={val}
                    style={{ display: 'inline-flex', alignItems: 'center', gap: 6, opacity: opt.disabled ? 0.45 : 1, cursor: opt.disabled ? 'not-allowed' : 'pointer' }}
                  >
                    <Checkbox value={val} disabled={opt.disabled}>
                      <CheckboxIndicator>✓</CheckboxIndicator>
                    </Checkbox>
                    {opt.label}
                  </label>
                );
              })}
            </CheckboxGroup>
          </div>
        )}
      </Field>
      <ArrayField name="rows" rowKey="id">
        {() => (
          <div style={{ display: 'flex', gap: 12, alignItems: 'center', marginBottom: 8 }}>
            <Field name="id" readOnly>{(f) => <code style={{ width: 88 }}>{String(f.value ?? '')}</code>}</Field>
            <Field name="price" title="单价" component={[NumberInline, { min: 0 }]} />
            <Field name="qty" title="数量" component={[NumberInline, { min: 1 }]} />
            <Field name="total" title="小计" readOnly component={[NumberInline]} />
          </div>
        )}
      </ArrayField>
      <button type="submit">提交</button>
      <pre>{result}</pre>
    </Form>
  );
}
```

- 同步数组行用 `watch` 而非 `reaction`：`watch` 仅在 `picked` 变化时运行，回调内读取 `d.rows.$value` 不会被收集为依赖，可在同步新行时保留已有行的单价修改。
- `reaction.each` 回调签名为 `(row, ctx)`，`row.total` 等句柄相对于当前行解析，见 [reaction.each](/guide/linkage#reactioneach每一行一个实例)。
- 缺货组合在 `enum` 中标记 `disabled: true`，复选框组自动禁用对应项。
- 也可单独维护一份 `overrides` 字典记录改过的单价，再用 `reaction` 派生整张数组。

## 上千个字段

**场景**：包含 1000 行、共 4000 个字段的价格表，每行根据成本和加价率实时计算建议价；修改其中一个单元格时，只重渲染并重算该行。

```tsx preview play=large
import { memo, useRef } from 'react';
import { ArrayField, Form, defineEffects, useField, useForm } from '@shikiform/react';

type Row = { id: string; cost: number; markup: number; suggested: number };
type Values = { rows: Row[] };

const effects = defineEffects<Values>(({ reaction, draft }) => {
  reaction.each(draft.rows, (row) => {
    row.suggested.$value = Math.round(row.cost.$value * (1 + row.markup.$value / 100));
  });
});

/** 每个单元格只订阅自己的 path；数字是它渲染过的次数。 */
function Cell({ name }: { name: keyof Row }) {
  const field = useField(name);
  const renders = useRef(0);
  renders.current += 1;
  return (
    <label style={{ marginRight: 8 }}>
      <input style={{ width: 72 }} aria-label={name} value={String(field.value ?? '')} readOnly={name === 'suggested'}
        onChange={(event) => field.onChange(Number(event.target.value))} />
      <sup>{renders.current}</sup>
    </label>
  );
}

const Line = memo(function Line({ index }: { index: number }) {
  return (
    <div style={{ height: 36 }}>
      <code style={{ display: 'inline-block', width: 48 }}>#{index}</code>
      <Cell name="cost" /><Cell name="markup" /><Cell name="suggested" />
    </div>
  );
});

function Table() {
  return (
    <ArrayField name="rows" virtualScroll={{ height: 300, rowHeight: 36, overscan: 4 }}>
      {({ index }) => <Line index={index} />}
    </ArrayField>
  );
}

export default function App() {
  const form = useForm<Values>({
    initialValues: { rows: Array.from({ length: 1000 }, (_, index) => ({ id: `r${index}`, cost: 10 + (index % 7), markup: 20, suggested: 0 })) },
    effects: [effects],
  });
  return (
    <Form form={form}>
      <p>成本 / 加价率 % / 建议价（上标是渲染次数）</p>
      <Table />
    </Form>
  );
}
```

- 修改第 3 行成本时，只有第 3 行的 `cost` 和 `suggested` 上标计数加 1，其余 999 行不重渲染。`<ArrayField>` 只订阅数组结构变更，不订阅单元格值变化。
- `<ArrayField virtualScroll>` 仅挂载视口内的行组件，未挂载行的 `reaction.each` 实例仍在引擎内运行，初始化时全表建议价已完成计算。
- 独立演示页提供 200 行与 1000 行切换，用于对比局部渲染次数。
