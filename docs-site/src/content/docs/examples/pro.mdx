## ArrayTable 全功能

场景：可编辑的明细表，要分页、勾选、拖动排序、调列宽、列配置、带确认的异步删除，可展开的行里再放一张只读子表。

```tsx preview play=array-table
import { Button, Space, message } from 'antd';
import { ArrayField, Field, Form, useForm } from '@shikiform/react';
import { ArrayTable, FormItem, Input, type ArrayTableColumn } from '@shikiform/antd';

type Item = { a1: string; a2: string };
type Row = Item & { id: string; subitems?: Item[] };

const rows: Row[] = Array.from({ length: 23 }, (_, i) => ({
  id: String(i), a1: `${i}.1`, a2: `${i}.2`,
  ...(i % 3 === 1 ? {} : { subitems: [{ a1: `子 ${i}.1`, a2: `子 ${i}.2` }] }),
}));
const cell = [FormItem, { style: { margin: 0 } }] as const; // 单元格：只要错误状态，不要外边距

function itemColumns<T extends Item>(): ArrayTableColumn<T>[] {
  return [
    { key: 'index', title: '#', width: 60, render: () => <ArrayTable.Index /> },
    { key: 'a1', title: 'A1', width: 160, render: () => <Field name="a1" required decorator={cell} component={[Input]} /> },
    { key: 'a2', title: 'A2', width: 160, render: () => <Field name="a2" decorator={cell} component={[Input]} /> },
  ];
}

// 子表是只读的：pattern 属于字段，逐个给 <Field> 标 readPretty
function readonlyItemColumns<T extends Item>(): ArrayTableColumn<T>[] {
  return itemColumns<T>().map((column) => column.key === 'index' ? column : {
    ...column,
    render: () => <Field name={column.key as keyof T & string} pattern="readPretty" decorator={cell} component={[Input]} />,
  });
}

function Toolbar() {
  const table = ArrayTable.useTable();
  return <Button size="small" onClick={() => message.info(`已选 ${table.selectedKeys.join(', ') || '无'}`)}>看看勾选了谁</Button>;
}

function Footer() {
  const table = ArrayTable.useTable();
  return <span>第 {table.page}/{table.pageCount} 页，共 {table.total} 条</span>;
}

export default function App() {
  const form = useForm<{ list: Row[] }>({ initialValues: { list: rows } });
  return (
    <Form form={form}>
      <ArrayField name="list" rowKey="id" component={[ArrayTable, {
        title: '明细',
        rowSelection: true,
        resizeable: true,
        toolbar: <Toolbar />,
        footer: <Footer />,
        extra: <ArrayTable.Addition defaultValue={() => ({ id: crypto.randomUUID(), a1: '', a2: '' })} />,
        onRemove: () => new Promise<void>((resolve) => setTimeout(resolve, 600)),
        expandable: {
          rowExpandable: (row) => Array.isArray(row.subitems),
          render: () => <ArrayField name="subitems" component={[ArrayTable, {
            showHeader: false,
            settings: false,
            columns: readonlyItemColumns<Item>(),
          }]} />,
        },
        columns: [
          { key: 'sort', width: 48, render: () => <ArrayTable.SortHandle /> },
          ...itemColumns<Row>(),
          {
            key: 'actions', title: '操作', width: 160,
            render: () => <Space size={0}><ArrayTable.Remove confirm /><ArrayTable.MoveUp /><ArrayTable.MoveDown /></Space>,
          },
        ],
      }]} />
    </Form>
  );
}
```

- 列的 `render` 运行在行作用域里：`<Field name="a1">` 就是 `list.N.a1`；展开区里的 `<ArrayField name="subitems" component={[ArrayTable, …]}>` 也相对这一行。
- `ArrayTable.useTable()` 在 `toolbar` / `footer` 里读同一张表的分页和勾选；行内用 `ArrayTable.useRow()` 或 headless 的 `useIndex()` / `useRecord()`。
- `onRemove` 返回 Promise：删除按钮先转圈，resolve 才删，reject 保留这一行。
- 表头右侧的齿轮是列配置：显隐、拖动排序、密度。完整版还有「加载 10 万行」。

## 用影子表单编辑行

场景：表格只读，每行「编辑」打开同一个弹窗；另有两个「添加」入口，一个弹窗、一个抽屉。确认才写回，取消什么都不动。

```tsx preview play=array-table-shadow
import { Button } from 'antd';
import { ArrayField, Field, Form, useForm } from '@shikiform/react';
import { useArrayShadow } from '@shikiform/hooks';
import { ArrayTable, FormItem, Input, ShadowModal } from '@shikiform/antd';

type Row = { id: string; domain: string; desc: string };

function RowFields() {
  return (
    <>
      <Field name="id" title="ID" required decorator={[FormItem]} component={[Input]} />
      <Field name="domain" title="域名" required decorator={[FormItem]} component={[Input]} />
      <Field name="desc" title="描述" decorator={[FormItem]} component={[Input.TextArea]} />
    </>
  );
}

function Domains() {
  const editor = useArrayShadow<Row>('list'); // 编辑第 N 行 → 确认后写回第 N 行
  const empty: Row = { id: '', domain: '', desc: '' };
  return (
    <>
      <ArrayField name="list" rowKey="id" component={[ArrayTable, {
        pagination: { pageSize: 5 },
        columns: [
          { key: 'id', title: 'ID', width: 80, render: () => <Field name="id" pattern="readPretty" component={[Input]} /> },
          { key: 'domain', title: '域名', render: () => <Field name="domain" pattern="readPretty" component={[Input]} /> },
          { key: 'edit', title: '操作', width: 90, render: ({ index }) => <Button type="link" onClick={() => void editor.edit(index)}>编辑</Button> },
        ],
        extra: (
          <>
            <ArrayTable.ShadowAddition<Row> title="弹窗添加" initialValues={empty}><RowFields /></ArrayTable.ShadowAddition>
            <ArrayTable.ShadowAddition<Row> title="抽屉添加" popType="drawer" initialValues={empty}><RowFields /></ArrayTable.ShadowAddition>
          </>
        ),
      }]} />
      <ShadowModal shadow={editor} title="编辑"><RowFields /></ShadowModal>
    </>
  );
}

export default function App() {
  const form = useForm<{ list: Row[] }>({
    initialValues: { list: [{ id: '1', domain: 'shikiform.dev', desc: 'ping' }, { id: '2', domain: 'example.com', desc: '' }] },
  });
  return <Form form={form}><Domains /></Form>;
}
```

- `useArrayShadow(name)` 是 `useShadowForm` 加上默认写回：`edit(index)` 拷贝这一行打开草稿，确认后写回同一行；可传 `edit(index, init)` 先映射或请求记录。
- `ArrayTable.ShadowAddition`（Pro.formily `ProAddition`）自带一个影子草稿：确认后追加一行，并翻到新行所在的页。
- 草稿是独立的引擎，字段名从草稿的根开始解析，不继承所在行的作用域。

## 影子表单的几种容器

场景：同一份「域名 + 描述」草稿，分别放进气泡确认框、页面内联区块和纯 Hook 里。

```tsx preview play=shadow-form
import { useState } from 'react';
import { Button, Card, Space } from 'antd';
import { Field, Form, Watch, useForm, type Draft } from '@shikiform/react';
import { useShadowForm } from '@shikiform/hooks';
import { FormItem, Input, ShadowForm, ShadowPopconfirm, useShadow } from '@shikiform/antd';

type Info = { domain: string; desc: string };

const fields = (
  <>
    <Field name="domain" title="域名" required decorator={[FormItem]} component={[Input]} />
    <Field name="desc" title="描述" decorator={[FormItem]} component={[Input]} />
  </>
);

/** 草稿里的子组件用 useShadow() 找到所在的草稿。 */
function SaveButton() {
  const shadow = useShadow<Info>();
  return <Button type="primary" onClick={() => void shadow.confirm()}>保存（内联）</Button>;
}

export default function App() {
  const form = useForm<{ info: Info }>({ initialValues: { info: { domain: 'shikiform.dev', desc: 'ping' } } });
  const [log, setLog] = useState('');
  const save = (label: string) => (values: Info) => {
    form.setValue('info', values);
    setLog(`${label}：${JSON.stringify(values)}`);
  };
  const record = () => form.getValue('info');
  const hook = useShadowForm<Info>({ onOk: save('纯 Hook') });
  return (
    <Form form={form}>
      <Watch select={(d: Draft<{ info: Info }>) => d.info.$value}>{(info) => <p>当前：{info.domain} · {info.desc}</p>}</Watch>
      <ShadowPopconfirm<Info> record={record} onOk={save('气泡')} trigger={<Button>气泡里编辑</Button>}>{fields}</ShadowPopconfirm>
      <Card size="small" title="内联草稿（ShadowForm）" style={{ marginTop: 12 }}>
        <ShadowForm<Info> record={record} onOk={save('内联')}>
          {fields}
          <SaveButton />
        </ShadowForm>
      </Card>
      <Card size="small" title="纯 Hook（useShadowForm）" style={{ marginTop: 12 }}>
        <Space>
          <Button onClick={() => void hook.open(record())}>打开草稿</Button>
          <Button disabled={!hook.visible} onClick={() => void hook.confirm()}>保存</Button>
          <Button disabled={!hook.visible} onClick={hook.cancel}>放弃</Button>
        </Space>
        {hook.visible && <Form form={hook.form}>{fields}</Form>}
      </Card>
      <pre aria-live="polite">{log}</pre>
    </Form>
  );
}
```

- 所有容器都是 `useShadowForm` 的外壳：`open(record)` 拷贝一份到新引擎，`confirm()` 校验通过后调用 `onOk(values)`，`cancel()` 丢弃。
- `ShadowForm` 内联渲染草稿，子组件用 `useShadow()` 拿到它；也可以写成 render prop `{(shadow) => …}`。
- 弹窗和抽屉（`ShadowModal` / `ShadowDrawer`）见 [Pro 组件 · 影子草稿](/guide/pro#影子草稿)。

## QueryList

场景：域名列表页。条件栅格超过两格时折叠，服务端分页，ID 表头排序发给服务端，「分类」列默认隐藏，状态显示成徽标，列表里的域名可以直接改。

```tsx preview play=query-list
import { Field } from '@shikiform/react';
import { useQueryTable } from '@shikiform/hooks';
import { FormItem, Input, QueryList, Select, type ArrayTableColumn } from '@shikiform/antd';

type Row = { id: number; status: number; domain: string; classify: number[] };
type Query = { domain?: string; status?: number };

const STATUS = [
  { label: '待审核', value: 0, color: 'default' }, { label: '运行中', value: 1, color: 'processing' },
  { label: '已上线', value: 2, color: 'success' }, { label: '异常', value: 3, color: 'error' },
];
const CLASSIFY = [{ label: '官网', value: 0 }, { label: '文档', value: 1 }, { label: '接口', value: 2 }];

/** 模拟服务：共 45 条，服务端分页和按 id 排序。 */
async function service(query: Query, page: number, pageSize: number, desc: boolean) {
  await new Promise((resolve) => setTimeout(resolve, 300));
  const all = Array.from({ length: 45 }, (_, i): Row => ({ id: i + 1, status: query.status ?? i % 4, domain: `${query.domain ?? 'site'}${i + 1}.dev`, classify: [i % 3] }));
  if (desc) all.reverse();
  return { data: all.slice((page - 1) * pageSize, page * pageSize), total: all.length };
}

const cell = [FormItem, { style: { margin: 0 } }] as const;
const columns: ArrayTableColumn<Row>[] = [
  { key: 'id', title: 'ID', width: 80, sorter: true, render: ({ record }) => record.id },
  { key: 'status', title: '状态', width: 120, render: () => <Field name="status" pattern="readPretty" decorator={cell} component={[Select, { prettyType: 'badge' }]} /> },
  { key: 'domain', title: '域名', render: () => <Field name="domain" required decorator={cell} component={[Input]} /> },
  { key: 'classify', title: '分类', width: 140, hidden: true, render: () => <Field name="classify" pattern="readPretty" decorator={cell} component={[Select, { mode: 'multiple', prettyType: 'tag' }]} /> },
];

export default function App() {
  const table = useQueryTable({
    initialValues: { query: {} as Query, list: [] as Row[] },
    fields: {
      'query.status': { enum: STATUS },
      'list.*.status': { enum: STATUS },
      'list.*.classify': { enum: CLASSIFY },
    },
    params: 'query',
    rows: 'list', // 每页结果写进 list，单元格就是普通字段
    request: ({ params, page, pageSize, sorter }) =>
      service(params, page, pageSize, sorter.some((item) => item.field === 'id' && item.order === 'descend')),
  });
  return (
    <QueryList table={table} columns={columns}>
      <Field name="query.domain" title="域名" decorator={[FormItem, { layout: 'vertical' }]} component={[Input]} />
      <Field name="query.status" title="状态" decorator={[FormItem, { layout: 'vertical' }]} component={[Select, { allowClear: true }]} />
      <Field name="query.note" title="备注（折叠）" decorator={[FormItem, { layout: 'vertical' }]} component={[Input]} />
    </QueryList>
  );
}
```

- `QueryList` = 可折叠的条件栅格 + 受控分页的 `ArrayTable` + 刷新按钮，状态全部在 `useQueryTable` 里。
- `rows: 'list'` 让每一页结果成为表单数组，所以单元格可以编辑、校验；这类写入不算用户修改（不置脏）。
- 列上写 `sorter: true`，表头排序随请求发给 `request` 的 `sorter` 并回到第 1 页；勾选保留。`hidden: true` 的列在列配置里可以再打开。
- `params: 'query'` 让请求参数的类型就是 `query` 字段的类型。

## QueryList 添加与编辑

场景：在查询结果上直接加一行、或用弹窗改某一行；真实项目里在 `onOk` 里调接口，然后 `table.refresh()`。

```tsx preview play=query-list-add
import { Button } from 'antd';
import { Field } from '@shikiform/react';
import { useArrayShadow, useQueryTable } from '@shikiform/hooks';
import { FormItem, Input, ArrayTable, QueryList, ShadowModal } from '@shikiform/antd';

type Row = { id: string; domain: string };

const DATA: Row[] = Array.from({ length: 12 }, (_, i) => ({ id: String(i + 1), domain: `site${i + 1}.dev` }));

function RowFields() {
  return (
    <>
      <Field name="id" title="ID" required decorator={[FormItem]} component={[Input]} />
      <Field name="domain" title="域名" required decorator={[FormItem]} component={[Input]} />
    </>
  );
}

export default function App() {
  const table = useQueryTable({
    initialValues: { query: { domain: '' }, list: [] as Row[] },
    params: 'query',
    rows: 'list',
    request: async ({ params, page, pageSize }) => {
      const rows = DATA.filter((row) => row.domain.includes(params.domain));
      return { data: rows.slice((page - 1) * pageSize, page * pageSize), total: rows.length };
    },
  });
  // QueryList 自己渲染 <Form>；把 Hook 的 form 交给影子，它就能写回 list。
  const editor = useArrayShadow<Row>('list', { form: table.form, rowKey: 'id' });
  return (
    <>
      <QueryList
        table={table}
        columns={[
          { key: 'id', title: 'ID', width: 80, render: ({ record }) => record.id },
          { key: 'domain', title: '域名', render: ({ record }) => record.domain },
          { key: 'edit', title: '操作', width: 90, render: ({ index }) => <Button type="link" onClick={() => void editor.edit(index)}>编辑</Button> },
        ]}
        extra={<ArrayTable.ShadowAddition<Row> title="添加条目" initialValues={{ id: '', domain: '' }}><RowFields /></ArrayTable.ShadowAddition>}
      >
        <Field name="query.domain" title="域名" decorator={[FormItem]} component={[Input]} />
      </QueryList>
      <ShadowModal shadow={editor} title="编辑"><RowFields /></ShadowModal>
    </>
  );
}
```

- 编辑和添加都只改当前页的 `list`，不会替你调用后端；要持久化就在 `onOk` 里请求，再 `table.refresh()`。
- `useArrayShadow` 不在 `<Form>` 里时，用 `{ form }` 指定写回哪个引擎。

## 枚举：静态、懒加载与阅读态

场景：一组选择类字段，有的选项是静态的，有的要远程加载（多个字段共用一次请求）；整组切到阅读态时，显示成文字、徽标或标签。

```tsx preview play=pro-enum
import { Button, Space } from 'antd';
import { EnumSource, Field, Form, ObjectField, useForm } from '@shikiform/react';
import { Checkbox, FormItem, FormLayout, Radio, Select } from '@shikiform/antd';

const LIST = [{ label: '啊', value: 'a', color: 'blue' }, { label: '哦', value: 'o', color: 'green' }, { label: '呃', value: 'e' }];
/** 懒加载枚举：加载一次，所有用它的字段共享。 */
const remote = EnumSource.from(async () => {
  await new Promise((resolve) => setTimeout(resolve, 800));
  return LIST;
});

type Demo = { select: string; lazySelect: string[]; radio: string; lazyCheckbox: string[]; badge: string[]; tag: string[] };
type Values = { demo: Demo };

export default function App() {
  const form = useForm<Values>({
    initialValues: { demo: { select: 'a', lazySelect: ['o'], radio: 'e', lazyCheckbox: ['a'], badge: ['a', 'o'], tag: ['a', 'e'] } },
  });
  const setPattern = (pattern?: 'readPretty') => form.batch((d) => { d.demo.$pattern = pattern; });
  return (
    <Form form={form}>
      <Space style={{ marginBottom: 12 }}>
        <Button onClick={() => setPattern('readPretty')}>阅读态</Button>
        <Button onClick={() => setPattern()}>编辑态</Button>
      </Space>
      <FormLayout labelCol={{ span: 6 }} wrapperCol={{ span: 18 }}>
        <ObjectField name="demo">
          <Field name="select" title="Select（徽标）" enum={LIST} decorator={[FormItem]} component={[Select, { prettyType: 'badge', style: { width: 200 } }]} />
          <Field name="lazySelect" title="懒加载 Select" enum={remote.with(null)} decorator={[FormItem]}
            component={[Select, { mode: 'multiple', prettyType: 'tag', style: { width: 200 } }]} />
          <Field name="radio" title="Radio" enum={LIST} decorator={[FormItem]} component={[Radio]} />
          <Field name="lazyCheckbox" title="懒加载 Checkbox" enum={remote.with(null)} decorator={[FormItem]} component={[Checkbox.Group]} />
          <Field name="badge" title="徽标（阅读态）" enum={LIST} pattern="readPretty" decorator={[FormItem]}
            component={[Select, { mode: 'multiple', prettyType: 'badge' }]} />
          <Field name="tag" title="标签（阅读态）" enum={LIST} pattern="readPretty" decorator={[FormItem]}
            component={[Select, { mode: 'multiple', prettyType: 'tag' }]} />
        </ObjectField>
      </FormLayout>
    </Form>
  );
}
```

- 选项写在 `<Field enum>` 上：静态数组，或远程的 `EnumSource.from(loader).with(params)`（同一个 source 加载一次，所有字段共享）。随别的字段变化的参数写在 reaction 里：`d.city.$enum = Cities.with(d.province.$value)`，参数变了会重新加载并作废旧请求。
- 选择类控件的 `options` 默认取字段枚举，加载中自动 `loading`；阅读态显示选项的 `label`（多个值是标签），`prettyType` 换成 `badge` 或彩色 `tag`（颜色取选项的 `color`）。
- 只显示、不编辑的字段（表格里的状态列）就是 `pattern="readPretty"` 的选择控件；不在表单里用纯视图 `<PreviewText.Select value options prettyType />`。

## Suggestion：远程联想

场景：输入关键字后联想商品名；值可以是字符串、`{ label, value }`，也可以多选。

```tsx preview play=suggestion
import { Field, Form, Watch, useForm, type Draft } from '@shikiform/react';
import { FormItem, FormLayout, Suggestion } from '@shikiform/antd';

const WORDS = ['手机壳', '手机支架', '手表', '书包', '鼠标', '鼠标垫', '水杯', '耳机', '咖啡机', '咖啡豆'];
/** 模拟远程接口：300ms 后返回匹配的词；新的输入会 abort 旧请求。 */
async function suggest(keyword: string, ctx?: { signal?: AbortSignal }) {
  await new Promise((resolve) => setTimeout(resolve, 300));
  if (ctx?.signal?.aborted) return [];
  return WORDS.filter((word) => [...keyword].some((char) => word.includes(char))).map((word) => ({ label: word, value: word }));
}

type Option = { label: string; value: string };
type Values = { text?: string; labeled?: Option; many?: string[]; manyLabeled?: Option[] };

export default function App() {
  const form = useForm<Values>({ initialValues: {} });
  return (
    <Form form={form}>
      <FormLayout labelCol={{ span: 8 }} wrapperCol={{ span: 16 }}>
        <Field name="text" title="string" decorator={[FormItem]} component={[Suggestion, { fetch: suggest, placeholder: '输入「手」' }]} />
        <Field name="labeled" title="labelInValue" decorator={[FormItem]} component={[Suggestion, { fetch: suggest, labelInValue: true }]} />
        <Field name="many" title="multiple" decorator={[FormItem]} component={[Suggestion, { fetch: suggest, multiple: true }]} />
        <Field name="manyLabeled" title="multiple + labelInValue" decorator={[FormItem]} component={[Suggestion, { fetch: suggest, multiple: true, labelInValue: true }]} />
      </FormLayout>
      <Watch select={(d: Draft<Values>) => d.$value}>{(values) => <pre>{JSON.stringify(values, null, 2)}</pre>}</Watch>
    </Form>
  );
}
```

- `fetch(keyword, { signal })` 返回选项；组件负责防抖（`debounce`，默认 300ms）、中止旧请求、丢弃迟到的结果。
- 无 UI 的同一套逻辑是 `useSuggestion`（`@shikiform/hooks`），不绑字段的组件是 `SuggestionView`。

## CascaderPlus：懒加载级联

场景：省 / 市 / 区三级地址，每一级都按需加载；表单里已有的值打开时要逐级反显，阅读态显示成「北京市 / 北京市 / 朝阳区」。

```tsx preview play=cascader-plus
import { Button, Space } from 'antd';
import { Field, Form, Watch, useForm, type Draft } from '@shikiform/react';
import type { CascaderOption } from '@shikiform/hooks';
import { CascaderPlus, FormItem, FormLayout } from '@shikiform/antd';

type Area = { code: string; name: string; children?: Area[] };
const AREA: Area[] = [
  { code: '110000', name: '北京市', children: [{ code: '11000000', name: '北京市', children: [{ code: '110105', name: '朝阳区' }, { code: '110108', name: '海淀区' }] }] },
  { code: '330000', name: '浙江省', children: [{ code: '330100', name: '杭州市', children: [{ code: '330106', name: '西湖区' }, { code: '330108', name: '滨江区' }] }] },
];
/** 一个 loadData(path) 服务所有层级：path 是已选的值，返回下一级。 */
async function loadArea(path: Array<string | number>): Promise<CascaderOption[]> {
  await new Promise((resolve) => setTimeout(resolve, 400));
  let level: Area[] | undefined = AREA;
  for (const code of path) level = level?.find((node) => node.code === String(code))?.children;
  return (level ?? []).map((node) => ({ label: node.name, value: node.code, isLeaf: path.length === 2 }));
}

type Values = { area: string[]; labeled: { label: string; value: string }[] };

export default function App() {
  const form = useForm<Values>({
    initialValues: {
      area: ['110000', '11000000', '110105'], // 只有编码：打开时逐级加载出标签
      labeled: [{ label: '浙江省', value: '330000' }, { label: '杭州市', value: '330100' }, { label: '西湖区', value: '330106' }],
    },
  });
  const setPattern = (pattern?: 'readPretty') => form.batch((d) => { d.area.$pattern = pattern; d.labeled.$pattern = pattern; });
  return (
    <Form form={form}>
      <Space style={{ marginBottom: 12 }}>
        <Button onClick={() => setPattern('readPretty')}>阅读态</Button>
        <Button onClick={() => setPattern()}>编辑态</Button>
      </Space>
      <FormLayout labelCol={{ span: 6 }} wrapperCol={{ span: 18 }}>
        <Field name="area" title="编码" decorator={[FormItem]} component={[CascaderPlus, { loadData: loadArea, style: { width: 280 } }]} />
        <Field name="labeled" title="labelInValue" decorator={[FormItem]} component={[CascaderPlus, { loadData: loadArea, labelInValue: true, style: { width: 280 } }]} />
      </FormLayout>
      <Watch select={(d: Draft<Values>) => d.$value}>{(values) => <pre>{JSON.stringify(values)}</pre>}</Watch>
    </Form>
  );
}
```

- `loadData(path)` 收到已选值组成的路径（根是 `[]`），返回下一级选项；最后一级标 `isLeaf: true`。
- 只有编码的值会通过同一个 `loadData` 解析出标签，编辑态和阅读态都能显示文字；`labelInValue` 的值自带标签。
- 选项一开始就是整棵树时不需要它：把树给 `<Field enum>`，用 [`Cascader`](/guide/controls)。不绑字段的组件是 `CascaderPlusView`，无 UI 的逻辑是 `useLazyCascader` / `useCascaderLabels`。
