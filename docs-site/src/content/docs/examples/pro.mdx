## useArrayTable：分页、勾选、排序与展开

**场景**：可编辑的明细表格，需要支持分页、行勾选、上下移排序、带确认提示的异步删除，以及展开行展示只读子表。

```tsx preview play=array-table
import { Fragment, useState } from 'react';
import { Field, Form, ObjectField, useArrayTable, useForm } from '@shikiform/react';
import {
  Badge,
  Button,
  Card,
  Checkbox,
  ChevronLeftIcon,
  ChevronRightIcon,
  Input,
  PlusIcon,
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
  Toolbar,
} from '@site/ui';

type Item = { a1: string; a2: string };
type Row = Item & { id: string; subitems?: Item[] };

const INITIAL_ROWS: Row[] = Array.from({ length: 12 }, (_, i) => ({
  id: String(i),
  a1: `${i}.1`,
  a2: `${i}.2`,
  ...(i % 3 === 1 ? {} : { subitems: [{ a1: `子 ${i}.1`, a2: `子 ${i}.2` }] }),
}));

let nextId = 100;

function TableDemo() {
  const [notice, setNotice] = useState('');
  const table = useArrayTable<Row>('list', {
    rowKey: 'id',
    pagination: { pageSize: 5 },
    rowExpandable: (row) => Array.isArray(row.subitems),
    onRemove: () => new Promise<void>((resolve) => setTimeout(resolve, 400)),
  });
  return (
    <Card title="可编辑明细表" description="支持分页、行勾选、上下排序、异步删除与只读子表展开">
      <div style={{ display: 'grid', gap: 12 }}>
        <Toolbar>
          <Button variant="primary" size="sm" icon={<PlusIcon size={13} />} onClick={() => table.add({ id: String(nextId++), a1: '', a2: '' })}>
            添加一行
          </Button>
          <Button variant="outline" size="sm" onClick={() => setNotice(`已选 ${table.selectedKeys.join(', ') || '无'}`)}>
            看看勾选了谁
          </Button>
          {notice && <Badge variant="brand">{notice}</Badge>}
        </Toolbar>
        <Table compact>
          <TableHead>
            <TableRow>
              <TableHeader style={{ width: 36 }} />
              <TableHeader style={{ width: 38 }} />
              <TableHeader style={{ width: 44 }}>#</TableHeader>
              <TableHeader>A1</TableHeader>
              <TableHeader>A2</TableHeader>
              <TableHeader>操作</TableHeader>
            </TableRow>
          </TableHead>
          <TableBody>
            {table.pageRows.map((row) => {
              const expanded = table.expandedKeys.includes(row.key);
              const expandable = table.isExpandable(row.index);
              const removing = table.removing.includes(row.key);
              const record = table.record(row.index);
              return (
                <Fragment key={row.key}>
                  <ObjectField name={`list.${row.index}`}>
                    <TableRow>
                      <TableCell>
                        <Checkbox
                          aria-label={`选择 ${row.key}`}
                          checked={table.selectedKeys.includes(row.key)}
                          onCheckedChange={() => table.toggleSelected(row.key)}
                        />
                      </TableCell>
                      <TableCell>
                        {expandable && (
                          <Button variant="ghost" size="sm" onClick={() => table.toggleExpanded(row.key)}>
                            {expanded ? '▾' : '▸'}
                          </Button>
                        )}
                      </TableCell>
                      <TableCell><Badge variant="outline">{row.position}</Badge></TableCell>
                      <TableCell>
                        <Field<Row, 'a1'> name="a1" required>
                          {(f) => (
                            <div style={{ display: 'flex', flexDirection: 'column', gap: 2 }}>
                              <Input
                                size="sm"
                                style={{ width: 104 }}
                                invalid={Boolean(f.errors[0])}
                                value={f.value ?? ''}
                                onChange={(e) => f.onChange(e.target.value)}
                                onBlur={f.onBlur}
                              />
                              {f.errors[0] && <small role="alert" className="sf-ui-form-error">{f.errors[0]}</small>}
                            </div>
                          )}
                        </Field>
                      </TableCell>
                      <TableCell>
                        <Field<Row, 'a2'> name="a2">
                          {(f) => <Input size="sm" style={{ width: 104 }} value={f.value ?? ''} onChange={(e) => f.onChange(e.target.value)} />}
                        </Field>
                      </TableCell>
                      <TableCell>
                        <div style={{ display: 'inline-flex', gap: 4 }}>
                          <Button variant="ghost" size="sm" disabled={row.index === 0} onClick={() => table.moveUp(row.index)}>上移</Button>
                          <Button variant="ghost" size="sm" disabled={row.index >= table.total - 1} onClick={() => table.moveDown(row.index)}>下移</Button>
                          <Button variant="danger" size="sm" loading={removing} onClick={() => void table.remove(row.index)}>
                            {removing ? '删除中…' : '删除'}
                          </Button>
                        </div>
                      </TableCell>
                    </TableRow>
                  </ObjectField>
                  {expanded && record.subitems && (
                    <TableRow style={{ background: 'var(--sf-ui-muted)' }}>
                      <TableCell colSpan={6} style={{ paddingLeft: 44 }}>
                        {record.subitems.map((sub, i) => (
                          <div key={i} style={{ display: 'flex', gap: 8, alignItems: 'center' }}>
                            <Badge variant="outline">子项 #{i + 1}</Badge>
                            <small>{sub.a1} / {sub.a2}</small>
                          </div>
                        ))}
                      </TableCell>
                    </TableRow>
                  )}
                </Fragment>
              );
            })}
          </TableBody>
        </Table>
        <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', justifyContent: 'space-between', gap: 10 }}>
          <span style={{ fontSize: 12, color: 'var(--sf-ui-fg-muted)' }}>
            第 {table.page}/{table.pageCount} 页，共 {table.total} 条
          </span>
          <div style={{ display: 'inline-flex', gap: 6 }}>
            <Button variant="outline" size="sm" disabled={table.page <= 1} icon={<ChevronLeftIcon size={13} />} onClick={() => table.setPage(table.page - 1)}>
              上一页
            </Button>
            <Button variant="outline" size="sm" disabled={table.page >= table.pageCount} onClick={() => table.setPage(table.page + 1)}>
              下一页
              <ChevronRightIcon size={13} />
            </Button>
          </div>
        </div>
      </div>
    </Card>
  );
}

export default function App() {
  const form = useForm<{ list: Row[] }>({ initialValues: { list: INITIAL_ROWS } });
  return (
    <Form form={form}>
      <TableDemo />
    </Form>
  );
}
```

- `useArrayTable('list', options)` 托管分页切片（`pageRows`）、勾选（`selectedKeys`）、展开（`expandedKeys`）、上下移（`moveUp` / `moveDown`）与异步删除（`onRemove` / `removing`），不绑定任何 UI 组件库。
- 每行包一层 ``<ObjectField name={`list.${row.index}`}>`` 后，单元格内的 `<Field name="a1">` 直接按相对路径绑定。
- 当 `onRemove` 返回 Promise 时，`table.removing` 包含当前行 key，Promise resolve 后才会移除该行，reject 则保留原行。

## 用草稿表单编辑与新增行

**场景**：表格本身设为只读，点击每行的「编辑」或顶部「添加」时打开草稿表单，确认校验通过后才写回列表，取消则保持原数据不变。

```tsx preview play=array-table-shadow
import type { ReactNode } from 'react';
import { Field, Form, useArrayTable, useFieldDecorator, useForm, useFormContext, useShadowForm } from '@shikiform/react';
import {
  Badge,
  Button,
  Card,
  FieldInput,
  PlusIcon,
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
  Toolbar,
} from '@site/ui';

type Row = { id: string; domain: string; desc: string };

function FieldRow({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return (
    <p className="sf-ui-form-item" style={{ margin: 0 }}>
      <span className="sf-ui-form-label">
        <label {...d.labelProps}>{d.label}：</label>
        {d.required && <span className="sf-ui-form-required" aria-hidden="true">*</span>}
      </span>
      {children}
      {d.errorProps && <small {...d.errorProps} className="sf-ui-form-error">{d.errors[0]}</small>}
    </p>
  );
}

function RowFields() {
  return (
    <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(160px, 1fr))', gap: 12 }}>
      <Field name="id" decorator={[FieldRow]} component={[FieldInput]} />
      <Field name="domain" decorator={[FieldRow]} component={[FieldInput]} />
      <Field name="desc" decorator={[FieldRow]} component={[FieldInput]} />
    </div>
  );
}

function EditableTable() {
  const form = useFormContext<{ list: Row[] }>();
  const table = useArrayTable<Row>('list', { rowKey: 'id', pagination: { pageSize: 5 } });
  const adder = useShadowForm<Row>({ name: 'list', add: { id: '', domain: '', desc: '' } });
  const editor = useShadowForm<Row, string>({
    fields: {
      id: { title: 'ID', required: true },
      domain: { title: '域名', required: true },
      desc: { title: '描述' },
    },
    onOk: (values, rowId) => {
      const list = form.getValue('list');
      const idx = list.findIndex((item) => item.id === rowId);
      if (idx >= 0) form.setValue(`list.${idx}`, values);
    },
  });
  return (
    <Card
      title="域名列表（草稿表单编辑）"
      description="新增与编辑都在独立草稿引擎里完成，确认校验通过才写回宿主"
      extra={
        <Button variant="primary" size="sm" icon={<PlusIcon size={13} />} onClick={() => void adder.open()}>
          添加一行
        </Button>
      }
    >
      <div style={{ display: 'grid', gap: 14 }}>
        <Table compact striped>
          <TableHead>
            <TableRow>
              <TableHeader style={{ width: 64 }}>ID</TableHeader>
              <TableHeader>域名</TableHeader>
              <TableHeader>描述</TableHeader>
              <TableHeader style={{ width: 92 }}>操作</TableHeader>
            </TableRow>
          </TableHead>
          <TableBody>
            {table.pageRows.map((row) => {
              const record = table.record(row.index);
              return (
                <TableRow key={row.key}>
                  <TableCell><Badge variant="outline">{record.id}</Badge></TableCell>
                  <TableCell><code>{record.domain}</code></TableCell>
                  <TableCell>{record.desc || '—'}</TableCell>
                  <TableCell>
                    <Button variant="outline" size="sm" onClick={() => void editor.open(record, record.id)}>编辑</Button>
                  </TableCell>
                </TableRow>
              );
            })}
          </TableBody>
        </Table>
        {adder.visible && (
          <fieldset className="sf-ui-card sf-ui-card--muted" style={{ padding: 16, margin: 0 }}>
            <legend style={{ padding: '0 6px', fontSize: 12, fontWeight: 600, color: 'var(--sf-ui-fg-muted)' }}>新增行</legend>
            <Form form={adder.form}>
              <RowFields />
              <Toolbar>
                <Button variant="primary" size="sm" onClick={() => void adder.confirm()}>确定</Button>
                <Button variant="outline" size="sm" onClick={adder.cancel}>取消</Button>
              </Toolbar>
            </Form>
          </fieldset>
        )}
        {editor.visible && (
          <fieldset className="sf-ui-card sf-ui-card--muted" style={{ padding: 16, margin: 0 }}>
            <legend style={{ padding: '0 6px', fontSize: 12, fontWeight: 600, color: 'var(--sf-ui-fg-muted)' }}>编辑行</legend>
            <Form form={editor.form}>
              <RowFields />
              <Toolbar>
                <Button variant="primary" size="sm" onClick={() => void editor.confirm()}>确定</Button>
                <Button variant="outline" size="sm" onClick={editor.cancel}>取消</Button>
              </Toolbar>
            </Form>
          </fieldset>
        )}
      </div>
    </Card>
  );
}

export default function App() {
  const form = useForm<{ list: Row[] }>({
    initialValues: { list: [{ id: '1', domain: 'shikiform.dev', desc: 'ping' }, { id: '2', domain: 'example.com', desc: '' }] },
    fields: {
      'list.*.id': { title: 'ID', required: true },
      'list.*.domain': { title: '域名', required: true },
      'list.*.desc': { title: '描述' },
    },
  });
  return (
    <Form form={form}>
      <EditableTable />
    </Form>
  );
}
```

- `useShadowForm({ name: 'list', add: template })` 会继承宿主 `list.*` 的字段配置（如 `title`、`required`），确认通过后自动向数组追加新行。
- 编辑草稿在 `open(record, record.id)` 时传入行 `id` 作为上下文，确认回调 `onOk` 按 `id` 定位并写回原行，即使编辑期间列表顺序变化也能准确更新。
- 由于草稿内部是独立的表单引擎，其中的 `<Field>` 路径从草稿根路径解析，不会污染宿主表单的 `dirty` 与错误状态。

## 草稿表单（useShadowForm）

**场景**：对宿主表单的 `info` 对象开启一份独立草稿：打开时拷贝数据到新引擎，确认校验通过后写回宿主，放弃则直接丢弃。

```tsx preview play=shadow-form
import { useState, type ReactNode } from 'react';
import { Field, Form, Watch, useFieldDecorator, useForm, useShadowForm, type Draft } from '@shikiform/react';
import { Alert, Button, Card, FieldInput, Toolbar } from '@site/ui';

type Info = { domain: string; desc: string };

function FieldRow({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return (
    <p className="sf-ui-form-item" style={{ margin: 0 }}>
      <span className="sf-ui-form-label">
        <label {...d.labelProps}>{d.label}：</label>
        {d.required && <span className="sf-ui-form-required" aria-hidden="true">*</span>}
      </span>
      {children}
      {d.errorProps && <small {...d.errorProps} className="sf-ui-form-error">{d.errors[0]}</small>}
    </p>
  );
}

export default function App() {
  const form = useForm<{ info: Info }>({ initialValues: { info: { domain: 'shikiform.dev', desc: 'ping' } } });
  const [log, setLog] = useState('');
  const shadow = useShadowForm<Info>({
    form,
    name: 'info',
    onOk: (values) => setLog(`已保存：${JSON.stringify(values)}`),
  });
  return (
    <Card title="独立草稿表单" description="草稿拥有独立引擎，放弃即丢弃、保存校验通过后写回宿主对象" style={{ maxWidth: 520 }}>
      <Form form={form}>
        <Alert variant="info">
          <Watch select={(d: Draft<{ info: Info }>) => d.info.$value}>
            {(info) => <p style={{ margin: 0 }}>当前宿主值：{info.domain} · {info.desc}</p>}
          </Watch>
        </Alert>
        <Toolbar>
          <Button variant="outline" size="sm" onClick={() => void shadow.open()}>打开草稿</Button>
          <Button variant="primary" size="sm" disabled={!shadow.visible} onClick={() => void shadow.confirm()}>保存</Button>
          <Button variant="ghost" size="sm" disabled={!shadow.visible} onClick={shadow.cancel}>放弃</Button>
        </Toolbar>
        {shadow.visible && (
          <fieldset className="sf-ui-card sf-ui-card--muted" style={{ padding: 16, margin: 0 }}>
            <legend style={{ padding: '0 6px', fontSize: 12, fontWeight: 600, color: 'var(--sf-ui-fg-muted)' }}>草稿表单</legend>
            <Form form={shadow.form}>
              <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(180px, 1fr))', gap: 12 }}>
                <Field name="domain" title="域名" required decorator={[FieldRow]} component={[FieldInput]} />
                <Field name="desc" title="描述" decorator={[FieldRow]} component={[FieldInput]} />
              </div>
            </Form>
          </fieldset>
        )}
        <pre aria-live="polite">{log}</pre>
      </Form>
    </Card>
  );
}
```

- 每次 `open()` 都会创建一份干净的表单引擎实例，`reset`、`dirty`、错误与联动都从零开始。
- `confirm()` 先执行校验，通过后调用 `onOk(values)` 并写回宿主 `info`（在 `onOk` 中返回 `false` 可跳过默认写回）；`cancel()` 直接关闭并丢弃草稿。

## useQueryTable：服务端分页、排序与行内编辑

**场景**：域名管理列表页。支持服务端分页与按 ID 排序、状态徽标展示，以及在表格内直接编辑域名字段。

```tsx preview play=query-list
import { Field, Form, enumItems, useQueryTable } from '@shikiform/react';
import {
  Badge,
  Button,
  Card,
  ChevronLeftIcon,
  ChevronRightIcon,
  ChevronsUpDownIcon,
  Input,
  SearchIcon,
  Select,
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
  Tag,
  Toolbar,
} from '@site/ui';

type Row = { id: number; status: number; domain: string; classify: number[] };
type Query = { domain?: string; status?: number };
type Values = { query: Query; list: Row[] };

const STATUS = [
  { label: '待审核', value: 0, color: 'default' },
  { label: '运行中', value: 1, color: 'processing' },
  { label: '已上线', value: 2, color: 'success' },
  { label: '异常', value: 3, color: 'error' },
];
const STATUS_VARIANT = { 0: 'default', 1: 'brand', 2: 'success', 3: 'danger' } as const;
const CLASSIFY = [{ label: '官网', value: 0 }, { label: '文档', value: 1 }, { label: '接口', value: 2 }];

/** 模拟服务：共 25 条，服务端分页和按 id 排序。 */
async function service(query: Query, page: number, pageSize: number, desc: boolean) {
  await new Promise((resolve) => setTimeout(resolve, 200));
  const all = Array.from({ length: 25 }, (_, i): Row => ({
    id: i + 1,
    status: query.status ?? i % 4,
    domain: `${query.domain ?? 'site'}${i + 1}.dev`,
    classify: [i % 3],
  }));
  if (desc) all.reverse();
  return { data: all.slice((page - 1) * pageSize, page * pageSize), total: all.length };
}

export default function App() {
  const table = useQueryTable<Values>()({
    initialValues: { query: {}, list: [] },
    fields: {
      'query.status': { enum: STATUS },
      'list.*.status': { enum: STATUS },
      'list.*.classify': { enum: CLASSIFY },
    },
    defaultPageSize: 5,
    params: 'query',
    rows: 'list', // 每页结果写进 list，单元格就是普通字段
    request: ({ params, page, pageSize, sorter }) =>
      service(params, page, pageSize, sorter.some((item) => item.field === 'id' && item.order === 'descend')),
  });
  const desc = table.sortOrderOf('id') === 'descend';
  const pages = Math.max(1, Math.ceil(table.total / table.pageSize));
  return (
    <Card title="域名管理列表" description="服务端分页、按 ID 排序、状态徽标与行内直接编辑">
      <Form form={table.form} onSubmit={() => table.submit()}>
        <Toolbar>
          <Field name="query.domain">
            {(f) => (
              <Input
                size="sm"
                placeholder="域名关键字"
                style={{ width: 160 }}
                value={String(f.value ?? '')}
                onChange={(e) => f.onChange(e.target.value)}
              />
            )}
          </Field>
          <Field name="query.status">
            {(f) => (
              <Select
                size="sm"
                allowClear
                placeholder="全部状态"
                style={{ width: 140 }}
                options={f.options}
                value={f.value ?? null}
                onChange={(next) => f.onChange(next)}
              />
            )}
          </Field>
          <Button type="submit" variant="primary" size="sm" icon={<SearchIcon size={13} />}>查询</Button>
          <Button variant="outline" size="sm" onClick={table.reset}>重置</Button>
          <Button
            variant="outline"
            size="sm"
            icon={<ChevronsUpDownIcon size={13} />}
            onClick={() => table.setSorter([{ field: 'id', order: desc ? 'ascend' : 'descend' }])}
          >
            ID {desc ? '降序 ↓' : '升序 ↑'}
          </Button>
        </Toolbar>
        <Table compact striped>
          <TableHead>
            <TableRow>
              <TableHeader style={{ width: 56 }}>ID</TableHeader>
              <TableHeader style={{ width: 108 }}>状态</TableHeader>
              <TableHeader>域名（可编辑）</TableHeader>
              <TableHeader style={{ width: 96 }}>分类</TableHeader>
            </TableRow>
          </TableHead>
          <TableBody>
            {table.data.map((row, index) => {
              const statusItem = enumItems(row.status, STATUS)[0];
              const tags = enumItems(row.classify, CLASSIFY, { colorful: true });
              const badgeVariant = STATUS_VARIANT[row.status as keyof typeof STATUS_VARIANT] ?? 'default';
              return (
                <TableRow key={row.id}>
                  <TableCell>{row.id}</TableCell>
                  <TableCell>
                    <Badge variant={badgeVariant} dot>{statusItem?.label}</Badge>
                  </TableCell>
                  <TableCell>
                    <Field name={`list.${index}.domain`} required>
                      {(f) => (
                        <div style={{ display: 'flex', flexDirection: 'column', gap: 2 }}>
                          <Input
                            size="sm"
                            style={{ width: 170 }}
                            invalid={Boolean(f.errors[0])}
                            value={String(f.value ?? '')}
                            onChange={(e) => f.onChange(e.target.value)}
                            onBlur={f.onBlur}
                          />
                          {f.errors[0] && <small role="alert" className="sf-ui-form-error">{f.errors[0]}</small>}
                        </div>
                      )}
                    </Field>
                  </TableCell>
                  <TableCell>
                    {tags.map((tag) => (
                      <Tag key={String(tag.value)} variant="outline" color={tag.color}>
                        {tag.label}
                      </Tag>
                    ))}
                  </TableCell>
                </TableRow>
              );
            })}
          </TableBody>
        </Table>
        <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', justifyContent: 'space-between', gap: 10 }}>
          <span style={{ fontSize: 12, color: 'var(--sf-ui-fg-muted)' }}>
            第 {table.page}/{pages} 页 · 共 {table.total} 条 {table.loading ? '· 加载中' : ''}
          </span>
          <div style={{ display: 'inline-flex', gap: 6 }}>
            <Button variant="outline" size="sm" disabled={table.page <= 1} icon={<ChevronLeftIcon size={13} />} onClick={() => table.onPageChange(table.page - 1)}>
              上一页
            </Button>
            <Button variant="outline" size="sm" disabled={table.page >= pages} onClick={() => table.onPageChange(table.page + 1)}>
              下一页
              <ChevronRightIcon size={13} />
            </Button>
          </div>
        </div>
      </Form>
    </Card>
  );
}
```

- 指定 `rows: 'list'` 后，接口返回的当页数据会写入表单的 `list` 数组字段，单元格中的 `<Field>` 可以直接编辑和校验，且引擎填充查询结果不会把表单标记为 `dirty`。
- 调用 `table.setSorter` 会自动携带 `sorter` 参数重新请求并回到第 1 页；声明 `params: 'query'` 会自动推导请求中 `params` 的 TypeScript 类型。

## 枚举：静态、懒加载与 enumItems

**场景**：演示静态枚举与远程懒加载枚举（多个字段共享同一次接口请求）的声明方式，以及使用 `enumItems` 在阅读态下翻译为标签与颜色。

```tsx preview play=pro-enum
import { EnumSource, Field, Form, ObjectField, enumItems, useField, useForm } from '@shikiform/react';
import { Badge, Button, Card, Checkbox, LoaderIcon, Tag } from '@site/ui';

const LIST = [
  { label: '啊', value: 'a', color: '#0f59a4' },
  { label: '哦', value: 'o', color: '#1a6840' },
  { label: '呃', value: 'e' },
];

/** 懒加载枚举：加载一次，所有用它的字段共享。 */
const remote = EnumSource.from(async () => {
  await new Promise((resolve) => setTimeout(resolve, 500));
  return LIST;
});

type Demo = { select: string; lazySelect: string[]; radio: string };
type Values = { demo: Demo };

function SingleSelect() {
  const f = useField<string>();
  const items = enumItems(f.value, f.options);
  return (
    <p className="sf-ui-form-item sf-ui-form-item--inline" style={{ margin: 0 }}>
      <label htmlFor={f.id} className="sf-ui-form-label" style={{ minWidth: 88 }}>{f.title}：</label>
      {f.readOnly ? (
        <span className="sf-ui-readonly-group">
          {items.map((item) => (
            <Badge key={String(item.value)} dot color={item.color}>
              {item.label}
            </Badge>
          ))}
        </span>
      ) : (
        <select id={f.id} value={f.value ?? ''} onChange={(e) => f.onChange(e.target.value)}>
          {f.options.map((opt) => <option key={String(opt.value)} value={String(opt.value)}>{opt.label}</option>)}
        </select>
      )}
    </p>
  );
}

function MultiChecks() {
  const f = useField<string[]>();
  const items = enumItems(f.value, f.options, { colorful: true });
  return (
    <p className="sf-ui-form-item sf-ui-form-item--inline" style={{ margin: 0 }}>
      <span className="sf-ui-form-label" style={{ minWidth: 88 }}>{f.title}：</span>
      {f.loading ? (
        <Badge variant="outline"><LoaderIcon size={12} /> 加载中…</Badge>
      ) : f.readOnly ? (
        <span className="sf-ui-readonly-group">
          {items.map((item) => (
            <Tag key={String(item.value)} variant="outline" color={item.color}>
              {item.label}
            </Tag>
          ))}
        </span>
      ) : (
        <span style={{ display: 'inline-flex', flexWrap: 'wrap', gap: 12 }}>
          {f.options.map((opt) => {
            const val = String(opt.value);
            const checked = (f.value ?? []).includes(val);
            return (
              <Checkbox
                key={val}
                checked={checked}
                onCheckedChange={(on) => f.onChange(on ? [...(f.value ?? []), val] : (f.value ?? []).filter((v) => v !== val))}
              >
                {opt.label}
              </Checkbox>
            );
          })}
        </span>
      )}
    </p>
  );
}

export default function App() {
  const form = useForm<Values>({
    initialValues: { demo: { select: 'a', lazySelect: ['o'], radio: 'e' } },
  });
  const setReadOnly = (readOnly?: true) => form.batch((d) => { d.demo.$readOnly = readOnly; });
  return (
    <Card
      title="枚举源与阅读态翻译"
      description="多个字段共享同一个 EnumSource，切至阅读态后通过 enumItems 自动翻译为徽标与标签"
      extra={
        <div style={{ display: 'inline-flex', gap: 6 }}>
          <Button variant="outline" size="sm" onClick={() => setReadOnly(true)}>阅读态</Button>
          <Button variant="primary" size="sm" onClick={() => setReadOnly()}>编辑态</Button>
        </div>
      }
      style={{ maxWidth: 520 }}
    >
      <Form form={form}>
        <ObjectField name="demo">
          <div style={{ display: 'grid', gap: 12 }}>
            <Field name="select" title="静态枚举" enum={LIST} component={[SingleSelect]} />
            <Field name="lazySelect" title="懒加载多选" enum={remote.with(null)} component={[MultiChecks]} />
            <Field name="radio" title="共享懒加载" enum={remote.with(null)} component={[SingleSelect]} />
          </div>
        </ObjectField>
      </Form>
    </Card>
  );
}
```

- 通过 `EnumSource.from(loader).with(params)` 定义的远程枚举会在相同参数下共享加载结果，加载期间字段的 `loading` 为 `true`；若参数依赖其他字段，在 `reaction` 中赋值 `d.city.$enum = Cities.with(d.province.$value)` 即可自动重拉并作废过期请求。
- `enumItems(value, options, { colorful })`（`@shikiform/react` / `@shikiform/core`）将枚举值解析为 `{ label, value, color, matched }`，便于在阅读态渲染徽标或标签。

## useSuggestion：远程联想

**场景**：输入关键字实时联想商品名称，内置防抖、中止过期请求，并记住已选项的显示文案。

```tsx preview play=suggestion
import { Field, Form, Watch, useField, useForm, useSuggestion, type Draft } from '@shikiform/react';
import { Badge, Button, Card, FormItem, Input, LoaderIcon, SearchIcon } from '@site/ui';

const WORDS = ['手机壳', '手机支架', '手表', '书包', '鼠标', '鼠标垫', '水杯', '耳机', '咖啡机', '咖啡豆'];

/** 模拟远程接口：250ms 后返回匹配的词；新的输入会 abort 旧请求。 */
async function suggest(keyword: string, ctx: { signal: AbortSignal }) {
  await new Promise((resolve) => setTimeout(resolve, 250));
  if (ctx.signal.aborted) return [];
  return WORDS.filter((word) => [...keyword].some((char) => word.includes(char))).map((word) => ({ label: word, value: word }));
}

type Values = { product: string };

function SuggestInput() {
  const f = useField<string>();
  const sug = useSuggestion({ fetch: suggest, debounce: 250 });
  return (
    <div style={{ display: 'grid', gap: 8 }}>
      <Input
        id={f.id}
        prefix={<SearchIcon size={14} />}
        suffix={sug.loading ? <LoaderIcon size={13} /> : undefined}
        placeholder="输入「手」或「鼠」"
        value={sug.keyword || f.value || ''}
        onChange={(e) => {
          sug.search(e.target.value);
          if (!e.target.value) f.onChange('');
        }}
      />
      {sug.loading && <Badge variant="outline" style={{ justifySelf: 'start' }}>搜索中…</Badge>}
      {sug.options.length > 0 && (
        <ul style={{ listStyle: 'none', padding: 0, margin: 0, display: 'flex', flexWrap: 'wrap', gap: 6 }}>
          {sug.options.map((opt) => (
            <li key={String(opt.value)}>
              <Button
                variant="outline"
                size="sm"
                onClick={() => {
                  sug.remember([opt]);
                  f.onChange(String(opt.value));
                  sug.search('');
                }}
              >
                {opt.label}
              </Button>
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}

export default function App() {
  const form = useForm<Values>({ initialValues: { product: '' } });
  return (
    <Card title="远程商品联想" description="内置防抖与旧请求中止，点击候选项写回表单值" style={{ maxWidth: 480 }}>
      <Form form={form}>
        <Field name="product" title="商品联想" decorator={[FormItem]} component={[SuggestInput]} />
        <Watch select={(d: Draft<Values>) => d.$value}>
          {(values) => <pre>{JSON.stringify(values, null, 2)}</pre>}
        </Watch>
      </Form>
    </Card>
  );
}
```

- `useSuggestion({ fetch, debounce })` 内置请求防抖、中止旧请求和丢弃过期响应的能力，`remember(options)` 保留已选项的 `label` 供后续反显。

## useLazyCascader：懒加载级联

**场景**：省 / 市 / 区三级地址按需逐级加载，并在打开已有编码数据时自动逐级反显完整路径标签。

```tsx preview play=cascader-plus
import { Field, Form, Watch, useField, useForm, useLazyCascader, type CascaderOption, type Draft } from '@shikiform/react';
import { Badge, Card, LoaderIcon } from '@site/ui';

type Area = { code: string; name: string; children?: Area[] };
const AREA: Area[] = [
  { code: '110000', name: '北京市', children: [{ code: '11000000', name: '北京市', children: [{ code: '110105', name: '朝阳区' }, { code: '110108', name: '海淀区' }] }] },
  { code: '330000', name: '浙江省', children: [{ code: '330100', name: '杭州市', children: [{ code: '330106', name: '西湖区' }, { code: '330108', name: '滨江区' }] }] },
];

/** 一个 load(path) 服务所有层级：path 是已选的值，返回下一级。 */
async function loadArea(path: Array<string | number>): Promise<CascaderOption[]> {
  await new Promise((resolve) => setTimeout(resolve, 250));
  let level: Area[] | undefined = AREA;
  for (const code of path) level = level?.find((node) => node.code === String(code))?.children;
  return (level ?? []).map((node) => ({ label: node.name, value: node.code, isLeaf: path.length === 2 }));
}

type Values = { area: string[] };

function AreaCascader() {
  const f = useField<string[]>();
  const path = f.value ?? [];
  const cascader = useLazyCascader({ load: loadArea, paths: path.length ? [path] : undefined });

  const level0 = cascader.options;
  const level1 = level0.find((n) => n.value === path[0])?.children ?? [];
  const level2 = level1.find((n) => n.value === path[1])?.children ?? [];

  const pick = (depth: number, code: string) => {
    const next = [...path.slice(0, depth), ...(code ? [code] : [])];
    f.onChange(next);
    if (code && depth < 2) void cascader.loadChildren(next);
  };

  return (
    <div style={{ display: 'grid', gap: 12 }}>
      <Card variant="muted" padding="sm">
        <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', gap: 8 }}>
          <Badge variant="outline">{f.title}</Badge>
          {cascader.loading ? (
            <span style={{ display: 'inline-flex', alignItems: 'center', gap: 6, fontSize: 13, color: 'var(--sf-ui-fg-muted)' }}>
              <LoaderIcon size={13} /> 加载中…
            </span>
          ) : (
            <strong style={{ fontSize: 14 }}>{cascader.labelsOf(path).join(' / ') || '未选择'}</strong>
          )}
        </div>
      </Card>
      <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(140px, 1fr))', gap: 8 }}>
        <select aria-label="省" value={path[0] ?? ''} onChange={(e) => pick(0, e.target.value)}>
          <option value="">请选择省</option>
          {level0.map((opt) => <option key={String(opt.value)} value={String(opt.value)}>{opt.label}</option>)}
        </select>
        <select aria-label="市" value={path[1] ?? ''} disabled={!level1.length} onChange={(e) => pick(1, e.target.value)}>
          <option value="">请选择市</option>
          {level1.map((opt) => <option key={String(opt.value)} value={String(opt.value)}>{opt.label}</option>)}
        </select>
        <select aria-label="区" value={path[2] ?? ''} disabled={!level2.length} onChange={(e) => pick(2, e.target.value)}>
          <option value="">请选择区</option>
          {level2.map((opt) => <option key={String(opt.value)} value={String(opt.value)}>{opt.label}</option>)}
        </select>
      </div>
    </div>
  );
}

export default function App() {
  const form = useForm<Values>({
    initialValues: { area: ['110000', '11000000', '110105'] }, // 只有编码：打开时逐级加载出标签
  });
  return (
    <Card title="省市区懒加载级联" description="初始仅有区域编码，按需逐级加载并自动回显完整路径标签" style={{ maxWidth: 540 }}>
      <Form form={form}>
        <Field name="area" title="地址" component={[AreaCascader]} />
        <Watch select={(d: Draft<Values>) => d.$value}>
          {(values) => <pre>{JSON.stringify(values)}</pre>}
        </Watch>
      </Form>
    </Card>
  );
}
```

- `useLazyCascader({ load, paths })` 初始化时会自动加载第一级选项，并按 `paths` 逐级补齐已有编码对应的上级菜单，调用 `labelsOf(path)` 就能拿到完整的中文路径。
- 选中某一层级后调用 `loadChildren(nextPath)` 按需加载下一级选项，最末级标记 `isLeaf: true`。
