## 动态行按行计算

订单明细中每一行有各自的单价、数量和小计。通过 `reaction.each` 为每一行建立独立联动实例，修改任意一行的单价或数量时，只重算该行小计；嵌套多层数组时可通过 `ctx.key`、`ctx.depth` 过滤，详见 [用 ctx 过滤](/guide/linkage#用-ctx-过滤)。

```tsx preview
import { useState, type ReactNode } from 'react';
import { defineEffects, Form, Submit, typed, useArrayField, useField, useForm, useIndex } from '@shikiform/react';
import { Button, Card, Input, PlusIcon, Table, TableBody, TableCell, TableHead, TableHeader, TableRow, Toolbar, TrashIcon } from '@site/ui';

type Line = { sku: string; price: number | null; qty: number | null; total: number };
type Values = { lines: Line[] };

const effects = defineEffects<Values>(({ reaction, draft }) => {
  reaction.each(draft.lines, (row) => {
    const total = (row.price.$value ?? 0) * (row.qty.$value ?? 0);
    row.total.$value = total;
  });
});

const blank = (): Line => ({ sku: '', price: 0, qty: 1, total: 0 });

function TextCell() {
  const f = useField<string>();
  return (
    <Input
      id={f.id}
      size="sm"
      aria-label={String(f.title ?? '')}
      placeholder="商品名称"
      style={{ width: 120 }}
      value={f.value ?? ''}
      onChange={(e) => f.onChange(e.target.value)}
      onBlur={f.onBlur}
    />
  );
}

function NumberCell({ min }: { min?: number }) {
  const f = useField<number | null>();
  return (
    <Input
      id={f.id}
      size="sm"
      type="number"
      min={min}
      aria-label={String(f.title ?? '')}
      style={{ width: 88 }}
      value={f.value ?? ''}
      onChange={(e) => f.onChange(e.target.value === '' ? null : Number(e.target.value))}
      onBlur={f.onBlur}
    />
  );
}

function RemoveRow() {
  const lines = useArrayField<Line>();
  const index = useIndex();
  return (
    <Button
      variant="ghost"
      size="sm"
      aria-label="移除该行"
      disabled={lines.keys.length <= 1}
      onClick={() => lines.remove(index)}
      icon={<TrashIcon size={13} />}
    />
  );
}

function Lines({ children }: { children?: ReactNode }) {
  const lines = useArrayField<Line>();
  return (
    <>
      <Table compact>
        <TableHead>
          <TableRow>
            <TableHeader>商品</TableHeader>
            <TableHeader>单价</TableHeader>
            <TableHeader>数量</TableHeader>
            <TableHeader>小计</TableHeader>
            <TableHeader style={{ width: 48 }} />
          </TableRow>
        </TableHead>
        <TableBody>{children}</TableBody>
      </Table>
      <Toolbar>
        <Button variant="outline" size="sm" icon={<PlusIcon size={13} />} onClick={() => lines.push(blank())}>
          添加一行
        </Button>
        <Submit>
          {(s) => <Button type="submit" variant="primary" size="sm" disabled={s.disabled}>提交</Button>}
        </Submit>
      </Toolbar>
    </>
  );
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({
    initialValues: { lines: [{ sku: '贴纸', price: 2, qty: 4, total: 0 }] },
    effects: [effects],
  });
  const F = typed(form);
  return (
    <Card title="订单明细" description="reaction.each 为每行建立独立联动实例，新增或编辑单行互不干扰">
      <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))}>
        <F.ArrayField name="lines" component={Lines}>
          {(row) => (
            <TableRow>
              <TableCell><row.Field name="sku" title="商品" component={[TextCell]} /></TableCell>
              <TableCell><row.Field name="price" title="单价" component={[NumberCell, { min: 0 }]} /></TableCell>
              <TableCell><row.Field name="qty" title="数量" component={[NumberCell, { min: 0 }]} /></TableCell>
              <TableCell><row.Field name="total" title="小计" component={[NumberCell]} /></TableCell>
              <TableCell><RemoveRow /></TableCell>
            </TableRow>
          )}
        </F.ArrayField>
        <pre aria-live="polite">{result}</pre>
      </Form>
    </Card>
  );
}
```

## 嵌套对象与只读行

**场景**：客户资料表单包含嵌套的地址信息与订单列表。希望一键切换整张表单的编辑态/只读态，同时要求「已发货」的订单行始终不可修改数量，并且地址详情放在抽屉中编辑。

```tsx preview play=nested
import { useState, type ReactNode } from 'react';
import { ArrayField, Field, Form, ObjectField, Watch, defineEffects, useField, useFieldDecorator, useForm, type Draft } from '@shikiform/react';
import { Badge, Button, Card, Checkbox, Divider, FieldInput, Input, Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@site/ui';

type Order = { id: string; title: string; qty: number; shipped: boolean };
type Values = { editable: boolean; profile: { name: string; address: { city: string; street: string } }; orders: Order[] };

const effects = defineEffects<Values>(({ reaction, draft }) => {
  reaction((d) => {
    const readOnly = d.editable.$value ? undefined : true; // undefined = 继承上级
    d.profile.$readOnly = readOnly;
    d.orders.$readOnly = readOnly;
  });
  reaction.each(draft.orders, (row) => {
    row.qty.$readOnly = row.shipped.$value ? true : undefined;
  });
});

function FieldRow({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return (
    <p className="sf-ui-form-item" style={{ margin: 0 }}>
      <label {...d.labelProps} className="sf-ui-form-label">
        <span>{d.label}：</span>
      </label>
      {children}
    </p>
  );
}

function NumberInline({ min }: { min?: number }) {
  const f = useField<number>();
  return (
    <label htmlFor={f.id} className="sf-ui-form-label" style={{ gap: 6 }}>
      <span>{f.title}</span>
      {f.readOnly ? (
        <Badge variant="outline">{f.value}</Badge>
      ) : (
        <Input
          id={f.id}
          type="number"
          size="sm"
          min={min}
          style={{ width: 72 }}
          value={f.value ?? 0}
          onChange={(e) => f.onChange(Number(e.target.value))}
          onBlur={f.onBlur}
        />
      )}
    </label>
  );
}

function ToggleInline() {
  const f = useField<boolean>();
  return (
    <Checkbox
      id={f.id}
      checked={Boolean(f.value)}
      disabled={f.readOnly || f.disabled}
      onCheckedChange={(checked) => f.onChange(checked)}
    >
      {f.title}
    </Checkbox>
  );
}

export default function App() {
  const [open, setOpen] = useState(false);
  const form = useForm<Values>({
    initialValues: {
      editable: true,
      profile: { name: 'Ada', address: { city: '伦敦', street: '贝克街 221B' } },
      orders: [{ id: 'o1', title: '未发货', qty: 1, shipped: false }, { id: 'o2', title: '已发货', qty: 2, shipped: true }],
    },
    fields: { editable: { omit: true } },
    effects: [effects],
  });
  return (
    <Form form={form}>
      <Card
        title="客户资料与订单"
        description="整表只读沿路径继承，已发货订单行始终锁定数量"
        extra={<Field name="editable" title="可编辑" component={[ToggleInline]} />}
      >
        <div style={{ display: 'grid', gap: 14 }}>
          <ObjectField name="profile">
            <div style={{ display: 'grid', gap: 10, maxWidth: 420 }}>
              <Field name="name" title="姓名" decorator={[FieldRow]} component={[FieldInput]} />
              <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', gap: 8 }}>
                <span className="sf-ui-form-label">地址：</span>
                <Watch select={(d: Draft<Values>) => d.profile.address.$value}>
                  {(address) => <Badge variant="outline">{address.city} · {address.street}</Badge>}
                </Watch>
                <Button variant="outline" size="sm" onClick={() => setOpen((prev) => !prev)}>
                  {open ? '收起' : '编辑'}
                </Button>
              </div>
              {open && (
                <fieldset className="sf-ui-card sf-ui-card--muted" style={{ padding: 14, margin: 0 }}>
                  <legend style={{ padding: '0 6px', fontSize: 12, fontWeight: 600, color: 'var(--sf-ui-fg-muted)' }}>
                    地址
                  </legend>
                  <ObjectField name="address">
                    <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(160px, 1fr))', gap: 10 }}>
                      <Field name="city" title="城市" decorator={[FieldRow]} component={[FieldInput]} />
                      <Field name="street" title="街道" decorator={[FieldRow]} component={[FieldInput]} />
                    </div>
                  </ObjectField>
                </fieldset>
              )}
            </div>
          </ObjectField>
          <Divider />
          <Table compact>
            <TableHead>
              <TableRow>
                <TableHeader>订单</TableHeader>
                <TableHeader>数量</TableHeader>
                <TableHeader>发货状态</TableHeader>
              </TableRow>
            </TableHead>
            <TableBody>
              <ArrayField name="orders" rowKey="id">
                {() => (
                  <TableRow>
                    <TableCell>
                      <Field name="title" readOnly>{(f) => <strong>{String(f.value ?? '')}</strong>}</Field>
                    </TableCell>
                    <TableCell>
                      <Field name="qty" title="数量" component={[NumberInline, { min: 1 }]} />
                    </TableCell>
                    <TableCell>
                      <Field name="shipped" title="已发货" component={[ToggleInline]} />
                    </TableCell>
                  </TableRow>
                )}
              </ArrayField>
            </TableBody>
          </Table>
        </div>
      </Card>
    </Form>
  );
}
```

- `$readOnly` 沿字段路径向下走：写 `d.profile` 和 `d.orders`，里面的字段跟着变。赋 `undefined` 丢掉这一层，回到上一级。
- 已发货行单独写了 `row.qty.$readOnly = true`，整表可编辑时这一格仍是预览。
- 弹出面板里的 `<ObjectField name="address">` 仍然在 `profile` 下面，可以直接写相对路径；收起面板后，填过的值和状态照常保存在表单里。

## 树：跨层移动

**场景**：实现一个支持跨父级拖拽的菜单树编辑器，要求节点的错误状态和输入状态能随着节点移动保留，同时禁止将父节点拖入自身的子孙节点中。

```tsx preview play=headless
import { useState } from 'react';
import { Form, TreeField, useField, useForm, useTreeField } from '@shikiform/react';
import { Button, Card, GripVerticalIcon, Input, Toolbar } from '@site/ui';

type Node = { id: string; name: string; children: Node[] };

function NodeRow() {
  const { pos, path, record, dragProps } = useTreeField<Node>();
  const name = useField<Node, 'name'>('name', { required: true });
  return (
    <div
      {...dragProps()}
      style={{
        display: 'flex',
        flexWrap: 'wrap',
        alignItems: 'center',
        gap: 8,
        cursor: 'grab',
        padding: '6px 10px',
        marginBottom: 6,
        borderRadius: 6,
        border: '1px solid var(--sf-ui-border)',
        background: 'var(--sf-ui-surface)',
      }}
    >
      <GripVerticalIcon size={14} style={{ color: 'var(--sf-ui-fg-muted)', flexShrink: 0 }} />
      <Input
        size="sm"
        aria-label="名称"
        style={{ width: 136 }}
        invalid={Boolean(name.errors[0])}
        value={name.value}
        onChange={(event) => name.onChange(event.target.value)}
        onBlur={name.onBlur}
      />
      <small className="sf-ui-badge sf-ui-badge--default">id {record.id} · pos [{pos.join(',')}] · {path}</small>
      {name.errors[0] && <small role="alert" className="sf-ui-form-error">{name.errors[0]}</small>}
    </div>
  );
}

function Menu({ onValidate }: { onValidate: () => void }) {
  const tree = useTreeField('menu', { rowKey: 'id' });
  const [result, setResult] = useState('');
  return (
    <>
      <ul style={{ paddingLeft: 0, listStyle: 'none', margin: 0 }}>
        <TreeField name="menu" rowKey="id">
          {(node) => (
            <li>
              <NodeRow />
              {node.children && (
                <ul style={{ paddingLeft: 24, listStyle: 'none', margin: 0, borderLeft: '1px dashed var(--sf-ui-border)' }}>
                  {node.children}
                </ul>
              )}
            </li>
          )}
        </TreeField>
      </ul>
      <Toolbar>
        <Button variant="primary" size="sm" onClick={onValidate}>校验</Button>
        <Button variant="outline" size="sm" onClick={() => setResult(JSON.stringify(tree.move([1, 0], [0, 0])))}>
          「撤销」移到「文件」最前
        </Button>
        <Button variant="outline" size="sm" onClick={() => setResult(JSON.stringify(tree.move([0], [0, 0])))}>
          「文件」移进自己的子节点
        </Button>
      </Toolbar>
      <pre>{result}</pre>
    </>
  );
}

export default function App() {
  const form = useForm({
    initialValues: {
      menu: [
        { id: 'file', name: '文件', children: [{ id: 'new', name: '', children: [] }, { id: 'open', name: '打开', children: [] }] },
        { id: 'edit', name: '编辑', children: [{ id: 'undo', name: '撤销', children: [] }] },
      ] as Node[],
    },
  });
  return (
    <Card title="菜单树编辑器" description="先校验让空节点报错，再执行跨层移动：错误状态随节点 id 自动迁移">
      <Form form={form}>
        <Menu onValidate={() => { void form.validate().catch(() => {}); }} />
      </Form>
    </Card>
  );
}
```

- 先点击「校验」让空名称节点报错，再将其拖拽到其他分组下：错误提示会跟随业务 `id` 一同迁移，同时节点的 `pos` 与 `path` 自动更新为新位置。
- `move(from, to)`（`useTreeField(name)` 或节点上的 `useTreeField()`）返回 `{ ok, reason? }`，移入自身子孙时拒绝并给出原因。节点里的 `useTreeField()` 给出当前 `path`、`pos` 和兄弟 `index`。
- `<TreeField>` 会自动记住每个节点的位置并把子节点传给 `node.children`，子组件里直接调用 `useTreeField()` 就能拿到当前节点的坐标、数据和 `dragProps()`，不用手拼整棵树的路径，详见 [Headless Hooks](/guide/headless)。

## 长列表：只挂载可见行

**场景**：2000 行名单只挂载视口内可见行，提交前全部 2000 行的必填规则仍参与校验。

```tsx preview play=virtual
import { useState } from 'react';
import { ArrayField, Field, Form, useForm } from '@shikiform/react';
import { Badge, Button, Card, Input } from '@site/ui';

type Row = { id: string; name: string };

export default function App() {
  const [errors, setErrors] = useState('');
  const form = useForm<{ rows: Row[] }>({
    initialValues: { rows: Array.from({ length: 2000 }, (_, index) => ({ id: `v${index}`, name: index % 500 === 499 ? '' : `第 ${index} 行` })) },
    fields: { 'rows.*.name': { required: true } }, // 行模板：每一行都有，挂没挂载都查
  });
  const validate = async () => {
    try {
      const values = await form.validate();
      setErrors(`无（${values.rows.length} 行通过）`);
    } catch (error) {
      const errors = Array.isArray(error) ? error as { path: string }[] : [];
      setErrors(errors.map((item) => item.path).join('、') || '无');
    }
  };
  return (
    <Card
      title="2,000 行虚拟滚动名单"
      description="行模板 rows.*.name 在引擎层生效，视口外未挂载的空行同样拦截校验"
      extra={<Badge variant="brand">virtualScroll</Badge>}
    >
      <Form form={form}>
        <ArrayField name="rows" rowKey="id" virtualScroll={{ height: 240, rowHeight: 42, overscan: 4 }}>
          {({ index }) => (
            <div style={{ display: 'flex', gap: 10, alignItems: 'center', height: 42, borderBottom: '1px solid var(--sf-ui-border)' }}>
              <Badge variant="outline" style={{ width: 58, justifyContent: 'center' }}>#{index}</Badge>
              <Field<Row, 'name'> name="name">
                {(f) => (
                  <>
                    <Input
                      size="sm"
                      aria-label={`第 ${index} 行名称`}
                      style={{ width: 210 }}
                      invalid={Boolean(f.errors[0])}
                      value={f.value ?? ''}
                      onChange={(e) => f.onChange(e.target.value)}
                      onBlur={f.onBlur}
                    />
                    {f.errors[0] && <small role="alert" className="sf-ui-form-error">{f.errors[0]}</small>}
                  </>
                )}
              </Field>
            </div>
          )}
        </ArrayField>
        <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', gap: 12 }}>
          <Button variant="primary" size="sm" onClick={validate}>校验全部 2000 行</Button>
          <p style={{ margin: 0, fontSize: 13 }}>错误：{errors}</p>
        </div>
      </Form>
    </Card>
  );
}
```

- 第 499、999、1499、1999 行处于视口外未挂载，点击校验仍会报出它们的空值错误：`fields` 中的行模板 `rows.*.name` 在引擎层生效，不依赖组件挂载。
- 通过时 `const values = await form.validate()` 拿到去掉 omit 后的提交值（这里是全部 2000 行）。失败仍 reject `FieldError[]`。
- `<ArrayField virtualScroll>` 仅挂载可见行并注入行作用域，只用于自定义列表行，不能与整表容器 `component` 同时使用。
