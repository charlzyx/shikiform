## Try an expense claim

Change a category to see each row adapt. Add an expense to see the total and approval route update.

```tsx preview
import { useEffect, useRef, useState, type ReactNode } from 'react';
import { enUS, defineEffects, Form, Submit, typed, useArrayField, useField, useFieldDecorator, useForm, useIndex, useWatch, type Draft } from '@shikiform/react';

type Expense = { category: 'travel' | 'meal'; amount: number | null; city: string; attendees: string };
type Values = { expenses: Expense[]; summary: { total: number; approval: 'standard' | 'manager' } };

const effects = defineEffects<Values>(({ reaction }) => {
  reaction.each((d) => d.expenses, (row) => {
    const { category, city, attendees } = row;
    const travel = category.$value === 'travel';
    city.$visible = travel;
    city.$required = travel;
    city.$omit = !travel;
    attendees.$visible = !travel;
    attendees.$required = !travel;
    attendees.$omit = travel;
  });
  reaction(({ expenses, summary }) => {
    const total = expenses.$value.reduce((sum, item) => sum + (item.amount ?? 0), 0);
    summary.$value.total = Math.round(total * 100) / 100;
    summary.$value.approval = total > 1000 ? 'manager' : 'standard';
  });
});

function FieldItem({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return <div className="claim-field">
    <label {...d.labelProps}>{d.label}{d.required && <span aria-hidden="true"> *</span>}</label>
    {children}
    {d.errorProps && <small {...d.errorProps} className="claim-error">{d.errors[0]}</small>}
  </div>;
}
function TextInput() {
  const field = useField<string>();
  const d = useFieldDecorator();
  return <input aria-required={field.required} aria-invalid={d.invalid || undefined} aria-describedby={d.describedBy} id={field.id} value={field.value ?? ''} onChange={(e) => field.onChange(e.target.value)} onBlur={field.onBlur} />;
}
function AmountInput() {
  const field = useField<number | null>();
  const d = useFieldDecorator();
  return <input aria-required={field.required} aria-invalid={d.invalid || undefined} aria-describedby={d.describedBy} id={field.id} type="number" min="0.01" step="0.01" inputMode="decimal" value={field.value ?? ''}
    onChange={(e) => field.onChange(e.target.value === '' ? null : Number(e.target.value))} onBlur={field.onBlur} />;
}
function CategorySelect() {
  const field = useField<string>();
  const d = useFieldDecorator();
  return <select aria-required={field.required} aria-invalid={d.invalid || undefined} aria-describedby={d.describedBy} id={field.id} value={field.value} onChange={(e) => field.onChange(e.target.value)} onBlur={field.onBlur}>
    {field.options.map((option) => <option key={String(option.value)} value={String(option.value)}>{option.label}</option>)}
  </select>;
}
function Expenses({ children }: { children?: ReactNode }) {
  const expenses = useArrayField<Expense>();
  return <div className="claim-lines">{children}
    <button className="claim-add" type="button" onClick={() => expenses.push({ category: 'travel', amount: null, city: '', attendees: '' })}>+ Add expense</button>
  </div>;
}
function RowHeader() {
  const expenses = useArrayField<Expense>();
  const index = useIndex();
  return <div className="claim-row-head"><span>ITEM {String(index + 1).padStart(2, '0')}</span>
    <button type="button" aria-label={`Remove ${index + 1}`} disabled={expenses.keys.length <= 1} onClick={() => expenses.remove(index)}>Remove</button>
  </div>;
}
function Summary() {
  const { total, approval } = useWatch((d: Draft<Values>) => d.summary.$value);
  return <div className="claim-summary" aria-live="polite">
    <div><span>Claim total</span><strong>¥ {total.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</strong></div>
    <div className="claim-approval"><b>{approval === 'manager' ? 'Manager approval' : 'Standard approval'}</b><small>Over ¥1,000 → manager review</small></div>
  </div>;
}
function ForgetResult({ clear }: { clear: () => void }) {
  const values = useWatch((d: Draft<Values>) => d.$value);
  const clearRef = useRef(clear);
  clearRef.current = clear;
  useEffect(() => { clearRef.current(); }, [values]);
  return null;
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({
    messages: enUS,
    initialValues: {
      expenses: [
        { category: 'travel', amount: 680, city: 'Shanghai', attendees: '' },
        { category: 'meal', amount: 240, city: '', attendees: 'Product team · 3 people' },
      ],
      summary: { total: 0, approval: 'standard' },
    },
    effects: [effects],
  });
  const F = typed(form);
  return <section className="claim-demo">
    <div className="claim-heading"><span className="claim-seal" aria-hidden="true">支</span><div><small>EXPENSE CLAIM · LIVE</small><h3>One trip. Two expenses.</h3></div><span className="claim-local">Local demo</span></div>
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values, null, 2))} onInvalid={() => setResult('')}>
      <ForgetResult clear={() => setResult('')} />
      <F.ArrayField name="expenses" component={Expenses}>
        {(row) => <div className="claim-row">
          <RowHeader />
          <div className="claim-fields">
            <row.Field name="category" title="Category" enum={[{ label: 'Business travel', value: 'travel' }, { label: 'Team meal', value: 'meal' }]} decorator={[FieldItem]} component={[CategorySelect]} />
            <row.Field name="amount" title="Amount (CNY)" required rules={{ min: 0.01 }} decorator={[FieldItem]} component={[AmountInput]} />
            <row.Field name="city" title="Destination city" decorator={[FieldItem]} component={[TextInput]} />
            <row.Field name="attendees" title="Attendees" decorator={[FieldItem]} component={[TextInput]} />
          </div>
        </div>}
      </F.ArrayField>
      <Summary />
      <div className="claim-actions"><span>Try changing a category or increasing the total.</span><Submit>Validate & preview JSON →</Submit></div>
      {result && <div className="claim-result"><strong>Submission payload · nothing sent</strong><pre aria-live="polite">{result}</pre></div>}
    </Form>
  </section>;
}
```

## How ShikiForm handles form state

A form instance owns the values, rules, validation results and UI state. Controls subscribe to individual paths. A `reaction` tracks the states it reads and updates dependent fields, including fields outside the current screen. Use `watch` to reset a dependent value or start a request after a change.

| Topic | ShikiForm | React Hook Form | Formily |
|---|---|---|---|
| Dependent fields | Assign to draft state in a reaction. | Components typically use `watch` and `setValue`. | Reactions often live alongside a schema. |
| UI controls | Bind a control with `useField`, `connect` or `useFieldDecorator`. | Use `Controller` for controlled inputs. | UI adapters are available as separate packages. |
| Visibility and submission | Hidden fields skip validation and still submit. Set `omit` to exclude them. | Registration and mount settings affect submission. Disabled inputs are excluded by default. | Hidden fields retain their value. The none display mode excludes them. |

## Installation

```bash
npm i @shikiform/react
# Headless query tables, draft editors and array tables
npm i @shikiform/hooks
```

Use `@shikiform/core` without React, `@shikiform/react` for fields and hooks, and `@shikiform/hooks` for tables, suggestions and draft editors. Start with the [first form](/en/guide/quick-start), work through the [interactive tutorial](/en/tutorial), or explore the [examples](/en/examples).
