## Try an expense claim

Change a category to see each row adapt. Add an expense to see the total and approval route update.

```tsx preview
import { useEffect, useRef, useState, type ReactNode } from 'react';
import {
  enUS,
  defineEffects,
  Form,
  Submit,
  typed,
  useArrayField,
  useForm,
  useIndex,
  useWatch,
  type Draft,
} from '@shikiform/react';
import {
  Alert,
  Badge,
  Button,
  Card,
  FieldInput,
  FieldNumber,
  FieldSelect,
  PlusIcon,
  siteKit,
  TrashIcon,
} from '@site/ui';

type Expense = { category: 'travel' | 'meal'; amount: number | null; city: string; attendees: string };
type Values = { expenses: Expense[]; summary: { total: number; approval: 'standard' | 'manager' } };

const effects = defineEffects<Values>(({ reaction, draft }) => {
  reaction.each(draft.expenses, (row) => {
    const { category, city, attendees } = row;
    const travel = category.$value === 'travel';
    city.$visible = travel;
    city.$required = travel;
    city.$omit = !travel;
    attendees.$visible = !travel;
    attendees.$required = !travel;
    attendees.$omit = travel;
  });
  reaction(({ expenses, summary }) => {
    const total = expenses.$value.reduce((sum, item) => sum + (item.amount ?? 0), 0);
    summary.$value.total = Math.round(total * 100) / 100;
    summary.$value.approval = total > 1000 ? 'manager' : 'standard';
  });
});

function Expenses({ children }: { children?: ReactNode }) {
  const expenses = useArrayField<Expense>();
  return (
    <div className="claim-lines">
      {children}
      <Button
        variant="outline"
        block
        icon={<PlusIcon size={14} />}
        style={{ borderStyle: 'dashed' }}
        onClick={() => expenses.push({ category: 'travel', amount: null, city: '', attendees: '' })}
      >
        Add expense
      </Button>
    </div>
  );
}

function RowCard({ children }: { children?: ReactNode }) {
  const expenses = useArrayField<Expense>();
  const index = useIndex();
  return (
    <Card
      padding="sm"
      title={`ITEM ${String(index + 1).padStart(2, '0')}`}
      style={{ borderInlineStart: '3px solid var(--sf-cinnabar, var(--sf-ui-brand))' }}
      extra={
        <Button
          variant="ghost"
          size="sm"
          icon={<TrashIcon size={13} />}
          aria-label={`Remove ${index + 1}`}
          disabled={expenses.keys.length <= 1}
          onClick={() => expenses.remove(index)}
        >
          Remove
        </Button>
      }
    >
      <div className="claim-fields">{children}</div>
    </Card>
  );
}

function Summary() {
  const { total, approval } = useWatch((d: Draft<Values>) => d.summary.$value);
  return (
    <div className="claim-summary" aria-live="polite">
      <div>
        <span>Claim total</span>
        <strong>¥ {total.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</strong>
      </div>
      <div style={{ display: 'grid', gap: 6, justifyItems: 'end' }}>
        <Badge variant={approval === 'manager' ? 'warning' : 'success'} dot>
          {approval === 'manager' ? 'Manager approval' : 'Standard approval'}
        </Badge>
        <small>Over ¥1,000 → manager review</small>
      </div>
    </div>
  );
}

function ForgetResult({ clear }: { clear: () => void }) {
  const values = useWatch((d: Draft<Values>) => d.$value);
  const clearRef = useRef(clear);
  clearRef.current = clear;
  useEffect(() => { clearRef.current(); }, [values]);
  return null;
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({
    messages: enUS,
    initialValues: {
      expenses: [
        { category: 'travel', amount: 680, city: 'Shanghai', attendees: '' },
        { category: 'meal', amount: 240, city: '', attendees: 'Product team · 3 people' },
      ],
      summary: { total: 0, approval: 'standard' },
    },
    effects: [effects],
  });
  const F = typed(form);
  return (
    <section style={{ maxWidth: 880, marginInline: 'auto' }}>
      <div className="claim-heading">
        <span className="claim-seal" aria-hidden="true">支</span>
        <div>
          <small>EXPENSE CLAIM · LIVE</small>
          <h3>One trip. Two expenses.</h3>
        </div>
        <Badge variant="outline" style={{ marginInlineStart: 'auto' }}>Local demo</Badge>
      </div>
      <Form
        form={form}
        kit={siteKit}
        onSubmit={(values) => setResult(JSON.stringify(values, null, 2))}
        onInvalid={() => setResult('')}
      >
        <ForgetResult clear={() => setResult('')} />
        <F.ArrayField name="expenses" component={Expenses}>
          {(row) => (
            <RowCard>
              <row.Field
                name="category"
                title="Category"
                enum={[{ label: 'Business travel', value: 'travel' }, { label: 'Team meal', value: 'meal' }]}
                component={FieldSelect}
              />
              <row.Field
                name="amount"
                title="Amount (CNY)"
                required
                rules={{ min: 0.01 }}
                component={[FieldNumber, { min: 0.01, step: 10, prefix: '¥' }]}
              />
              <row.Field name="city" title="Destination city" component={[FieldInput, { placeholder: 'e.g. Shanghai' }]} />
              <row.Field name="attendees" title="Attendees" component={[FieldInput, { placeholder: 'e.g. Product team · 3' }]} />
            </RowCard>
          )}
        </F.ArrayField>
        <Summary />
        <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', justifyContent: 'space-between', gap: 12 }}>
          <span style={{ fontSize: 12.5, color: 'var(--sf-ui-fg-muted)' }}>Try changing a category or increasing the total.</span>
          <Submit asChild><Button variant="primary">Validate & preview JSON →</Button></Submit>
        </div>
        {result && (
          <Alert variant="success" title="Submission payload · nothing sent">
            <pre aria-live="polite" style={{ marginTop: 6 }}>{result}</pre>
          </Alert>
        )}
      </Form>
    </section>
  );
}
```

## How ShikiForm handles form state

A form instance owns the values, rules, validation results and UI state. Controls subscribe to individual paths. A `reaction` tracks the states it reads and updates dependent fields, including fields outside the current screen. Use `watch` to reset a dependent value or start a request after a change.

| Topic | ShikiForm | React Hook Form | Formily |
|---|---|---|---|
| Dependent fields | Assign to draft state in a reaction. | Components typically use `watch` and `setValue`. | Reactions often live alongside a schema. |
| UI controls | Bind a control with `useField`, `connect` or `useFieldDecorator`. | Use `Controller` for controlled inputs. | UI adapters are available as separate packages. |
| Visibility and submission | Hidden fields skip validation and still submit. Set `omit` to exclude them. | Registration and mount settings affect submission. Disabled inputs are excluded by default. | Hidden fields retain their value. The none display mode excludes them. |

## Installation

Install `@shikiform/react` for React projects — it bundles the core engine, headless React bindings, and pro hooks for query tables, draft editors, and editable array tables:

```bash
npm i @shikiform/react
```

| Package | Role |
|---|---|
| `@shikiform/react` | Headless React bindings (`Form`, `Field`, `useField`, `connect`, `useFieldDecorator`), pro hooks (`useQueryTable`, `useShadowForm`, `useArrayTable`, `useColumnSettings`), and common `@shikiform/core` re-exports |
| `@shikiform/core` | React-free TypeScript form engine: state store, drafts, `reaction` / `watch`, and validation (install directly only without React) |
