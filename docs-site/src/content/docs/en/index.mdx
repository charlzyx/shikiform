## Try the invoice form

Switch invoice type, edit or add line items, then submit to see the JSON. Open the code to see nested object updates and per-row calculations in effects.

```tsx preview
import { useEffect, useRef, useState, type ReactNode } from 'react';
import { enUS, defineEffects, Form, Submit, typed, useArrayField, useField, useFieldDecorator, useForm, useIndex, useWatch, type Draft } from '@shikiform/react';

type Item = { price: number | null; qty: number | null; subtotal: number };
type Values = {
  invoice: { kind: 'personal' | 'company'; taxId: string };
  items: Item[];
  summary: { total: number };
};

const effects = defineEffects<Values>(({ reaction }) => {
  reaction(({ invoice }) => {
    const { kind, taxId } = invoice;
    const company = kind.$value === 'company';
    taxId.$visible = company;
    taxId.$required = company;
    taxId.$omit = !company;
  });
  reaction.each((d) => d.items, (row) => {
    const { price, qty, subtotal } = row;
    const unitPrice = price.$value ?? 0;
    const quantity = qty.$value ?? 0;
    subtotal.$value = unitPrice * quantity;
    subtotal.$readOnly = true;
  });
  reaction((d) => {
    d.summary.$value.total = d.items.$value.reduce((sum, item) => sum + item.subtotal, 0);
  });
});

function FieldItem({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return (
    <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', gap: 8 }}>
      <label {...d.labelProps}>
        {d.required && <span style={{ color: 'crimson' }}>* </span>}
        {d.label}
      </label>
      {children}
      {d.errorProps && <small {...d.errorProps} style={{ color: 'crimson' }}>{d.errors[0]}</small>}
    </div>
  );
}

function TextInput() {
  const field = useField<string>();
  return <input id={field.id} value={field.value ?? ''} style={{ maxWidth: '100%' }}
    onChange={(e) => field.onChange(e.target.value)} onBlur={field.onBlur} />;
}

function NumberInput() {
  const field = useField<number | null>();
  return <input id={field.id} type="number" min={0} step="any" readOnly={field.readOnly}
    value={field.value ?? ''} style={{ width: 100, maxWidth: '100%' }}
    onChange={(e) => field.onChange(e.target.value === '' ? null : Number(e.target.value))}
    onBlur={field.onBlur} />;
}

function RadioGroup() {
  const field = useField<string>();
  return (
    <span style={{ display: 'inline-flex', flexWrap: 'wrap', gap: 12 }}>
      {field.options.map((o) => (
        <label key={String(o.value)}>
          <input type="radio" name={field.id} checked={field.value === o.value}
            onChange={() => field.onChange(String(o.value))} />
          {' '}{o.label}
        </label>
      ))}
    </span>
  );
}

function Items({ children }: { children?: ReactNode }) {
  const items = useArrayField<Item>();
  return <div>
    {children}
    <button type="button" onClick={() => items.push({ price: 0, qty: 1, subtotal: 0 })}>Add item</button>
  </div>;
}

function RemoveItem() {
  const items = useArrayField<Item>();
  const index = useIndex();
  return <button type="button" onClick={() => items.remove(index)}>Remove</button>;
}

function ForgetResult({ clear }: { clear: () => void }) {
  const values = useWatch((d: Draft<Values>) => d.$value);
  const clearRef = useRef(clear);
  clearRef.current = clear;
  useEffect(() => { clearRef.current(); }, [values]);
  return null;
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({
    messages: enUS,
    initialValues: {
      invoice: { kind: 'personal', taxId: '' },
      items: [{ price: 10, qty: 2, subtotal: 0 }],
      summary: { total: 0 },
    },
    effects: [effects],
  });
  const F = typed(form);
  return (
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values, null, 2))} onInvalid={() => setResult('')}>
      <ForgetResult clear={() => setResult('')} />
      <F.Field name="invoice.kind" title="Invoice type" enum={[{ label: 'Personal', value: 'personal' }, { label: 'Company', value: 'company' }]}
        decorator={[FieldItem]} component={[RadioGroup]} />
      <F.Field name="invoice.taxId" title="Tax ID" decorator={[FieldItem]} component={[TextInput]} />
      <F.ArrayField name="items" component={Items}>
        {(row) => (
          <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', gap: 12, marginBottom: 12 }}>
            <row.Field name="price" title="Unit price" required rules={{ min: 0 }} decorator={[FieldItem]} component={[NumberInput]} />
            <row.Field name="qty" title="Quantity" required rules={{ min: 1 }} decorator={[FieldItem]} component={[NumberInput]} />
            <row.Field name="subtotal" title="Subtotal" decorator={[FieldItem]} component={[NumberInput]} />
            <RemoveItem />
          </div>
        )}
      </F.ArrayField>
      <F.Field name="summary.total" title="Total" readOnly decorator={[FieldItem]} component={[NumberInput]} />
      <Submit>Submit</Submit>
      <pre aria-live="polite">{result}</pre>
    </Form>
  );
}
```

## How ShikiForm handles form state

A form instance owns the values, rules, validation results and UI state. Controls subscribe to individual paths. A `reaction` tracks the states it reads and updates dependent fields, including fields outside the current screen. Use `watch` to reset a dependent value or start a request after a change.

| Topic | ShikiForm | React Hook Form | Formily |
|---|---|---|---|
| Dependent fields | Assign to draft state in a reaction. | Components typically use `watch` and `setValue`. | Reactions often live alongside a schema. |
| UI controls | Bind a control with `useField`, `connect` or `useFieldDecorator`. | Use `Controller` for controlled inputs. | UI adapters are available as separate packages. |
| Visibility and submission | Hidden fields skip validation and still submit. Set `omit` to exclude them. | Registration and mount settings affect submission. Disabled inputs are excluded by default. | Hidden fields retain their value. The none display mode excludes them. |

## Installation

```bash
npm i @shikiform/react
# Headless query tables, draft editors and array tables
npm i @shikiform/hooks
```

Use `@shikiform/core` without React, `@shikiform/react` for fields and hooks, and `@shikiform/hooks` for tables, suggestions and draft editors. Start with the [first form](/en/guide/quick-start), work through the [interactive tutorial](/en/tutorial), or explore the [examples](/en/examples).
