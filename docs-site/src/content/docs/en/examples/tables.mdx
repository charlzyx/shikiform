## Query table

Choose a supplier to change the available models. Collapsing the filters preserves input. A new query starts on page one, while paging preserves row selection.

```tsx preview play=query
import { useState } from 'react';
import { Field, Form, defineEffects, useField, useQueryTable } from '@shikiform/react';
import {
  Badge,
  Button,
  Card,
  Checkbox,
  ChevronDownIcon,
  ChevronLeftIcon,
  ChevronRightIcon,
  ChevronUpIcon,
  Empty,
  Input,
  SearchIcon,
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
  Tag,
  Toolbar,
} from '@site/ui';

type Log = { id: string; keyword: string; provider: string; model: string; status: string };
type Query = { keyword: string; provider: string; model: string; user: string };
type Values = { query: Query; list: Log[] };

const LOGS: Log[] = Array.from({ length: 12 }, (_, i) => ({
  id: String(i + 1),
  keyword: `${i % 2 ? 'beta' : 'alpha'}-${i + 1}`,
  provider: i % 2 ? 'anthropic' : 'openai',
  model: i % 2 ? 'claude' : 'gpt',
  status: i % 3 ? 'ok' : 'error',
}));
const MODELS: Record<string, string[]> = { openai: ['gpt'], anthropic: ['claude'], '': ['gpt', 'claude'] };
const options = (values: string[]) => values.map((value) => ({ label: value, value }));

const effects = defineEffects<Values>(({ reaction, watch }) => {
  reaction((d) => {
    d.query.model.$enum = options(MODELS[d.query.provider.$value] ?? []);
  });
  watch((d) => d.query.provider.$value, (d) => {
    if (!(MODELS[d.query.provider.$value] ?? []).includes(d.query.model.$value)) d.query.model.$value = '';
  });
});

function TextFilter() {
  const f = useField<string>();
  return (
    <label htmlFor={f.id} className="sf-ui-form-label" style={{ gap: 6 }}>
      <span>{f.title}</span>
      <Input id={f.id} size="sm" style={{ width: 112 }} value={f.value ?? ''} onChange={(e) => f.onChange(e.target.value)} />
    </label>
  );
}

function SelectFilter() {
  const f = useField<string>();
  return (
    <label htmlFor={f.id} className="sf-ui-form-label" style={{ gap: 6 }}>
      <span>{f.title}</span>
      <select
        id={f.id}
        style={{ minWidth: 112, minHeight: 30, padding: '3px 28px 3px 10px' }}
        value={f.value ?? ''}
        onChange={(e) => f.onChange(e.target.value)}
      >
        <option value="">All</option>
        {f.options.map((opt) => (
          <option key={String(opt.value)} value={String(opt.value)}>{opt.label}</option>
        ))}
      </select>
    </label>
  );
}

export default function App() {
  const [expanded, setExpanded] = useState(false);
  const table = useQueryTable({
    initialValues: { query: { keyword: '', provider: '', model: '', user: '' }, list: [] } as Values,
    fields: { 'query.provider': { enum: options(['openai', 'anthropic']) } },
    effects: [effects],
    defaultPageSize: 5,
    params: 'query',
    rows: 'list',
    request: async ({ params, page, pageSize, filters }) => {
      const rows = LOGS.filter((row) =>
        (!params.keyword || row.keyword.includes(params.keyword)) &&
        (!params.provider || row.provider === params.provider) &&
        (!params.model || row.model === params.model) &&
        (!params.user || row.id === params.user) &&
        (!filters.status || filters.status.includes(row.status)));
      return { data: rows.slice((page - 1) * pageSize, page * pageSize), total: rows.length };
    },
  });
  const pages = Math.max(1, Math.ceil(table.total / table.pageSize));
  const statusFilter = table.filteredValueOf('status')?.[0] ?? '';
  const toggleSelect = (id: string, checked: boolean) =>
    table.setSelectedKeys(checked ? [...table.selectedKeys, id] : table.selectedKeys.filter((k) => k !== id));
  return (
    <Card title="Log query" description="Linked supplier/model filters, collapsible inputs and persistent paging selection">
      <Form form={table.form} onSubmit={() => table.submit()}>
        <Card variant="muted" padding="sm">
          <div style={{ display: 'flex', flexWrap: 'wrap', gap: 10, alignItems: 'center' }}>
            <Field name="query.keyword" title="Keyword" component={[TextFilter]} />
            <Field name="query.provider" title="Supplier" component={[SelectFilter]} />
            <Field name="query.model" title="Model" component={[SelectFilter]} />
            <span style={{ display: expanded ? 'inline-flex' : 'none' }}>
              <Field name="query.user" title="User ID" component={[TextFilter]} />
            </span>
            <Button type="submit" variant="primary" size="sm" icon={<SearchIcon size={13} />}>Search</Button>
            <Button variant="outline" size="sm" onClick={table.reset}>Reset</Button>
            <Button
              variant="ghost"
              size="sm"
              icon={expanded ? <ChevronUpIcon size={13} /> : <ChevronDownIcon size={13} />}
              onClick={() => setExpanded((v) => !v)}
            >
              {expanded ? 'Collapse' : 'Expand'}
            </Button>
          </div>
        </Card>
        <Toolbar>
          <span style={{ fontSize: 12, color: 'var(--sf-ui-fg-muted)' }}>Status filter:</span>
          {(['', 'ok', 'error'] as const).map((status) => (
            <Button
              key={status || 'all'}
              size="sm"
              variant={statusFilter === status ? 'primary' : 'outline'}
              onClick={() => table.setFilters(status ? { status: [status] } : {})}
            >
              {status === '' ? 'All' : status === 'ok' ? 'Success' : 'Failed'}
            </Button>
          ))}
        </Toolbar>
        <Table compact striped>
          <TableHead>
            <TableRow>
              <TableHeader style={{ width: 38 }} />
              <TableHeader>Keyword</TableHeader>
              <TableHeader>Supplier</TableHeader>
              <TableHeader>Model</TableHeader>
              <TableHeader>Status</TableHeader>
            </TableRow>
          </TableHead>
          <TableBody>
            {table.data.length === 0 ? (
              <TableRow>
                <TableCell colSpan={5}><Empty compact title="No matching logs" /></TableCell>
              </TableRow>
            ) : (
              table.data.map((row) => (
                <TableRow key={row.id}>
                  <TableCell>
                    <Checkbox
                      aria-label={`Select ${row.id}`}
                      checked={table.selectedKeys.includes(row.id)}
                      onCheckedChange={(checked) => toggleSelect(row.id, checked)}
                    />
                  </TableCell>
                  <TableCell><code>{row.keyword}</code></TableCell>
                  <TableCell><Tag variant="outline">{row.provider}</Tag></TableCell>
                  <TableCell>{row.model}</TableCell>
                  <TableCell>
                    <Badge variant={row.status === 'ok' ? 'success' : 'danger'} dot>{row.status}</Badge>
                  </TableCell>
                </TableRow>
              ))
            )}
          </TableBody>
        </Table>
        <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', justifyContent: 'space-between', gap: 10 }}>
          <span style={{ fontSize: 12, color: 'var(--sf-ui-fg-muted)' }}>
            Page {table.page} / {pages} of {table.total} rows · selected {table.selectedKeys.length}
          </span>
          <div style={{ display: 'inline-flex', gap: 6 }}>
            <Button
              variant="outline"
              size="sm"
              disabled={table.page <= 1}
              icon={<ChevronLeftIcon size={13} />}
              onClick={() => table.onPageChange(table.page - 1)}
            >
              Previous
            </Button>
            <Button
              variant="outline"
              size="sm"
              disabled={table.page >= pages}
              onClick={() => table.onPageChange(table.page + 1)}
            >
              Next
              <ChevronRightIcon size={13} />
            </Button>
          </div>
        </div>
      </Form>
    </Card>
  );
}
```

## Native query table

Use the same `useQueryTable` hook with minimal controls and a table.

```tsx preview play=query-native
import { Field, Form, useQueryTable } from '@shikiform/react';
import {
  Badge,
  Button,
  Card,
  Checkbox,
  ChevronLeftIcon,
  ChevronRightIcon,
  Input,
  SearchIcon,
  Table,
  TableBody,
  TableCell,
  TableRow,
  Toolbar,
} from '@site/ui';

type Log = { id: string; keyword: string };
const LOGS: Log[] = Array.from({ length: 12 }, (_, i) => ({ id: String(i + 1), keyword: `${i % 2 ? 'beta' : 'alpha'}-${i + 1}` }));

export default function App() {
  const table = useQueryTable({
    initialValues: { keyword: '' },
    defaultPageSize: 5,
    params: (values: { keyword: string }) => values,
    request: async ({ params, page, pageSize }) => {
      const rows = LOGS.filter((row) => row.keyword.includes(params.keyword));
      return { data: rows.slice((page - 1) * pageSize, page * pageSize), total: rows.length };
    },
  });
  const pages = Math.max(1, Math.ceil(table.total / table.pageSize));
  const toggle = (id: string, on: boolean) =>
    table.setSelectedKeys(on ? [...table.selectedKeys, id] : table.selectedKeys.filter((key) => key !== id));
  return (
    <Card title="Minimal query table" description="Headless useQueryTable state wired to clean controls and Enter-to-search">
      <Form form={table.form} onSubmit={() => table.submit()}>
        <Toolbar>
          <Field name="keyword">
            {(field) => (
              <Input
                size="sm"
                aria-label="Keyword"
                placeholder="alpha"
                style={{ width: 160 }}
                value={field.value}
                onChange={(event) => field.onChange(event.target.value)}
              />
            )}
          </Field>
          <Button type="submit" variant="primary" size="sm" icon={<SearchIcon size={13} />}>Search</Button>
          <Button variant="outline" size="sm" onClick={table.reset}>Reset</Button>
          <Button variant="ghost" size="sm" onClick={table.refresh}>Refresh</Button>
        </Toolbar>
        <Table compact striped>
          <TableBody>
            {table.data.map((row) => (
              <TableRow key={row.id}>
                <TableCell style={{ width: 38 }}>
                  <Checkbox
                    aria-label={`Select ${row.id}`}
                    checked={table.selectedKeys.includes(row.id)}
                    onCheckedChange={(checked) => toggle(row.id, checked)}
                  />
                </TableCell>
                <TableCell><code>{row.keyword}</code></TableCell>
              </TableRow>
            ))}
          </TableBody>
        </Table>
        <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', justifyContent: 'space-between', gap: 10 }}>
          <Badge variant="default">
            {table.page} / {pages} · selected {table.selectedKeys.length}{table.loading ? ' · Loading' : ''}
          </Badge>
          <div style={{ display: 'inline-flex', gap: 6 }}>
            <Button
              variant="outline"
              size="sm"
              disabled={table.page <= 1}
              icon={<ChevronLeftIcon size={13} />}
              onClick={() => table.onPageChange(table.page - 1)}
            >
              Previous
            </Button>
            <Button
              variant="outline"
              size="sm"
              disabled={table.page >= pages}
              onClick={() => table.onPageChange(table.page + 1)}
            >
              Next
              <ChevronRightIcon size={13} />
            </Button>
          </div>
        </div>
      </Form>
    </Card>
  );
}
```

## Edit a row in a panel

Open a draft for one row. Cancel discards edits. Confirm validates and writes back to the record with the same ID.

```tsx preview play=row-edit
import { useState, type ReactNode } from 'react';
import { enUS, Field, Form, ObjectField, Watch, useFieldDecorator, useForm, type Draft } from '@shikiform/react';
import {
  Badge,
  Button,
  Card,
  FieldInput,
  Input,
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
  Toolbar,
} from '@site/ui';

type Row = { id: string; name: string; price: number };
type Values = { rows: Row[]; editor: Row & { open: boolean } };

function FieldRow({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return (
    <p className="sf-ui-form-item" style={{ margin: 0 }}>
      <span className="sf-ui-form-label">
        <label {...d.labelProps}>{d.label}: </label>
        {d.required && <span className="sf-ui-form-required" aria-hidden="true">*</span>}
      </span>
      {children}
      {d.errorProps && <small {...d.errorProps} className="sf-ui-form-error">{d.errors[0]}</small>}
    </p>
  );
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({
    messages: enUS,
    initialValues: {
      rows: [{ id: 'a', name: 'Pen', price: 12 }, { id: 'b', name: 'Ink', price: 8 }],
      editor: { open: false, id: '', name: '', price: 0 },
    },
    fields: { editor: { omit: true } }, // Exclude the editor from submitted data.
  });
  const open = (row: Row) => form.setValue('editor', { ...row, open: true });
  const close = () => form.setValue('editor.open', false);
  const save = async () => {
    let editor: Values['editor'];
    try { editor = await form.validate('editor'); } catch { return; }
    const { id, name, price } = editor;
    const index = form.getValue('rows').findIndex((row) => row.id === id);
    form.batch((d) => {
      d.rows[index].name.$value = name;
      d.rows[index].price.$value = price;
      d.editor.open.$value = false;
    });
  };
  return (
    <Card title="Read-only table + panel editor" description="editor is marked omit and written back by ID in one form.batch">
      <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))}>
        <Watch select={(d: Draft<Values>) => d.rows.$value}>
          {(rows) => (
            <Table compact striped>
              <TableHead>
                <TableRow>
                  <TableHeader>Name</TableHeader>
                  <TableHeader>Price</TableHeader>
                  <TableHeader style={{ width: 96 }}>Actions</TableHeader>
                </TableRow>
              </TableHead>
              <TableBody>
                {rows.map((row) => (
                  <TableRow key={row.id}>
                    <TableCell>{row.name}</TableCell>
                    <TableCell><Badge variant="outline">${row.price}</Badge></TableCell>
                    <TableCell>
                      <Button variant="outline" size="sm" onClick={() => open(row)}>Edit</Button>
                    </TableCell>
                  </TableRow>
                ))}
              </TableBody>
            </Table>
          )}
        </Watch>
        <Watch select={(d: Draft<Values>) => d.editor.open.$value}>
          {(visible) => visible ? (
            <fieldset className="sf-ui-card sf-ui-card--muted" style={{ padding: 16, margin: 0 }}>
              <legend style={{ padding: '0 6px', fontSize: 12, fontWeight: 600, color: 'var(--sf-ui-fg-muted)' }}>
                Edit row
              </legend>
              <div style={{ display: 'grid', gap: 12 }}>
                <ObjectField name="editor">
                  <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(180px, 1fr))', gap: 12 }}>
                    <Field<Row, 'name'> name="name" title="Name" required decorator={[FieldRow]} component={[FieldInput]} />
                    <Field<Row, 'price'> name="price" title="Price">
                      {(f) => (
                        <p className="sf-ui-form-item" style={{ margin: 0 }}>
                          <label htmlFor={f.id} className="sf-ui-form-label">{f.title}: </label>
                          <Input
                            id={f.id}
                            type="number"
                            min={0}
                            value={f.value ?? 0}
                            onChange={(e) => f.onChange(Number(e.target.value))}
                            onBlur={f.onBlur}
                          />
                        </p>
                      )}
                    </Field>
                  </div>
                </ObjectField>
                <Toolbar>
                  <Button variant="primary" size="sm" onClick={() => void save()}>Confirm</Button>
                  <Button variant="outline" size="sm" onClick={close}>Cancel</Button>
                </Toolbar>
              </div>
            </fieldset>
          ) : null}
        </Watch>
        <div>
          <Button type="submit" variant="primary">Submit without the editor</Button>
        </div>
        <pre>{result}</pre>
      </Form>
    </Card>
  );
}
```

`form.validate('editor')` validates just the `editor` section and returns its values (even though `editor` is marked `omit`, validating it directly still returns its values while keeping it out of the whole-form submit). `form.batch` updates the row and closes the panel in one render. For a completely separate form instance, see [draft forms](/en/examples/pro#draft-containers).
