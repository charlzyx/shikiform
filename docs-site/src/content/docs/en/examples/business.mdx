## SKU price matrix

Edit a merged color cell to update every row in that group, and toggle the optional channel column without losing existing prices or stock. Adjacent colors are merged only at render time while the data stays flat.

```tsx preview play=sku-editor
import { useState } from 'react';
import { enUS, Field, Form, useForm, useFormContext, useWatch, type Draft } from '@shikiform/react';
import {
  Badge,
  Button,
  Card,
  Checkbox,
  Input,
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
} from '@site/ui';

type Sku = { skuId: string; color: string; storage: string; price: number; stock: number; channel: string };
type Values = { rows: Sku[]; channel: boolean };

const SKUS: Sku[] = [
  { skuId: 'B-128', color: 'Black', storage: '128G', price: 999, stock: 12, channel: '' },
  { skuId: 'B-256', color: 'Black', storage: '256G', price: 1499, stock: 3, channel: '' },
  { skuId: 'W-128', color: 'White', storage: '128G', price: 999, stock: 0, channel: '' },
  { skuId: 'W-256', color: 'White', storage: '256G', price: 1499, stock: 8, channel: '' },
];

function computeRowSpans<T>(rows: T[], groupOf: (row: T) => string): number[] {
  const spans = new Array<number>(rows.length).fill(0);
  let i = 0;
  while (i < rows.length) {
    const key = groupOf(rows[i]!);
    let end = i + 1;
    while (end < rows.length && groupOf(rows[end]!) === key) end++;
    spans[i] = end - i;
    i = end;
  }
  return spans;
}

function SkuTable() {
  const form = useFormContext<Values>();
  const rows = useWatch((d: Draft<Values>) => d.rows.$value);
  const channel = useWatch((d: Draft<Values>) => d.channel.$value);
  const spans = computeRowSpans(rows, (row) => row.color);
  const paint = (from: string, to: string) =>
    form.setValue('rows', rows.map((row) => (row.color === from ? { ...row, color: to } : row)));
  return (
    <Card
      title="SKU price matrix"
      description="Merged color cells update the whole group while toggling optional columns preserves edits"
      extra={
        <Checkbox checked={channel} onCheckedChange={(checked) => form.setValue('channel', checked)}>
          Show channel column
        </Checkbox>
      }
    >
      <Table compact bordered>
        <TableHead>
          <TableRow>
            <TableHeader>Color</TableHeader>
            <TableHeader>Storage</TableHeader>
            <TableHeader>Price</TableHeader>
            <TableHeader>Stock</TableHeader>
            {channel && <TableHeader>Channel</TableHeader>}
          </TableRow>
        </TableHead>
        <TableBody>
          {rows.map((row, index) => (
            <TableRow key={row.skuId}>
              {spans[index]! > 0 && (
                <TableCell rowSpan={spans[index]}>
                  <Input
                    size="sm"
                    aria-label="Color"
                    style={{ width: 88 }}
                    value={row.color}
                    onChange={(e) => paint(row.color, e.target.value)}
                  />
                </TableCell>
              )}
              <TableCell><Badge variant="outline">{row.storage}</Badge></TableCell>
              <TableCell>
                <Field name={`rows.${index}.price`}>
                  {(f) => (
                    <Input
                      size="sm"
                      type="number"
                      min={0}
                      style={{ width: 84 }}
                      value={Number(f.value ?? 0)}
                      onChange={(e) => f.onChange(Number(e.target.value))}
                    />
                  )}
                </Field>
              </TableCell>
              <TableCell>
                <Field name={`rows.${index}.stock`}>
                  {(f) => (
                    <Input
                      size="sm"
                      type="number"
                      min={0}
                      style={{ width: 72 }}
                      value={Number(f.value ?? 0)}
                      onChange={(e) => f.onChange(Number(e.target.value))}
                    />
                  )}
                </Field>
              </TableCell>
              {channel && (
                <TableCell>
                  <Field name={`rows.${index}.channel`}>
                    {(f) => (
                      <Input
                        size="sm"
                        placeholder="Channel code"
                        style={{ width: 96 }}
                        value={String(f.value ?? '')}
                        onChange={(e) => f.onChange(e.target.value)}
                      />
                    )}
                  </Field>
                </TableCell>
              )}
            </TableRow>
          ))}
        </TableBody>
      </Table>
    </Card>
  );
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({ messages: enUS, initialValues: { rows: SKUS, channel: false }, fields: { channel: { omit: true } } });
  return (
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))}>
      <SkuTable />
      <div>
        <Button type="submit" variant="primary">Submit</Button>
      </div>
      <pre>{result}</pre>
    </Form>
  );
}
```

## Suppliers and products

Keep products nested under their supplier and merge cells for display. A product validator can read the parent record.

```tsx preview play=merged
import { enUS, Field, Form, useArrayField, useField, useForm, useFormContext, useWatch, type Draft } from '@shikiform/react';
import {
  Button,
  Card,
  Input,
  PlusIcon,
  Select,
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
} from '@site/ui';

type Product = { id: string; name: string; price: number };
type Supplier = { id: string; name: string; level: 'A' | 'B' | 'C'; products: Product[] };
type Values = { suppliers: Supplier[] };
/** One product per table row. Keep its location in the nested data. */
type Line = { id: string; s: number; p: number };

const LEVELS = [
  { label: 'A grade', value: 'A' },
  { label: 'B grade', value: 'B' },
  { label: 'C grade', value: 'C' },
];

let nextProductId = 4;

function computeRowSpans<T>(rows: T[], groupOf: (row: T) => string): number[] {
  const spans = new Array<number>(rows.length).fill(0);
  let i = 0;
  while (i < rows.length) {
    const key = groupOf(rows[i]!);
    let end = i + 1;
    while (end < rows.length && groupOf(rows[end]!) === key) end++;
    spans[i] = end - i;
    i = end;
  }
  return spans;
}

function TextCell() {
  const f = useField<string>();
  return (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 3 }}>
      <Input
        size="sm"
        style={{ width: 120 }}
        invalid={Boolean(f.errors[0])}
        value={f.value ?? ''}
        onChange={(e) => f.onChange(e.target.value)}
        onBlur={f.onBlur}
      />
      {f.errors[0] && <small role="alert" className="sf-ui-form-error">{f.errors[0]}</small>}
    </div>
  );
}

function NumberCell() {
  const f = useField<number>();
  return (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 3 }}>
      <Input
        size="sm"
        type="number"
        style={{ width: 96 }}
        invalid={Boolean(f.errors[0])}
        value={f.value ?? 0}
        onChange={(e) => f.onChange(Number(e.target.value))}
        onBlur={f.onBlur}
      />
      {f.errors[0] && <small role="alert" className="sf-ui-form-error">{f.errors[0]}</small>}
    </div>
  );
}

function AddProduct({ supplierIndex }: { supplierIndex: number }) {
  const products = useArrayField(`suppliers.${supplierIndex}.products`);
  return (
    <Button
      variant="outline"
      size="sm"
      icon={<PlusIcon size={12} />}
      onClick={() => products.push({ id: `p${nextProductId++}`, name: 'New product', price: 1 })}
    >
      Add product
    </Button>
  );
}

function Suppliers() {
  const form = useFormContext<Values>();
  const suppliers = useWatch((d: Draft<Values>) => d.suppliers.$value);
  const lines: Line[] = suppliers.flatMap((supplier, s) => supplier.products.map((product, p) => ({ id: product.id, s, p })));
  const spans = computeRowSpans(lines, (line) => suppliers[line.s]!.id);
  return (
    <Card
      title="Suppliers & products"
      description="Nested data with merged cells — validate duplicates via ctx.array and grade caps via ctx.parent"
    >
      <div style={{ display: 'grid', gap: 12 }}>
        <Table compact bordered>
          <TableHead>
            <TableRow>
              <TableHeader>Supplier</TableHeader>
              <TableHeader>Product</TableHeader>
              <TableHeader>Unit price</TableHeader>
            </TableRow>
          </TableHead>
          <TableBody>
            {lines.map((line, index) => (
              <TableRow key={line.id}>
                {spans[index]! > 0 && (
                  <TableCell rowSpan={spans[index]} style={{ verticalAlign: 'top' }}>
                    <div style={{ display: 'flex', flexDirection: 'column', gap: 8 }}>
                      <Field
                        name={`suppliers.${line.s}.name`}
                        component={[TextCell]}
                        rules={[
                          { required: true },
                          { validator: (value, ctx) => ((ctx.array as Supplier[] | undefined)?.filter((item) => item.name === value).length ?? 0) > 1 ? 'Duplicate supplier name' : null },
                        ]}
                      />
                      <Field name={`suppliers.${line.s}.level`}>
                        {(f) => (
                          <Select
                            size="sm"
                            style={{ width: 120 }}
                            options={LEVELS}
                            value={String(f.value ?? 'A')}
                            onChange={(next) => f.onChange(next)}
                          />
                        )}
                      </Field>
                      <AddProduct supplierIndex={line.s} />
                    </div>
                  </TableCell>
                )}
                <TableCell>
                  <Field name={`suppliers.${line.s}.products.${line.p}.name`} component={[TextCell]} />
                </TableCell>
                <TableCell>
                  <Field
                    name={`suppliers.${line.s}.products.${line.p}.price`}
                    component={[NumberCell]}
                    rules={{
                      validator: (value, ctx) => (ctx.parent?.level === 'C' && Number(value) > 1000 ? 'Grade C unit prices cannot exceed 1000' : null),
                    }}
                  />
                </TableCell>
              </TableRow>
            ))}
          </TableBody>
        </Table>
        <div>
          <Button variant="primary" size="sm" onClick={() => { void form.validate().catch(() => {}); }}>
            Validate
          </Button>
        </div>
      </div>
    </Card>
  );
}

export default function App() {
  const form = useForm<Values>({
    messages: enUS,
    initialValues: {
      suppliers: [
        { id: 'a', name: 'Supplier A', level: 'C', products: [{ id: 'p1', name: 'Screw', price: 1200 }, { id: 'p2', name: 'Nut', price: 20 }] },
        { id: 'b', name: 'Supplier A', level: 'A', products: [{ id: 'p3', name: 'Washer', price: 5000 }] },
      ],
    },
  });
  return <Form form={form}><Suppliers /></Form>;
}
```

The `void` on Validate discards the promise. `form.validate()` still resolves the whole submit value when the nested rules pass.

## Orders and line items

Expand an order to edit its line items. Recalculate each line amount and the order total. Mark shipped orders read-only at the parent path.

```tsx preview play=orders
import { Fragment, useState } from 'react';
import { enUS, ArrayField, Field, Form, defineEffects, useField, useForm, useRecord } from '@shikiform/react';
import {
  Badge,
  Button,
  Card,
  Checkbox,
  Input,
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
} from '@site/ui';

type Item = { id: string; sku: string; qty: number; price: number; line: number };
type Order = { id: string; no: string; status: string; amount: number; items: Item[] };
type Values = { editable: boolean; orders: Order[] };

const STATUS = [
  { label: 'Pending', value: 'pending' },
  { label: 'Paid', value: 'paid' },
  { label: 'Shipped', value: 'shipped' },
];
const STATUS_VARIANT = { pending: 'warning', paid: 'brand', shipped: 'success' } as const;

const effects = defineEffects<Values>(({ reaction, draft }) => {
  reaction((d) => {
    d.orders.$readOnly = d.editable.$value ? undefined : true;
  });
  reaction.each(draft.orders, (order, { d, depth }) => {
    if (depth !== 0) return; // Only handle orders at this depth.
    order.$readOnly = order.status.$value === 'shipped' ? true : undefined; // Show shipped orders in read-only mode.
    order.status.$readOnly = !d.editable.$value; // The status field can override the parent and follow the editing toggle.
    let amount = 0;
    order.items.$value.forEach((item, index) => {
      const line = item.qty * item.price;
      order.items[index].line.$value = line;
      amount += line;
    });
    order.amount.$value = amount;
  });
});

function NumberCell({ min }: { min?: number }) {
  const f = useField<number>();
  return (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 2 }}>
      {f.readOnly ? (
        <span style={{ fontWeight: 500 }}>{f.value}</span>
      ) : (
        <Input
          size="sm"
          type="number"
          min={min}
          style={{ width: 76 }}
          invalid={Boolean(f.errors[0])}
          value={f.value ?? 0}
          onChange={(e) => f.onChange(Number(e.target.value))}
          onBlur={f.onBlur}
        />
      )}
      {f.errors[0] && <small role="alert" className="sf-ui-form-error">{f.errors[0]}</small>}
    </div>
  );
}

function ItemRow() {
  const item = useRecord<Item>();
  return (
    <TableRow>
      <TableCell><code>{item.sku}</code></TableCell>
      <TableCell>
        <Field
          name="qty"
          component={[NumberCell, { min: 1 }]}
          rules={{
            validator: (value, ctx) => (ctx.parent?.status === 'pending' && Number(value) > 99 ? 'A pending line cannot exceed 99 units' : null),
          }}
        />
      </TableCell>
      <TableCell><Field name="price" component={[NumberCell, { min: 0 }]} /></TableCell>
      <TableCell><Field name="line" readOnly component={[NumberCell]} /></TableCell>
    </TableRow>
  );
}

function OrderRow({ expanded, onToggle }: { expanded: boolean; onToggle: () => void }) {
  const record = useRecord<Order>();
  return (
    <Fragment>
      <TableRow>
        <TableCell style={{ width: 42 }}>
          <Button variant="ghost" size="sm" onClick={onToggle}>{expanded ? '▾' : '▸'}</Button>
        </TableCell>
        <TableCell><strong>{record.no}</strong></TableCell>
        <TableCell>
          <Field name="status" enum={STATUS}>
            {(f) => f.readOnly ? (
              <Badge variant={STATUS_VARIANT[String(f.value) as keyof typeof STATUS_VARIANT] ?? 'default'} dot>
                {f.options.find((o) => o.value === f.value)?.label ?? String(f.value)}
              </Badge>
            ) : (
              <select
                style={{ minWidth: 112, minHeight: 30, padding: '3px 28px 3px 10px' }}
                value={String(f.value ?? '')}
                onChange={(e) => f.onChange(e.target.value)}
              >
                {f.options.map((opt) => (
                  <option key={String(opt.value)} value={String(opt.value)}>{opt.label}</option>
                ))}
              </select>
            )}
          </Field>
        </TableCell>
        <TableCell>
          <Badge variant="brand">
            <Field name="amount" readOnly component={[NumberCell]} />
          </Badge>
        </TableCell>
      </TableRow>
      {expanded && (
        <TableRow style={{ background: 'var(--sf-ui-muted)' }}>
          <TableCell colSpan={4} style={{ padding: '10px 14px 10px 36px' }}>
            <Table compact>
              <TableHead>
                <TableRow>
                  <TableHeader>Product</TableHeader>
                  <TableHeader>Quantity</TableHeader>
                  <TableHeader>Unit price</TableHeader>
                  <TableHeader>Line total</TableHeader>
                </TableRow>
              </TableHead>
              <TableBody>
                <ArrayField name="items" rowKey="id">
                  {() => <ItemRow />}
                </ArrayField>
              </TableBody>
            </Table>
          </TableCell>
        </TableRow>
      )}
    </Fragment>
  );
}

export default function App() {
  const [expanded, setExpanded] = useState<Record<string, boolean>>({ o1: true });
  const form = useForm<Values>({
    messages: enUS,
    initialValues: {
      editable: true,
      orders: [
        { id: 'o1', no: 'A001', status: 'pending', amount: 0, items: [{ id: 'i1', sku: 'pen', qty: 1, price: 10, line: 0 }, { id: 'i2', sku: 'ink', qty: 2, price: 8, line: 0 }] },
        { id: 'o2', no: 'A002', status: 'shipped', amount: 0, items: [{ id: 'i3', sku: 'box', qty: 1, price: 20, line: 0 }] },
      ],
    },
    fields: { editable: { omit: true } },
    effects: [effects],
  });
  const toggle = (id: string) => setExpanded((prev) => ({ ...prev, [id]: !prev[id] }));
  return (
    <Form form={form}>
      <Card
        title="Orders & line items"
        description="Derives per-order totals; shipped orders inherit read-only while status stays switchable"
        extra={
          <Field<Values, 'editable'> name="editable" title="Editable">
            {(f) => (
              <Checkbox id={f.id} checked={Boolean(f.value)} onCheckedChange={(checked) => f.onChange(checked)}>
                {f.title}
              </Checkbox>
            )}
          </Field>
        }
      >
        <Table compact>
          <TableHead>
            <TableRow>
              <TableHeader style={{ width: 42 }} />
              <TableHeader>Order number</TableHeader>
              <TableHeader>Status</TableHeader>
              <TableHeader>Total</TableHeader>
            </TableRow>
          </TableHead>
          <TableBody>
            <ArrayField name="orders" rowKey="id">
              {({ key }) => <OrderRow expanded={Boolean(expanded[key])} onToggle={() => toggle(key)} />}
            </ArrayField>
          </TableBody>
        </Table>
      </Card>
    </Form>
  );
}
```

## Policy Matrix Form (Visibility Matrix + Adapter + Risk Gate)

**Scenario**: A multi-mode campaign/release policy form. Switching the policy type shows only its matching section while automatically marking inactive fields `$omit` so they are excluded from validation and submit payloads; `fromApi / toApi` keeps UI units decoupled from backend DTOs, and high-budget submissions pass through a confirmation gate.

```tsx preview
import { useState } from 'react';
import { defineEffects, enUS, Field, Form, Submit, useForm, useWatch, type Draft } from '@shikiform/react';
import {
  Alert,
  Badge,
  Button,
  Card,
  Dialog,
  FieldInput,
  FieldNumber,
  FieldRadioGroup,
  siteKit,
} from '@site/ui';

type PolicyType = 'flash' | 'bundle' | 'referral';
type PolicyDto = {
  policy_id: string;
  policy_type: PolicyType;
  title: string;
  budget_cents: number;
  flash_cap?: number;
  bundle_min_items?: number;
  referral_bonus?: number;
};
type PolicyValues = {
  type: PolicyType;
  title: string;
  budget: number;
  flashCap: number;
  bundleMinItems: number;
  referralBonus: number;
};

const SHOW_MAP: Record<PolicyType, { flashCap: boolean; bundleMinItems: boolean; referralBonus: boolean }> = {
  flash: { flashCap: true, bundleMinItems: false, referralBonus: false },
  bundle: { flashCap: false, bundleMinItems: true, referralBonus: false },
  referral: { flashCap: false, bundleMinItems: false, referralBonus: true },
};

const fromApi = (dto: PolicyDto): PolicyValues => ({
  type: dto.policy_type,
  title: dto.title,
  budget: Math.round(dto.budget_cents / 100),
  flashCap: dto.flash_cap ?? 500,
  bundleMinItems: dto.bundle_min_items ?? 2,
  referralBonus: dto.referral_bonus ?? 30,
});

const toApi = (values: Partial<PolicyValues>, id: string): PolicyDto => ({
  policy_id: id,
  policy_type: values.type ?? 'flash',
  title: (values.title ?? '').trim(),
  budget_cents: Math.round((values.budget ?? 0) * 100),
  ...(values.flashCap !== undefined ? { flash_cap: values.flashCap } : {}),
  ...(values.bundleMinItems !== undefined ? { bundle_min_items: values.bundleMinItems } : {}),
  ...(values.referralBonus !== undefined ? { referral_bonus: values.referralBonus } : {}),
});

const policyEffects = defineEffects<PolicyValues>(({ reaction }) => {
  reaction((d) => {
    const show = SHOW_MAP[d.type.$value] ?? SHOW_MAP.flash;
    d.flashCap.$visible = show.flashCap;
    d.flashCap.$omit = !show.flashCap;
    d.bundleMinItems.$visible = show.bundleMinItems;
    d.bundleMinItems.$omit = !show.bundleMinItems;
    d.referralBonus.$visible = show.referralBonus;
    d.referralBonus.$omit = !show.referralBonus;
  });
});

function RiskBanner() {
  const budget = useWatch((d: Draft<PolicyValues>) => d.budget.$value);
  if ((budget ?? 0) < 50000) return null;
  return (
    <Alert variant="warning" title="High-budget risk gate">
      Budget is ≥ $50,000 — submitting opens an extra confirmation dialog.
    </Alert>
  );
}

const INITIAL_DTO: PolicyDto = {
  policy_id: 'PL-2026-01',
  policy_type: 'flash',
  title: 'Spring Flash Campaign',
  budget_cents: 6000000,
  flash_cap: 300,
};

export default function App() {
  const [pendingDto, setPendingDto] = useState<PolicyDto | null>(null);
  const [savedDto, setSavedDto] = useState<PolicyDto | null>(null);

  const form = useForm<PolicyValues>({
    messages: enUS,
    initialValues: fromApi(INITIAL_DTO),
    fields: {
      type: {
        title: 'Policy type',
        required: true,
        enum: [
          { label: 'Flash sale', value: 'flash' },
          { label: 'Bundle deal', value: 'bundle' },
          { label: 'Referral bonus', value: 'referral' },
        ] as const,
      },
      title: { title: 'Policy title', required: true },
      budget: { title: 'Budget cap ($)', required: true, rules: { min: 100 } },
      flashCap: { title: 'Per-user cap', required: true, rules: { min: 1 } },
      bundleMinItems: { title: 'Min bundle items', required: true, rules: { min: 2 } },
      referralBonus: { title: 'Bonus per referral ($)', required: true, rules: { min: 1 } },
    },
    effects: [policyEffects],
  });

  return (
    <Card
      title="Campaign policy"
      description="Visibility matrix + fromApi / toApi adapter + high-budget confirmation gate"
      extra={<Badge variant="outline">{INITIAL_DTO.policy_id}</Badge>}
      style={{ maxWidth: 520 }}
    >
      <Form
        form={form}
        kit={siteKit}
        onSubmit={(values) => {
          const dto = toApi(values, INITIAL_DTO.policy_id);
          if (dto.budget_cents >= 5000000) {
            setPendingDto(dto);
            return;
          }
          setSavedDto(dto);
        }}
      >
        <Field<PolicyValues, 'type'> name="type" component={[FieldRadioGroup]} />
        <Field<PolicyValues, 'title'> name="title" component={[FieldInput]} />
        <Field<PolicyValues, 'budget'> name="budget" component={[FieldNumber, { min: 100, step: 1000, prefix: '$' }]} />
        <RiskBanner />
        <Field<PolicyValues, 'flashCap'> name="flashCap" component={[FieldNumber, { min: 1 }]} />
        <Field<PolicyValues, 'bundleMinItems'> name="bundleMinItems" component={[FieldNumber, { min: 2 }]} />
        <Field<PolicyValues, 'referralBonus'> name="referralBonus" component={[FieldNumber, { min: 1, prefix: '$' }]} />
        <div>
          <Submit>
            {(s) => <Button type="submit" variant="primary" disabled={s.disabled}>Publish policy</Button>}
          </Submit>
        </div>
        <Dialog
          open={Boolean(pendingDto)}
          onOpenChange={(open) => { if (!open) setPendingDto(null); }}
          title="Confirm high-budget publish"
          description={pendingDto ? `Confirm publishing a $${(pendingDto.budget_cents / 100).toLocaleString()} campaign budget?` : undefined}
          size="sm"
          footer={
            <>
              <Button variant="outline" size="sm" onClick={() => setPendingDto(null)}>Cancel</Button>
              <Button
                variant="primary"
                size="sm"
                onClick={() => {
                  if (pendingDto) setSavedDto(pendingDto);
                  setPendingDto(null);
                }}
              >
                Confirm
              </Button>
            </>
          }
        >
          {pendingDto && (
            <pre style={{ margin: 0 }}>{JSON.stringify(pendingDto, null, 2)}</pre>
          )}
        </Dialog>
        {savedDto && (
          <Card variant="muted" padding="sm" title="Submitted DTO payload">
            <pre style={{ margin: 0 }}>{JSON.stringify(savedDto, null, 2)}</pre>
          </Card>
        )}
      </Form>
    </Card>
  );
}
```

## Shared Single-Drawer Table + Live Inspector (`scopeKey` + `onValuesChange`)

**Scenario**: Every table row shares a single `useShadowForm({ scopeKey })` overlay instance (instead of mounting one drawer per row). Switching workspaces cancels any open draft via `scopeKey`, while the live inspector below uses `useForm({ values, onValuesChange })` to sync edits immediately without a submit button.

```tsx preview
import { useState } from 'react';
import { enUS, Field, Form, useForm, useShadowForm } from '@shikiform/react';
import {
  Badge,
  Button,
  Card,
  Drawer,
  FieldInput,
  FieldNumber,
  FieldSwitch,
  siteKit,
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
  Tabs,
  TabsList,
  TabsTab,
} from '@site/ui';

type Rule = { id: string; name: string; weight: number; enabled: boolean };

const WORKSPACE_RULES: Record<string, Rule[]> = {
  'ws-east': [
    { id: 'R-01', name: 'New user welcome perk', weight: 80, enabled: true },
    { id: 'R-02', name: 'Weekend member boost', weight: 50, enabled: false },
  ],
  'ws-west': [
    { id: 'R-09', name: 'Regional free shipping', weight: 90, enabled: true },
  ],
};

export default function App() {
  const [workspace, setWorkspace] = useState<'ws-east' | 'ws-west'>('ws-east');
  const [store, setStore] = useState(WORKSPACE_RULES);
  const [selectedId, setSelectedId] = useState('R-01');
  const rows = store[workspace] ?? [];
  const activeRow = rows.find((r) => r.id === selectedId) ?? rows[0];

  const editor = useShadowForm<Rule, { id: string }>({
    scopeKey: workspace,
    formOptions: { messages: enUS },
    fields: {
      name: { title: 'Rule name', required: true },
      weight: { title: 'Priority weight', required: true, rules: { min: 1, max: 100 } },
      enabled: { title: 'Enabled' },
    },
    onOk: (values, ctx) => {
      if (!ctx) return;
      setStore((prev) => ({
        ...prev,
        [workspace]: (prev[workspace] ?? []).map((r) => (r.id === ctx.id ? { ...r, ...values } : r)),
      }));
    },
  });

  const inspector = useForm<Rule>({
    messages: enUS,
    values: activeRow,
    onValuesChange: (next) => {
      if (!activeRow) return;
      setStore((prev) => ({
        ...prev,
        [workspace]: (prev[workspace] ?? []).map((r) => (r.id === activeRow.id ? { ...r, ...next } : r)),
      }));
    },
  });

  return (
    <Card
      title="Multi-tenant rules"
      description="Single shared drawer draft (auto-cancelled by scopeKey) + instant live inspector"
      extra={
        <Tabs
          variant="segmented"
          value={workspace}
          onValueChange={(next) => {
            const ws = next as 'ws-east' | 'ws-west';
            setWorkspace(ws);
            setSelectedId(ws === 'ws-east' ? 'R-01' : 'R-09');
          }}
        >
          <TabsList className="sf-ui-tabs-list" style={{ marginBottom: 0 }}>
            <TabsTab value="ws-east" className="sf-ui-tabs-tab">East</TabsTab>
            <TabsTab value="ws-west" className="sf-ui-tabs-tab">West</TabsTab>
          </TabsList>
        </Tabs>
      }
    >
      <div style={{ display: 'grid', gap: 14 }}>
        <Table compact striped>
          <TableHead>
            <TableRow>
              <TableHeader style={{ width: 76 }}>ID</TableHeader>
              <TableHeader>Rule name</TableHeader>
              <TableHeader style={{ width: 76 }}>Weight</TableHeader>
              <TableHeader style={{ width: 96 }}>Status</TableHeader>
              <TableHeader style={{ width: 196 }}>Actions</TableHeader>
            </TableRow>
          </TableHead>
          <TableBody>
            {rows.map((row) => (
              <TableRow key={row.id}>
                <TableCell><code>{row.id}</code></TableCell>
                <TableCell>{row.name}</TableCell>
                <TableCell><Badge variant="outline">{row.weight}</Badge></TableCell>
                <TableCell>
                  <Badge variant={row.enabled ? 'success' : 'default'} dot>
                    {row.enabled ? 'Active' : 'Paused'}
                  </Badge>
                </TableCell>
                <TableCell>
                  <div style={{ display: 'inline-flex', gap: 6 }}>
                    <Button
                      variant={activeRow?.id === row.id ? 'secondary' : 'ghost'}
                      size="sm"
                      onClick={() => setSelectedId(row.id)}
                    >
                      Inspect
                    </Button>
                    <Button variant="outline" size="sm" onClick={() => void editor.open(row, { id: row.id })}>
                      Edit in drawer
                    </Button>
                  </div>
                </TableCell>
              </TableRow>
            ))}
          </TableBody>
        </Table>

        <Drawer
          open={editor.visible}
          onOpenChange={(open) => { if (!open) editor.cancel(); }}
          title={`Shared drawer · ${editor.context?.id ?? ''}`}
          description="Switching workspace closes and discards this draft via scopeKey"
          footer={
            <>
              <Button variant="outline" size="sm" onClick={() => editor.cancel()}>Cancel</Button>
              <Button variant="primary" size="sm" onClick={() => void editor.confirm()}>Save</Button>
            </>
          }
        >
          <Form form={editor.form} kit={siteKit}>
            <Field<Rule, 'name'> name="name" component={[FieldInput]} />
            <Field<Rule, 'weight'> name="weight" component={[FieldNumber, { min: 1, max: 100 }]} />
            <Field<Rule, 'enabled'> name="enabled" component={[FieldSwitch]} />
          </Form>
        </Drawer>

        {activeRow && (
          <Card
            variant="muted"
            padding="sm"
            title={`Live inspector · ${activeRow.id}`}
            description="Syncs changes immediately via values + onValuesChange without a submit button"
          >
            <Form form={inspector} kit={siteKit}>
              <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(170px, 1fr))', gap: 12, alignItems: 'end' }}>
                <Field<Rule, 'name'> name="name" title="Rule name" component={[FieldInput]} />
                <Field<Rule, 'weight'> name="weight" title="Priority weight" component={[FieldNumber, { min: 1, max: 100 }]} />
                <Field<Rule, 'enabled'> name="enabled" title="Enabled" component={[FieldSwitch]} />
              </div>
            </Form>
          </Card>
        )}
      </div>
    </Card>
  );
}
```
