## Login validation

Submit an empty form, an invalid email and a short password. The submit handler runs only when the field rules pass.

```tsx preview
import { useState } from 'react';
import { enUS, Form, Submit, typed, useField, useForm } from '@shikiform/react';

type Values = { account: string; email: string; password: string };

function TextInput({ type = 'text' }: { type?: string }) {
  const f = useField<string>();
  return (
    <p>
      <label htmlFor={f.id}>{f.title}：</label>
      <input
        id={f.id}
        type={type}
        value={f.value ?? ''}
        onChange={(e) => f.onChange(e.target.value)}
        onBlur={f.onBlur}
      />
      {f.errors[0] && <small role="alert"> {f.errors[0]}</small>}
    </p>
  );
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({ messages: enUS, initialValues: { account: '', email: '', password: '' } });
  const F = typed(form);
  return (
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))}>
      <F.Field name="account" title="Account" required component={[TextInput]} />
      <F.Field
        name="email"
        title="Email"
        required
        rules={{ format: 'email', message: "Invalid email address" }}
        component={[TextInput]}
      />
      <F.Field
        name="password"
        title="Password"
        required
        rules={{ minLength: 8, message: "At least 8 characters" }}
        component={[TextInput, { type: 'password' }]}
      />
      <Submit>Log in</Submit>
      <pre aria-live="polite">{result}</pre>
    </Form>
  );
}
```

## Cross-field and asynchronous validation

Compare passwords with `ctx.record`. Pass `ctx.signal` to a request so typing a new value cancels obsolete work. When simulating a request with a timer, clean it up and settle the promise on abort.

```tsx preview
import { useState } from 'react';
import { enUS, Form, Submit, typed, useField, useForm } from '@shikiform/react';

type Values = { username: string; password: string; confirm: string };

function lookupName(username: string, signal?: AbortSignal): Promise<string | undefined> {
  return new Promise((resolve) => {
    const timer = setTimeout(() => {
      resolve(username.trim().toLowerCase() === 'admin' ? "Already taken" : undefined);
    }, 400);
    signal?.addEventListener(
      'abort',
      () => {
        clearTimeout(timer);
        resolve(undefined);
      },
      { once: true },
    );
  });
}

function TextInput({ type = 'text' }: { type?: string }) {
  const f = useField<string>();
  return (
    <p>
      <label htmlFor={f.id}>{f.title}：</label>
      <input
        id={f.id}
        type={type}
        value={f.value ?? ''}
        onChange={(e) => f.onChange(e.target.value)}
        onBlur={f.onBlur}
      />
      {f.validating && <small style={{ marginLeft: 6, opacity: 0.7 }}>Validating…</small>}
      {f.errors[0] && <small role="alert"> {f.errors[0]}</small>}
    </p>
  );
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { username: '', password: '', confirm: '' },
    asyncDebounceMs: 300,
  });
  const F = typed(form);
  return (
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))}>
      <F.Field
        name="username"
        title="Username"
        required
        rules={{ validator: (value, ctx) => lookupName(String(value ?? ''), ctx.signal) }}
        component={[TextInput]}
      />
      <F.Field
        name="password"
        title="Password"
        required
        rules={{ minLength: 8, message: "At least 8 characters" }}
        component={[TextInput, { type: 'password' }]}
      />
      <F.Field
        name="confirm"
        title="Confirm password"
        required
        omit
        rules={{
          validator: (value, ctx) => (value && value !== ctx.record?.password ? "Passwords do not match" : undefined),
        }}
        component={[TextInput, { type: 'password' }]}
      />
      <Submit>Register</Submit>
      <pre aria-live="polite">{result}</pre>
    </Form>
  );
}
```

## Registration with a schema

Pass a Zod schema to `validate` to check nested records and compare fields across the form.

```tsx preview play=schema
import { useState } from 'react';
import { z } from 'zod';
import { isFieldErrors } from '@shikiform/core';
import { enUS, Field, Form, ObjectField, useField, useForm } from '@shikiform/react';

const schema = z
  .object({
    email: z.string().email("Invalid email address"),
    password: z.string().min(6, "At least 6 characters"),
    confirm: z.string(),
    nick: z.string().min(1, "Nickname is required"),
    address: z.object({ city: z.string().min(1, "City is required"), street: z.string() }),
  })
  .refine((values) => values.password === values.confirm, { path: ['confirm'], message: "Passwords do not match" });

type Values = z.infer<typeof schema>;

function TextInput({ type = 'text' }: { type?: string }) {
  const f = useField<string>();
  return (
    <p>
      <label htmlFor={f.id}>{f.title}：</label>
      <input
        id={f.id}
        type={type}
        value={f.value ?? ''}
        onChange={(e) => f.onChange(e.target.value)}
        onBlur={f.onBlur}
      />
      {f.errors[0] && <small role="alert" style={{ color: 'crimson' }}> {f.errors[0]}</small>}
    </p>
  );
}

export default function App() {
  const [out, setOut] = useState('');
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { email: '', password: '', confirm: '', nick: '', address: { city: '', street: '' } },
    validate: schema,
  });
  const submit = async () => {
    try {
      const values = await form.submit();
      setOut(`Submitted: ${JSON.stringify(values)}`);
    } catch (error) {
      const errors = isFieldErrors(error) ? error : [];
      setOut(`Errors: ${errors.map((item) => item.path).join(', ')}`);
    }
  };
  const setNick = (mode: 'visible' | 'hidden' | 'omit') => form.batch((d) => {
    d.nick.$visible = mode !== 'hidden';
    d.nick.$omit = mode === 'omit';
  });
  return (
    <Form form={form}>
      <Field name="email" title="Email" component={[TextInput]} />
      <Field name="password" title="Password" component={[TextInput, { type: 'password' }]} />
      <Field name="confirm" title="Confirm password" component={[TextInput, { type: 'password' }]} />
      <Field name="nick" title="Nickname" component={[TextInput]} />
      <ObjectField name="address">
        <Field name="city" title="City" component={[TextInput]} />
        <Field name="street" title="Street" component={[TextInput]} />
      </ObjectField>
      <div style={{ display: 'flex', flexWrap: 'wrap', gap: 8 }}>
        <button type="button" onClick={submit}>Submit</button>
        <button type="button" onClick={() => setNick('hidden')}>Hide nickname</button>
        <button type="button" onClick={() => setNick('omit')}>Omit nickname</button>
        <button type="button" onClick={() => setNick('visible')}>Show nickname</button>
      </div>
      <pre aria-live="polite">{out}</pre>
    </Form>
  );
}
```

## Visibility and submission

Toggle the contact group's flags and compare what renders, what validates and what reaches the submit callback.

```tsx preview play=display
import { useState } from 'react';
import { isFieldErrors } from '@shikiform/core';
import { enUS, Field, Form, ObjectField, defineEffects, useField, useForm } from '@shikiform/react';

type Mode = 'visible' | 'hidden' | 'omit' | 'readOnly';
type Values = { mode: Mode; contact: { email: string; phone: string } };

const effects = defineEffects<Values>(({ reaction }) => {
  reaction((d) => {
    const mode = d.mode.$value;
    d.contact.$visible = mode !== 'hidden';
    d.contact.$omit = mode === 'omit';
    d.contact.$readOnly = mode === 'readOnly';
  });
});

function TextInput() {
  const f = useField<string>();
  return (
    <p>
      <label htmlFor={f.id}>{f.title}：</label>
      {f.readOnly ? (
        <span>{f.value || '-'}</span>
      ) : (
        <input id={f.id} value={f.value ?? ''} onChange={(e) => f.onChange(e.target.value)} onBlur={f.onBlur} />
      )}
      {f.errors[0] && <small role="alert" style={{ color: 'crimson' }}> {f.errors[0]}</small>}
    </p>
  );
}

export default function App() {
  const [out, setOut] = useState('');
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { mode: 'visible', contact: { email: '', phone: '13800000000' } },
    fields: { 'contact.email': { required: true } },
    effects: [effects],
  });
  const submit = async () => {
    try {
      const values = await form.submit();
      setOut(`Submitted: ${JSON.stringify(values)}`);
    } catch (error) {
      const errors = isFieldErrors(error) ? error : [];
      setOut(`Blocked: ${errors.map((item) => item.path).join(', ')}`);
    }
  };
  return (
    <Form form={form}>
      <Field<Values, 'mode'> name="mode" title="contact" omit enum={['visible', 'hidden', 'omit', 'readOnly']}>
        {(f) => (
          <div style={{ display: 'flex', gap: 12, marginBottom: 12 }}>
            <span>{f.title}：</span>
            {f.options.map((opt) => (
              <label key={String(opt.value)}>
                <input
                  type="radio"
                  name={f.id}
                  checked={f.value === opt.value}
                  onChange={() => f.onChange(opt.value as Mode)}
                />{' '}
                {opt.label}
              </label>
            ))}
          </div>
        )}
      </Field>
      <ObjectField name="contact">
        <Field name="email" title="Email" component={[TextInput]} />
        <Field name="phone" title="Phone" component={[TextInput]} />
      </ObjectField>
      <button type="button" onClick={submit}>Submit</button>
      <pre aria-live="polite">{out}</pre>
    </Form>
  );
}
```

## Tabs and unmounted fields

Declare required fields in the engine so submitting still checks inputs on a tab that has unmounted.

```tsx preview play=tabs
import { useState } from 'react';
import { isFieldErrors } from '@shikiform/core';
import { enUS, Field, Form, Watch, useField, useForm, type Draft } from '@shikiform/react';

type Values = { tab: 'base' | 'extra'; name: string; code: string; note: string };

function TextInput() {
  const f = useField<string>();
  return (
    <p>
      <label htmlFor={f.id}>{f.title}：</label>
      <input id={f.id} value={f.value ?? ''} onChange={(e) => f.onChange(e.target.value)} onBlur={f.onBlur} />
      {f.errors[0] && <small role="alert" style={{ color: 'crimson' }}> {f.errors[0]}</small>}
    </p>
  );
}

export default function App() {
  const [out, setOut] = useState('');
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { tab: 'base', name: '', code: '', note: '' },
    fields: {
      name: { required: true }, // Rules in the model validate mounted and unmounted fields.
      code: { required: true },
    },
  });
  const submit = async () => {
    try {
      await form.submit();
      setOut("Passed");
    } catch (error) {
      const errors = isFieldErrors(error) ? error : [];
      setOut(`Blocked: ${errors.map((item) => item.path).join(', ')}`);
    }
  };
  return (
    <Form form={form}>
      <Field<Values, 'tab'> name="tab" omit enum={[{ label: "Basic", value: 'base' }, { label: "Extra", value: 'extra' }]}>
        {(f) => (
          <div style={{ display: 'flex', gap: 8, marginBottom: 12 }}>
            {f.options.map((opt) => (
              <button
                key={String(opt.value)}
                type="button"
                style={{ fontWeight: f.value === opt.value ? 700 : 400 }}
                onClick={() => f.onChange(opt.value as Values['tab'])}
              >
                {opt.label}
              </button>
            ))}
          </div>
        )}
      </Field>
      <Watch select={(d: Draft<Values>) => d.tab.$value}>
        {(tab) => tab === 'base' ? (
          <Field name="name" title="Name" component={[TextInput]} />
        ) : (
          <>
            <Field name="code" title="Code" component={[TextInput]} />
            {/* JSX rules apply while mounted. */}
            <Field name="note" title="Notes" required component={[TextInput]} />
          </>
        )}
      </Watch>
      <button type="button" onClick={submit}>Submit</button>
      <pre aria-live="polite">{out}</pre>
    </Form>
  );
}
```
