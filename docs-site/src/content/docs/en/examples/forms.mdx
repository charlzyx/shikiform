## Login validation

Submit an empty form, an invalid email and a short password. The submit handler runs only when the field rules pass.

```tsx preview
import { useState, type ReactNode } from 'react';
import { enUS, Form, Submit, typed, useFieldDecorator, useForm } from '@shikiform/react';
import { Button, Card, FieldInput } from '@site/ui';

type Values = { account: string; email: string; password: string };

function FieldRow({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return (
    <p className="sf-ui-form-item" style={{ margin: 0 }}>
      <span className="sf-ui-form-label">
        <label {...d.labelProps}>{d.label}: </label>
        {d.required && <span className="sf-ui-form-required" aria-hidden="true">*</span>}
      </span>
      {children}
      {d.errorProps && <small {...d.errorProps} className="sf-ui-form-error">{d.errors[0]}</small>}
    </p>
  );
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({ messages: enUS, initialValues: { account: '', email: '', password: '' } });
  const F = typed(form);
  return (
    <Card title="Sign in" description="onSubmit runs only when all field rules pass" style={{ maxWidth: 440 }}>
      <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))}>
        <F.Field
          name="account"
          title="Account"
          required
          decorator={[FieldRow]}
          component={[FieldInput, { placeholder: 'e.g. alice' }]}
        />
        <F.Field
          name="email"
          title="Email"
          required
          rules={{ format: 'email', message: 'Invalid email address' }}
          decorator={[FieldRow]}
          component={[FieldInput, { placeholder: 'alice@example.com' }]}
        />
        <F.Field
          name="password"
          title="Password"
          required
          rules={{ minLength: 8, message: 'At least 8 characters' }}
          decorator={[FieldRow]}
          component={[FieldInput, { type: 'password', placeholder: 'At least 8 characters' }]}
        />
        <div>
          <Submit>
            {(s) => (
              <Button type="submit" variant="primary" disabled={s.disabled} loading={s.submitting}>
                Log in
              </Button>
            )}
          </Submit>
        </div>
        <pre aria-live="polite">{result}</pre>
      </Form>
    </Card>
  );
}
```

## Cross-field and asynchronous validation

Compare passwords with `ctx.record`. Pass `ctx.signal` to a request so typing a new value cancels obsolete work. When simulating a request with a timer, clean it up and settle the promise on abort.

```tsx preview
import { useState, type ReactNode } from 'react';
import { enUS, Form, Submit, typed, useField, useFieldDecorator, useForm } from '@shikiform/react';
import { Button, Card, FieldInput, LoaderIcon } from '@site/ui';

type Values = { username: string; password: string; confirm: string };

function lookupName(username: string, signal?: AbortSignal): Promise<string | undefined> {
  return new Promise((resolve) => {
    const timer = setTimeout(() => {
      resolve(username.trim().toLowerCase() === 'admin' ? 'Already taken' : undefined);
    }, 400);
    signal?.addEventListener(
      'abort',
      () => {
        clearTimeout(timer);
        resolve(undefined);
      },
      { once: true },
    );
  });
}

function FieldRow({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  const f = useField();
  return (
    <p className="sf-ui-form-item" style={{ margin: 0 }}>
      <span className="sf-ui-form-label">
        <label {...d.labelProps}>{d.label}: </label>
        {d.required && <span className="sf-ui-form-required" aria-hidden="true">*</span>}
      </span>
      {children}
      {f.validating && (
        <small className="sf-ui-form-desc" style={{ display: 'inline-flex', alignItems: 'center', gap: 4 }}>
          <LoaderIcon size={12} /> Validating…
        </small>
      )}
      {d.errorProps && <small {...d.errorProps} className="sf-ui-form-error">{d.errors[0]}</small>}
    </p>
  );
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { username: '', password: '', confirm: '' },
    asyncDebounceMs: 300,
  });
  const F = typed(form);
  return (
    <Card title="Create account" description='Try "admin" for async conflict; confirm password is omitted on submit' style={{ maxWidth: 440 }}>
      <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))}>
        <F.Field
          name="username"
          title="Username"
          required
          rules={{ validator: (value, ctx) => lookupName(String(value ?? ''), ctx.signal) }}
          decorator={[FieldRow]}
          component={[FieldInput, { placeholder: 'Try admin or alice' }]}
        />
        <F.Field
          name="password"
          title="Password"
          required
          rules={{ minLength: 8, message: 'At least 8 characters' }}
          decorator={[FieldRow]}
          component={[FieldInput, { type: 'password', placeholder: 'At least 8 characters' }]}
        />
        <F.Field
          name="confirm"
          title="Confirm password"
          required
          omit
          rules={{
            validator: (value, ctx) => (value && value !== ctx.record?.password ? 'Passwords do not match' : undefined),
          }}
          decorator={[FieldRow]}
          component={[FieldInput, { type: 'password', placeholder: 'Re-enter password' }]}
        />
        <div>
          <Submit>
            {(s) => (
              <Button type="submit" variant="primary" disabled={s.disabled} loading={s.submitting}>
                Register
              </Button>
            )}
          </Submit>
        </div>
        <pre aria-live="polite">{result}</pre>
      </Form>
    </Card>
  );
}
```

## Registration with a schema

Pass a Zod schema to `validate` to check nested records and compare fields across the form.

```tsx preview play=schema
import { useState, type ReactNode } from 'react';
import { z } from 'zod';
import { isFieldErrors } from '@shikiform/core';
import { enUS, Field, Form, ObjectField, useFieldDecorator, useForm } from '@shikiform/react';
import { Badge, Button, Card, FieldInput, Toolbar } from '@site/ui';

const schema = z
  .object({
    email: z.string().email('Invalid email address'),
    password: z.string().min(6, 'At least 6 characters'),
    confirm: z.string(),
    nick: z.string().min(1, 'Nickname is required'),
    address: z.object({ city: z.string().min(1, 'City is required'), street: z.string() }),
  })
  .refine((values) => values.password === values.confirm, { path: ['confirm'], message: 'Passwords do not match' });

type Values = z.infer<typeof schema>;

function FieldRow({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return (
    <p className="sf-ui-form-item" style={{ margin: 0 }}>
      <label {...d.labelProps} className="sf-ui-form-label">
        <span>{d.label}: </span>
      </label>
      {children}
      {d.errorProps && <small {...d.errorProps} className="sf-ui-form-error">{d.errors[0]}</small>}
    </p>
  );
}

export default function App() {
  const [out, setOut] = useState('');
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { email: '', password: '', confirm: '', nick: '', address: { city: '', street: '' } },
    validate: schema,
  });
  const submit = async () => {
    try {
      const values = await form.submit();
      setOut(`Submitted: ${JSON.stringify(values)}`);
    } catch (error) {
      const errors = isFieldErrors(error) ? error : [];
      setOut(`Errors: ${errors.map((item) => item.path).join(', ')}`);
    }
  };
  const setNick = (mode: 'visible' | 'hidden' | 'omit') => form.batch((d) => {
    d.nick.$visible = mode !== 'hidden';
    d.nick.$omit = mode === 'omit';
  });
  return (
    <Card
      title="Registration"
      description="Whole-form Standard Schema (Zod) validation with nested address paths"
      extra={<Badge variant="brand">Zod</Badge>}
      style={{ maxWidth: 520 }}
    >
      <Form form={form}>
        <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(190px, 1fr))', gap: 12 }}>
          <Field name="email" title="Email" decorator={[FieldRow]} component={[FieldInput, { placeholder: 'a@b.co' }]} />
          <Field name="nick" title="Nickname" decorator={[FieldRow]} component={[FieldInput, { placeholder: 'ada' }]} />
          <Field name="password" title="Password" decorator={[FieldRow]} component={[FieldInput, { type: 'password', placeholder: 'At least 6 characters' }]} />
          <Field name="confirm" title="Confirm password" decorator={[FieldRow]} component={[FieldInput, { type: 'password', placeholder: 'Re-enter password' }]} />
        </div>
        <Card variant="muted" padding="sm" title="Shipping address">
          <ObjectField name="address">
            <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(180px, 1fr))', gap: 12 }}>
              <Field name="city" title="City" decorator={[FieldRow]} component={[FieldInput, { placeholder: 'Hangzhou' }]} />
              <Field name="street" title="Street" decorator={[FieldRow]} component={[FieldInput, { placeholder: 'Optional' }]} />
            </div>
          </ObjectField>
        </Card>
        <Toolbar>
          <Button variant="primary" size="sm" onClick={submit}>Submit</Button>
          <Button variant="outline" size="sm" onClick={() => setNick('hidden')}>Hide nickname</Button>
          <Button variant="outline" size="sm" onClick={() => setNick('omit')}>Omit nickname</Button>
          <Button variant="outline" size="sm" onClick={() => setNick('visible')}>Show nickname</Button>
        </Toolbar>
        <pre aria-live="polite">{out}</pre>
      </Form>
    </Card>
  );
}
```

## Visibility and submission

Toggle the contact group's flags and compare what renders, what validates and what reaches the submit callback.

```tsx preview play=display
import { useState, type ReactNode } from 'react';
import { isFieldErrors } from '@shikiform/core';
import { enUS, Field, Form, ObjectField, defineEffects, useFieldDecorator, useForm } from '@shikiform/react';
import { Button, Card, FieldInput, FieldRadioGroup, FormItem } from '@site/ui';

type Mode = 'visible' | 'hidden' | 'omit' | 'readOnly';
type Values = { mode: Mode; contact: { email: string; phone: string } };

const effects = defineEffects<Values>(({ reaction }) => {
  reaction((d) => {
    const mode = d.mode.$value;
    d.contact.$visible = mode !== 'hidden';
    d.contact.$omit = mode === 'omit';
    d.contact.$readOnly = mode === 'readOnly';
  });
});

function FieldRow({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return (
    <p className="sf-ui-form-item" style={{ margin: 0 }}>
      <span className="sf-ui-form-label">
        <label {...d.labelProps}>{d.label}: </label>
        {d.required && <span className="sf-ui-form-required" aria-hidden="true">*</span>}
      </span>
      {children}
      {d.errorProps && <small {...d.errorProps} className="sf-ui-form-error">{d.errors[0]}</small>}
    </p>
  );
}

export default function App() {
  const [out, setOut] = useState('');
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { mode: 'visible', contact: { email: '', phone: '13800000000' } },
    fields: { 'contact.email': { required: true } },
    effects: [effects],
  });
  const submit = async () => {
    try {
      const values = await form.submit();
      setOut(`Submitted: ${JSON.stringify(values)}`);
    } catch (error) {
      const errors = isFieldErrors(error) ? error : [];
      setOut(`Blocked: ${errors.map((item) => item.path).join(', ')}`);
    }
  };
  return (
    <Card title="Visibility modes" description="Switch the contact group flags to compare rendering, validation and submit payloads" style={{ maxWidth: 520 }}>
      <Form form={form}>
        <Field<Values, 'mode'>
          name="mode"
          title="contact"
          omit
          enum={['visible', 'hidden', 'omit', 'readOnly']}
          decorator={[FormItem, { layout: 'inline' }]}
          component={[FieldRadioGroup]}
        />
        <Card variant="muted" padding="sm" title="Contact details">
          <ObjectField name="contact">
            <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(180px, 1fr))', gap: 12 }}>
              <Field name="email" title="Email" decorator={[FieldRow]} component={[FieldInput, { placeholder: 'a@b.co' }]} />
              <Field name="phone" title="Phone" decorator={[FieldRow]} component={[FieldInput]} />
            </div>
          </ObjectField>
        </Card>
        <div>
          <Button variant="primary" onClick={submit}>Submit</Button>
        </div>
        <pre aria-live="polite">{out}</pre>
      </Form>
    </Card>
  );
}
```

## Tabs and unmounted fields

Declare required fields in the engine so submitting still checks inputs on a tab that has unmounted.

```tsx preview play=tabs
import { useState, type ReactNode } from 'react';
import { isFieldErrors } from '@shikiform/core';
import { enUS, Field, Form, Watch, useFieldDecorator, useForm, type Draft } from '@shikiform/react';
import { Button, Card, FieldInput, Tabs, TabsList, TabsTab } from '@site/ui';

type Values = { tab: 'base' | 'extra'; name: string; code: string; note: string };

function FieldRow({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return (
    <p className="sf-ui-form-item" style={{ margin: 0 }}>
      <span className="sf-ui-form-label">
        <label {...d.labelProps}>{d.label}: </label>
        {d.required && <span className="sf-ui-form-required" aria-hidden="true">*</span>}
      </span>
      {children}
      {d.errorProps && <small {...d.errorProps} className="sf-ui-form-error">{d.errors[0]}</small>}
    </p>
  );
}

export default function App() {
  const [out, setOut] = useState('');
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { tab: 'base', name: '', code: '', note: '' },
    fields: {
      name: { required: true }, // Rules in the model validate mounted and unmounted fields.
      code: { required: true },
    },
  });
  const submit = async () => {
    try {
      await form.submit();
      setOut('Passed');
    } catch (error) {
      const errors = isFieldErrors(error) ? error : [];
      setOut(`Blocked: ${errors.map((item) => item.path).join(', ')}`);
    }
  };
  return (
    <Card title="Tabbed validation" description="Engine-declared fields validate while unmounted; JSX rules apply only while mounted" style={{ maxWidth: 460 }}>
      <Form form={form}>
        <Field<Values, 'tab'> name="tab" omit enum={[{ label: 'Basic', value: 'base' }, { label: 'Extra', value: 'extra' }]}>
          {(f) => (
            <Tabs value={f.value ?? 'base'} onValueChange={(v) => f.onChange(v as Values['tab'])} variant="segmented">
              <TabsList className="sf-ui-tabs-list">
                {f.options.map((opt) => (
                  <TabsTab key={String(opt.value)} value={String(opt.value)} className="sf-ui-tabs-tab">
                    {opt.label}
                  </TabsTab>
                ))}
              </TabsList>
            </Tabs>
          )}
        </Field>
        <Watch select={(d: Draft<Values>) => d.tab.$value}>
          {(tab) => tab === 'base' ? (
            <Field name="name" title="Name" decorator={[FieldRow]} component={[FieldInput, { placeholder: 'Required (in fields)' }]} />
          ) : (
            <div style={{ display: 'grid', gap: 12 }}>
              <Field name="code" title="Code" decorator={[FieldRow]} component={[FieldInput, { placeholder: 'Required (in fields)' }]} />
              {/* JSX rules apply while mounted. */}
              <Field name="note" title="Notes" required decorator={[FieldRow]} component={[FieldInput, { placeholder: 'Required (while mounted)' }]} />
            </div>
          )}
        </Watch>
        <div>
          <Button variant="primary" onClick={submit}>Submit</Button>
        </div>
        <pre aria-live="polite">{out}</pre>
      </Form>
    </Card>
  );
}
```
