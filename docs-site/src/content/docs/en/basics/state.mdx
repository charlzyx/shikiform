## Four independent flags

| Setting | Display | Validation | Submit |
|---|---|---|---|
| Default | Editable control | Runs | Included |
| `visible: false` | Hidden | Skipped, existing errors clear | Included |
| `disabled: true` | Disabled | Skipped, existing errors clear | Included |
| `readOnly: true` | Read-only preview | Skipped, existing errors clear | Included |
| `omit: true` | Unchanged | Unchanged | Excluded |

Toggle the flags and compare the field with the submitted JSON. Omitted and computed fields are excluded when calculating whether the form is dirty. Hidden fields still count.

```tsx preview
import { useState } from 'react';
import { Select, SelectContent, SelectItem, SelectItemIndicator, SelectTrigger, SelectValue } from '@ui-sdk/react/select';
import { enUS, defineEffects, Form, Submit, typed, useForm, Watch, type Draft } from '@shikiform/react';

type Mode = 'visible' | 'hidden' | 'omit' | 'readOnly';
type Values = { mode: Mode; nick: string };

const effects = defineEffects<Values>(({ reaction }) => {
  reaction((d) => {
    const mode = d.mode.$value;
    d.nick.$visible = mode !== 'hidden';
    d.nick.$omit = mode === 'omit';
    d.nick.$readOnly = mode === 'readOnly';
  });
});

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({ messages: enUS, initialValues: { mode: 'visible', nick: 'ada' }, effects: [effects] });
  const F = typed(form);
  return (
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))}>
      <F.Field name="mode" title="Status" enum={['visible', 'hidden', 'omit', 'readOnly']}>
        {(f) => {
          const items = f.options.map((opt) => ({ value: String(opt.value), label: opt.label }));
          return (
            <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
              <label htmlFor={f.id}>{f.title}: </label>
              <Select items={items} value={f.value} onValueChange={(next) => next && f.onChange(next as Mode)}>
                <SelectTrigger id={f.id}><SelectValue /></SelectTrigger>
                <SelectContent>
                  {items.map((opt) => (
                    <SelectItem key={opt.value} value={opt.value}>
                      {opt.label}
                      <SelectItemIndicator aria-hidden="true">✓</SelectItemIndicator>
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </div>
          );
        }}
      </F.Field>
      <F.Field name="nick" title="Nickname" required>
        {(f) => (
          <p>
            <label htmlFor={f.id}>{f.title}: </label>
            {f.readOnly ? (
              <span>{f.value || '-'}</span>
            ) : (
              <input
                id={f.id}
                value={f.value ?? ''}
                disabled={f.disabled}
                onChange={(e) => f.onChange(e.target.value)}
                onBlur={f.onBlur}
              />
            )}
            {f.errors[0] && <small role="alert" style={{ color: 'crimson' }}> {f.errors[0]}</small>}
          </p>
        )}
      </F.Field>
      <Watch select={(d: Draft<Values>) => d.nick.$visible}>{(shown) => <p>Read $visible: {String(shown)}</p>}</Watch>
      <Submit>Submit</Submit>
      <pre>{result}</pre>
    </Form>
  );
}
```

## Inheritance and defaults

- **`visible`, `disabled`, `readOnly`**: Inherit from parent fields, and any child can override with an explicit `true` or `false` (for example, a child with `disabled: false` stays editable inside a disabled parent). If both `disabled` and `readOnly` are `true`, `readOnly` wins.
- **`omit`**: If any ancestor has `omit: true`, the entire subtree is excluded from submit—a child's `omit: false` cannot opt back in.
- **Clearing errors & keeping values**: Hiding a field keeps its value when shown again. Turning off validation (`visible: false`, `disabled: true`, or `readOnly: true`) clears existing errors immediately; re-enabling does not re-run validation until the next blur, `validate`, or `submit`.
- **Disabled still submits**: Unlike react-hook-form, disabled fields are included in submit values unless you also set `omit: true`.
- **Resetting overrides**: Assign `undefined` in a `reaction` (e.g. `d.nick.$disabled = undefined`) to remove the reaction override and fall back to the field's default. For a whole read-only detail page, pass `useForm({ readOnly: true })` or call `form.setReadOnly(true)`.
