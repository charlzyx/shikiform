## Array rows

Inside an array row, `useArrayField()`, `useIndex()` and `useRecord()` supply the array, position and reactive row value. A container renders its children and can provide an add button even when the array is empty.

```tsx preview
import { type ReactNode } from 'react';
import { Checkbox, CheckboxIndicator } from '@ui-sdk/react/checkbox';
import { enUS, ArrayField, Form, Watch, useArrayField, useField, useForm, useIndex, useRecord, type Draft } from '@shikiform/react';

type Todo = { id: number; title: string; done: boolean };
type Values = { todos: Todo[] };
let nextId = 4;

function TitleInput() {
  const field = useField<Todo, 'title'>('title', { required: true });
  return (
    <span>
      <input aria-label="Title" value={field.value ?? ''} onChange={(event) => field.onChange(event.target.value)} onBlur={field.onBlur} />
      {field.errors[0] && <small style={{ color: 'crimson' }}> {field.errors[0]}</small>}
    </span>
  );
}

function DoneBox() {
  const field = useField<Todo, 'done'>('done');
  return (
    <Checkbox aria-label="Done" checked={Boolean(field.value)} onCheckedChange={(checked) => field.onChange(checked === true)} onBlur={field.onBlur}>
      <CheckboxIndicator aria-hidden="true">✓</CheckboxIndicator>
    </Checkbox>
  );
}

function Row() {
  const array = useArrayField<Todo>();
  const index = useIndex();
  const record = useRecord<Todo>();
  return (
    <li {...array.dragProps(index)} style={{ display: 'flex', alignItems: 'center', gap: 6, marginBottom: 6, cursor: 'grab', opacity: record.done ? 0.5 : 1 }}>
      ⠿ <DoneBox /> <TitleInput /> <small>#{index} · id {record.id}</small>
      <button type="button" onClick={() => array.remove(index)}>Delete</button>
    </li>
  );
}

function TodoList({ children }: { children?: ReactNode }) {
  const array = useArrayField<Todo>();
  return (
    <>
      <ul style={{ paddingLeft: 0, listStyle: 'none' }}>{children}</ul>
      <button type="button" onClick={() => array.push({ id: nextId++, title: '', done: false })}>Add</button>
    </>
  );
}

export default function App() {
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { todos: [{ id: 1, title: "Read the docs", done: true }, { id: 2, title: "Write the form", done: false }, { id: 3, title: "Drag to reorder", done: false }] },
  });
  return (
    <Form form={form}>
      <ArrayField name="todos" rowKey="id" component={TodoList}>{() => <Row />}</ArrayField>
      <Watch select={(d: Draft<Values>) => d.todos.$value.filter((todo) => todo.done).length}>
        {(done) => <p>Completed {done} items</p>}
      </Watch>
    </Form>
  );
}
```

## Tree nodes

Within a `TreeField` node, `useTreeField()` gives the current path, position, record and editing operations. Outside the tree, pass its name to obtain the root handle.

```tsx
function NodeRow() {
  const { record, dragProps } = useTreeField<Node>()
  const field = useField<Node, 'name'>('name')
  return <div {...dragProps()}>⠿ <input value={field.value} onChange={(e) => field.onChange(e.target.value)} /> {record.children?.length ?? 0}</div>
}

<ul>
  <TreeField name="menu" rowKey="id">
    {(node) => (
      <li>
        <NodeRow />
        {node.children && <ul>{node.children}</ul>}
      </li>
    )}
  </TreeField>
</ul>
```

## Adapt a control

`connect` and `mapProps` are exported directly from `@shikiform/react` with zero UI-library dependencies. Labels, required marks, and validation messages come from [`useFieldDecorator`](/en/reference/react-hooks#usefielddecorator), while submit and reset actions use headless [`Submit` / `Reset`](/en/reference/react-hooks#submit).

Every UI library integration follows one boundary: **the UI library renders controls and layout only, while ShikiForm owns field values, validation, effects, and submission**. Render one ShikiForm `<Form form={form}>` and skip the UI library's own `<Form>` / `Form.useForm()` so two form stores never compete for control state.

> `@site/ui` in this site's interactive demos is an internal docs-only kit built on `@ui-sdk/react`. In your application, install a public UI library package and adapt it with the patterns below.

Connecting a library takes three steps:

1. **Wrap controls with `connect(Comp, mapProps(...))`**: forwards `value`, `onChange`, `onBlur`, `disabled`, `readOnly`, and `id` with stable generated event handlers while mounted. The built-in `valueFromEvent` handles both DOM events (`e.target.value` / `e.target.checked`) and direct values (passing the first argument through). Use `mapProps` to opt into `options`, `loading`, `status`, `format` / `parse`, `blurOn`, or a read-only `preview`.
2. **Build a `FormItem` decorator with `useFieldDecorator()`**: map `d.label`, `d.required`, `d.error`, `d.warnings`, and `d.description` onto the UI library's form-item layout without passing the UI library's own `name` or `rules`.
3. **Register once with `<Form kit={{ decorator, components }}>` or `<FieldKit>`**: write `<Field name="email" title="Email" component="Input" />` across your screens.

Start with the complete **Ant Design** and **TDesign React** guides below, followed by general adapter patterns and recipes for UI SDK, shadcn/ui, Mantine, and MUI.

### Ant Design

**Install & stylesheet**:

```bash
npm i @shikiform/react antd
```

Import Ant Design's reset stylesheet once in your app entry (`antd` v5 and v6 inject component CSS via CSS-in-JS):

```ts
import 'antd/dist/reset.css'
```

- Supported versions: `npm i @shikiform/react antd` installs `antd` v6 by default (verified on `6.7.2`, requires `react >= 18`, native React 19 support). Projects on React 16.8 / 17 can install `antd@5` (`npm i @shikiform/react antd@5`, verified on `5.29.3`); when using `antd` v5 with React 19, install and import `@ant-design/v5-patch-for-react-19` per Ant Design's docs.
- Single form store: render ShikiForm's `<Form form={form}>` without `antd`'s `<AntForm>` or `AntForm.useForm()`. When `<AntForm.Item>` receives no `name`, `rules`, or `dependencies`, it skips internal form-field registration and renders layout chrome only.

Copy-pasteable minimal form covering `Input`, `Select`, `Switch`, validation messages, reset, and submit:

```tsx
import { useState, type ReactNode } from 'react'
import { Form as AntForm, Button, Input as AntInput, Select as AntSelect, Space, Switch as AntSwitch } from 'antd'
import { enUS, Form, Reset, Submit, connect, mapProps, typed, useFieldDecorator, useForm } from '@shikiform/react'
// Import reset CSS once in your app entry:
// import 'antd/dist/reset.css'

// Input emits a DOM ChangeEvent; connect's default valueFromEvent extracts e.target.value. Supports native readOnly.
const Input = connect(
  AntInput,
  mapProps(
    { status: 'status', required: 'aria-required' },
    (props, field) => ({
      ...props,
      'aria-invalid': field.errors.length > 0 || undefined,
    }),
  ),
)

// Select emits (value, option) directly; unset or allowClear values are undefined.
// AntSelect has no readOnly prop, so map readOnly -> disabled for UI rendering only.
const Select = connect(
  AntSelect,
  mapProps(
    { options: 'options', loading: 'loading', status: 'status', readOnly: 'disabled', required: 'aria-required' },
    (props, field) => ({
      ...props,
      'aria-invalid': field.errors.length > 0 || undefined,
    }),
  ),
)

// Switch uses checked, omits onBlur, and has no readOnly prop.
const Switch = connect(AntSwitch, mapProps({ value: 'checked', readOnly: 'disabled', blurOn: 'onChange' }))

// Layout-only AntForm.Item: omit name and rules so Ant Design does not create a second store.
function FormItem({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator()
  return (
    <AntForm.Item
      layout="vertical"
      label={d.label}
      htmlFor={d.labelProps.htmlFor}
      required={d.required}
      validateStatus={d.invalid ? 'error' : d.warnings.length ? 'warning' : d.validating ? 'validating' : undefined}
      help={d.error ?? d.warnings[0]}
      extra={d.description}
    >
      {children}
    </AntForm.Item>
  )
}

type Values = { name: string; city: string | undefined; notify: boolean }

export default function App() {
  const [saved, setSaved] = useState('')
  const form = useForm<Values>({ messages: enUS, initialValues: { name: '', city: undefined, notify: false } })
  const F = typed(form)

  return (
    <Form form={form} kit={{ decorator: FormItem }} onSubmit={(values) => setSaved(JSON.stringify(values))}>
      <F.Field
        name="name"
        title="Name"
        description="Public display name"
        required
        rules={{ minLength: 2, message: 'Enter at least 2 characters' }}
        component={[Input, { placeholder: 'Enter your name', allowClear: true }]}
      />
      <F.Field
        name="city"
        title="City"
        required
        enum={[
          { label: 'Hangzhou', value: 'hz' },
          { label: 'Nanjing', value: 'nj' },
        ]}
        component={[Select, { placeholder: 'Choose a city', allowClear: true }]}
      />
      <F.Field name="notify" title="Notifications" component={Switch} />
      <Space>
        {/* Ant Design Button uses htmlType for the native button type; forward it with a render prop. */}
        <Submit>
          {(s) => (
            <Button type="primary" htmlType={s.type} loading={s.loading} disabled={s.disabled} onClick={s.onClick}>
              Submit
            </Button>
          )}
        </Submit>
        <Reset afterReset={() => setSaved('')}>
          {(s) => (
            <Button htmlType="button" disabled={s.disabled} onClick={s.onClick}>
              Reset
            </Button>
          )}
        </Reset>
      </Space>
      {saved && <pre aria-live="polite">{saved}</pre>}
    </Form>
  )
}
```

Key Ant Design details:

- **`Input` vs `Select` callbacks & clearing**: `AntInput` fires `onChange(ChangeEvent)` (`allowClear` emits `''`); `AntSelect` fires `onChange(value, option)` directly (`allowClear` emits `undefined`, and `placeholder` only displays when `value === undefined`). Store empty single-select values as `undefined` (`string | undefined`).
- **`onBlur`, readOnly & accessibility**: `AntInput` and `AntSelect` support `onBlur`, `disabled`, `id`, `aria-required`, and `aria-invalid`, while `htmlFor={d.labelProps.htmlFor}` links the label; `AntSwitch` omits `onBlur`, so configure `blurOn: 'onChange'`. For read-only mode, `AntInput` supports native `readOnly`, whereas `AntSelect` and `AntSwitch` do not—`mapProps({ readOnly: 'disabled' })` maps `readOnly` to `disabled` on those controls strictly for UI rendering while the ShikiForm field remains `readOnly` (hiding required asterisks and skipping validation). To render plain text in read-only mode, pass a `preview` component in `mapProps` (see the `preview` usage in [Runnable adapter](#runnable-adapter)).
- **Submit & reset buttons**: Ant Design `<Button>` uses `type` for visual styling (`'primary' | 'default'`) and `htmlType` for the native form action. Use `<Submit>` / `<Reset>` render props to pass `htmlType`, `loading`, `disabled`, and `onClick`.
- **More Ant Design controls**: the optional `DatePicker` snippet imports `dayjs` directly—in projects with strict dependency resolution (such as pnpm), install it explicitly with `npm i dayjs`:

```tsx
import dayjs, { type Dayjs } from 'dayjs'
import { Checkbox as AntCheckbox, DatePicker as AntDatePicker, InputNumber as AntInputNumber } from 'antd'
import { connect, mapProps } from '@shikiform/react'

// InputNumber supports readOnly; emits null when cleared, so type the field as number | null.
const InputNumber = connect(AntInputNumber, mapProps({ status: 'status' }))
// Checkbox supports onBlur (valueFromEvent reads e.target.checked); map readOnly to disabled for UI rendering.
const Checkbox = connect(AntCheckbox, mapProps({ value: 'checked', readOnly: 'disabled' }))
// Store YYYY-MM-DD strings and convert to/from Dayjs at the control boundary; map readOnly to disabled.
const DatePicker = connect(
  AntDatePicker,
  mapProps({
    status: 'status',
    readOnly: 'disabled',
    format: (value: string | undefined) => (value ? dayjs(value) : null),
    parse: (date: Dayjs | null) => (date?.isValid() ? date.format('YYYY-MM-DD') : undefined),
  }),
)
```

### TDesign React

**Install & stylesheet**:

```bash
npm i @shikiform/react tdesign-react
```

Components imported from `tdesign-react`'s default ESM entry (`es/index.js`) pull in their own component CSS automatically. Import the base stylesheet once in your app entry (`tdesign-react/dist/tdesign.css` is also available for the full stylesheet bundle):

```ts
import 'tdesign-react/es/style/index.css'
```

- Supported versions: `tdesign-react` 1.x (`peerDependencies`: `react >= 16.13.1`, verified on `1.19.0`). On React 19, import `tdesign-react/es/_util/react-19-adapter` in your entry when using TDesign imperative/portal overlays.
- Single form store: render ShikiForm's `<Form form={form}>` without TDesign's `<TForm>` or `TForm.useForm()`. Because `<TForm.FormItem>` clones any direct custom-component child to inject its own `value` / `onChange`, wrap `{children}` in a plain `<div style={{ width: '100%' }}>` so TDesign never touches controlled props, and pass `showErrorMessage` explicitly (its default outside `<TForm>` is `undefined`) so `t-is-error` classes render.

Copy-pasteable minimal form covering `Input`, `Select`, `Switch`, validation messages, reset, and submit:

```tsx
import { useState, type ReactNode } from 'react'
import { Button, Form as TForm, Input as TInput, Select as TSelect, Space, Switch as TSwitch } from 'tdesign-react'
import { enUS, Form, Reset, Submit, connect, mapProps, typed, useFieldDecorator, useForm } from '@shikiform/react'
// Import base CSS once in your app entry:
// import 'tdesign-react/es/style/index.css'

// Both TDesign Input and Select pass the next value directly as the first onChange argument: (value, context).
// TInput forwards unknown props to its outer wrapper div rather than the inner <input>, so set id: false.
const Input = connect(TInput, mapProps({ id: false, status: 'status' }))

// Single-select TSelect emits null when cleared via clearable; type the field as string | null.
const Select = connect(TSelect, mapProps({ id: false, options: 'options', loading: 'loading', status: 'status' }))

// TSwitch has no onBlur prop; blurOn: 'onChange' triggers blur validation on toggle.
const Switch = connect(TSwitch, mapProps({ id: false, blurOn: 'onChange' }))

// Layout-only TForm.FormItem: omit name / rules, enable showErrorMessage, and wrap children in a div
// to stop TForm.FormItem from cloning its own value / onChange onto the connected control.
function FormItem({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator()
  return (
    <TForm.FormItem
      labelAlign="top"
      label={d.label}
      requiredMark={d.required}
      showErrorMessage
      status={d.invalid ? 'error' : d.warnings.length ? 'warning' : d.validating ? 'validating' : undefined}
      tips={d.error ?? d.warnings[0]}
      help={d.description}
    >
      <div style={{ width: '100%' }}>{children}</div>
    </TForm.FormItem>
  )
}

type Values = { name: string; city: string | null; notify: boolean }

export default function App() {
  const [saved, setSaved] = useState('')
  const form = useForm<Values>({ messages: enUS, initialValues: { name: '', city: null, notify: false } })
  const F = typed(form)

  return (
    <Form form={form} kit={{ decorator: FormItem }} onSubmit={(values) => setSaved(JSON.stringify(values))}>
      <F.Field
        name="name"
        title="Name"
        description="Public display name"
        required
        rules={{ minLength: 2, message: 'Enter at least 2 characters' }}
        component={[Input, { placeholder: 'Enter your name', clearable: true }]}
      />
      <F.Field
        name="city"
        title="City"
        required
        enum={[
          { label: 'Hangzhou', value: 'hz' },
          { label: 'Nanjing', value: 'nj' },
        ]}
        component={[Select, { placeholder: 'Choose a city', clearable: true }]}
      />
      <F.Field name="notify" title="Notifications" component={Switch} />
      <Space>
        {/* TDesign Button uses type for native HTML button type and theme for visual style. */}
        <Submit>
          {(s) => (
            <Button theme="primary" type={s.type} loading={s.loading} disabled={s.disabled}>
              Submit
            </Button>
          )}
        </Submit>
        <Reset afterReset={() => setSaved('')} asChild>
          <Button variant="outline">Reset</Button>
        </Reset>
      </Space>
      {saved && <pre aria-live="polite">{saved}</pre>}
    </Form>
  )
}
```

Key TDesign React details:

- **Direct-value callbacks & clearing**: both `TInput` and `TSelect` call `onChange(value, context)` with the new value first, so `connect` forwards it unchanged. Clearing differs by control: `TInput` emits `''`, single-select `TSelect` emits `null` (multi-select emits `[]`). Typing the field as `string | null` stores the cleared value directly; add `parse: (v) => (v == null ? undefined : v)` if you prefer `undefined`.
- **`onBlur`, disabled & `id` / accessibility**: `TInput` and `TSelect` accept `onBlur`, `disabled`, `readOnly`, and `status`; `TSwitch` and `TCheckbox` omit `onBlur`, so pass `blurOn: 'onChange'`. Because `TInput` spreads extra props onto `.t-input__wrap` and `TSelect` does not forward `id` to its input, `id: false` avoids placing an unreachable `id` on wrapper divs, while `status`, `tips`, and `help` render the error state and message.
- **Submit & reset buttons**: TDesign `<Button>` uses native `type` (`'submit' | 'reset' | 'button'`) and `theme="primary"` for styling. Pass `type={s.type}`, `loading={s.loading}`, and `disabled={s.disabled}` in `<Submit>`'s render prop so clicks and Enter trigger `<Form onSubmit>`; `<Reset asChild>` merges directly onto `<Button variant="outline">`.
- **More TDesign React controls**:

```tsx
import { Checkbox as TCheckbox, DatePicker as TDatePicker, InputNumber as TInputNumber } from 'tdesign-react'
import { connect, mapProps } from '@shikiform/react'

const InputNumber = connect(TInputNumber, mapProps({ id: false, status: 'status' }))
// TCheckbox uses checked and omits onBlur.
const Checkbox = connect(TCheckbox, mapProps({ id: false, value: 'checked', blurOn: 'onChange' }))
// TDatePicker stores formatted date strings (e.g. YYYY-MM-DD) directly.
const DatePicker = connect(TDatePicker, mapProps({ id: false, status: 'status' }))
```

### Labels and input bindings

For a widget with its own label, use `useFieldDecorator().control` for display props and `useField()` for values. For a plain input, bind it with `connect` and render a separate decorator.

```tsx
import { Field, connect, useField, useFieldDecorator } from '@shikiform/react'

// ✗ control has no value or onChange
function Bad() {
  const { control } = useFieldDecorator()
  return <TextInput {...control} />
}

// ✓ Widgets with their own labels use control for display and useField for the value.
function NameInput() {
  const field = useField<string>()
  const { control } = useFieldDecorator()
  return (
    <TextInput
      {...control}
      value={field.value ?? ''}
      onChange={(event) => field.onChange(event.currentTarget.value)}
    />
  )
}

// ✓ Connect a plain input and wrap it in a FormItem decorator.
const Input = connect(ShadcnInput)
<Field name="name" title="Name" decorator={[FormItem]} component={[Input]} />
```

### Widgets without onBlur

Choose when interaction is complete. `blurOn: 'onOpenChange'` blurs when a popup closes. `blurOn: 'onChange'` blurs after a change. Use `onBlur: false` to opt out explicitly. Handler props created by `connect` (`onChange`, `onBlur`, `blurOn`) keep one identity while mounted; other function props pass through unchanged.

```tsx
connect(SelectControl, mapProps({ options: 'options', onChange: 'onValueChange', blurOn: 'onOpenChange' }))
connect(Picker, mapProps({ onBlur: false }))
```

### UI SDK

[UI SDK](https://ui-sdk.dev) (`@ui-sdk/react` + `@ui-sdk/core`) ships unstyled, accessible UI components imported per component and styled via `data-slot` attributes:

- `Select` (`@ui-sdk/react/select`): controlled with `value` (`string | null`) and `onValueChange`. Because the root has no `onBlur`, configure `blurOn: 'onOpenChange'` so closing the popup triggers blur validation. Passing `items` on the root keeps labels available during SSR and while closed.
- `Checkbox` (`@ui-sdk/react/checkbox`): accepts `onBlur` directly; map `value: 'checked'`, `onChange: 'onCheckedChange'`, and `parse: (c) => c === true`.
- `DatePicker` (`@ui-sdk/react/date-picker`): uses ISO `"YYYY-MM-DD"` strings (or `null`) natively with no date library required; map `onChange: 'onValueChange'` and `blurOn: 'onOpenChange'`.

The runnable example below adapts `@ui-sdk/react` `Select`, `Checkbox`, and `DatePicker` with `connect` + `mapProps` and registers them via `<Form kit>`:

```tsx preview
import { useState, type ReactNode } from 'react';
import { Checkbox as UiCheckbox, CheckboxIndicator } from '@ui-sdk/react/checkbox';
import { DatePicker as UiDatePicker, DatePickerContent, DatePickerInput, DatePickerTrigger } from '@ui-sdk/react/date-picker';
import { Select as UiSelect, SelectContent, SelectItem, SelectItemIndicator, SelectTrigger, SelectValue } from '@ui-sdk/react/select';
import { enUS, Field, Form, Reset, Submit, connect, mapProps, useFieldDecorator, useForm } from '@shikiform/react';

function TextControl(props: {
  id?: string;
  value?: string;
  placeholder?: string;
  disabled?: boolean;
  readOnly?: boolean;
  onChange?: (event: { target: { value: string } }) => void;
  onBlur?: () => void;
}) {
  return (
    <input
      id={props.id}
      placeholder={props.placeholder}
      disabled={props.disabled}
      readOnly={props.readOnly}
      value={props.value ?? ''}
      onChange={props.onChange}
      onBlur={props.onBlur}
    />
  );
}

function SelectControl(props: {
  id?: string;
  value?: string;
  disabled?: boolean;
  options?: { label: string; value: string; disabled?: boolean }[];
  onValueChange?: (value: string) => void;
  onOpenChange?: (open: boolean) => void;
}) {
  const items = (props.options ?? []).map((o) => ({ value: String(o.value), label: o.label, disabled: o.disabled }));
  return (
    <UiSelect
      items={items}
      value={props.value || null}
      disabled={props.disabled}
      onValueChange={(next) => props.onValueChange?.(next ?? '')}
      onOpenChange={(open) => props.onOpenChange?.(open)}
    >
      <SelectTrigger id={props.id}>
        <SelectValue placeholder="Choose an option" />
      </SelectTrigger>
      <SelectContent>
        {items.map((option) => (
          <SelectItem key={option.value} value={option.value} disabled={option.disabled}>
            {option.label}
            <SelectItemIndicator aria-hidden="true">✓</SelectItemIndicator>
          </SelectItem>
        ))}
      </SelectContent>
    </UiSelect>
  );
}

function CheckboxControl(props: {
  id?: string;
  checked?: boolean;
  disabled?: boolean;
  label?: string;
  onCheckedChange?: (checked: boolean | 'indeterminate') => void;
  onBlur?: () => void;
}) {
  return (
    <label style={{ display: 'inline-flex', alignItems: 'center', gap: 6 }}>
      <UiCheckbox
        id={props.id}
        checked={Boolean(props.checked)}
        disabled={props.disabled}
        onCheckedChange={props.onCheckedChange}
        onBlur={props.onBlur}
      >
        <CheckboxIndicator aria-hidden="true">✓</CheckboxIndicator>
      </UiCheckbox>
      {props.label}
    </label>
  );
}

function DatePickerControl(props: {
  id?: string;
  value?: string;
  disabled?: boolean;
  onValueChange?: (value: string) => void;
  onOpenChange?: (open: boolean) => void;
}) {
  return (
    <UiDatePicker
      value={props.value || null}
      disabled={props.disabled}
      onValueChange={(next) => props.onValueChange?.(typeof next === 'string' ? next : '')}
      onOpenChange={(open) => props.onOpenChange?.(open)}
    >
      <DatePickerInput id={props.id} aria-label="Start date" placeholder="YYYY-MM-DD" />
      <DatePickerTrigger aria-label="Open calendar">Calendar</DatePickerTrigger>
      <DatePickerContent />
    </UiDatePicker>
  );
}

const Input = connect(TextControl, mapProps({ preview: ({ value }: { value?: string }) => <span>{value || '—'}</span> }));
const Select = connect(SelectControl, mapProps({ options: 'options', onChange: 'onValueChange', blurOn: 'onOpenChange' }));
const Checkbox = connect(CheckboxControl, mapProps({
  value: 'checked',
  onChange: 'onCheckedChange',
  parse: (checked: boolean | 'indeterminate') => checked === true,
}));
const DatePicker = connect(DatePickerControl, mapProps({ onChange: 'onValueChange', blurOn: 'onOpenChange' }));

function FormItem({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return (
    <div style={{ display: 'grid', gap: 4, marginBottom: 12 }}>
      {d.label ? <label {...d.labelProps}>{d.label}{d.required ? <span aria-hidden="true"> *</span> : null}</label> : null}
      {children}
      {d.descriptionProps ? <p {...d.descriptionProps} style={{ margin: 0, color: '#666' }}>{d.description}</p> : null}
```

### MUI

```tsx
      id={d.id}
      label={d.label}

const Agree = connect(Checkbox, mapProps({ value: 'checked' }))

function toDayjs(value: string | undefined): Dayjs | null {
  const match = value && /^(\d{4})-(\d{2})-(\d{2})$/.exec(value)
  if (!match) return null
  return dayjs(new Date(Number(match[1]), Number(match[2]) - 1, Number(match[3])))
}

// Forward the label target to the DatePicker's inner TextField.

  slotProps: { textField: { id: field.id } },
})))
```
