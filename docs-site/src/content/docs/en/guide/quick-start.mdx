## Install

```bash
npm i @shikiform/react
```

`@shikiform/react` includes the core engine (`@shikiform/core`), headless React bindings, and pro hooks for query tables, editable arrays, column settings, and temporary draft editors. Install `@shikiform/core` directly only when using the engine without React.

Requires `react` and `react-dom` (`^16.8.0 || ^17 || ^18 || ^19`). On React 18+, omitting `<Form id>` auto-generates hydration-safe field IDs via `useId()`; React 16.8/17 SSR apps should pass an explicit `<Form id="…">` to avoid hydration mismatches. ShikiForm has zero UI-library dependencies and works with native HTML or any React UI library.

## Your first form

Create one engine with `useForm`. `<Form>` provides it to descendants and handles native submit events. A field binds a path, a decorator renders its label and errors, and a component renders the control. Fill in the fields and submit to see the resulting values.

```tsx preview
import { useState, type ReactNode } from 'react';
import { enUS, Form, Reset, Submit, typed, useField, useFieldDecorator, useForm } from '@shikiform/react';

type Values = { name: string; email: string; age: number | null };

function FieldItem({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return (
    <div style={{ marginBottom: 10 }}>
      <label {...d.labelProps} style={{ display: 'inline-block', width: 64 }}>
        {d.required && <span style={{ color: 'crimson' }}>* </span>}
        {d.label}
      </label>
      {children}
      {d.errorProps && <small {...d.errorProps} style={{ color: 'crimson', marginLeft: 8 }}>{d.errors[0]}</small>}
    </div>
  );
}

function TextInput({ placeholder }: { placeholder?: string }) {
  const field = useField<string>();
  return (
    <input id={field.id} value={field.value ?? ''} placeholder={placeholder}
      disabled={field.disabled} readOnly={field.readOnly}
      onChange={(e) => field.onChange(e.target.value)} onBlur={field.onBlur} />
  );
}

function NumberInput() {
  const field = useField<number | null>();
  return (
    <input id={field.id} type="number" value={field.value ?? ''}
      disabled={field.disabled} readOnly={field.readOnly}
      onChange={(e) => field.onChange(e.target.value === '' ? null : Number(e.target.value))}
      onBlur={field.onBlur} />
  );
}

export default function App() {
  const [result, setResult] = useState("No submission yet");
  const form = useForm<Values>({ messages: enUS, initialValues: { name: '', email: '', age: 18 } });
  const F = typed(form);
  return (
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))}>
      <F.Field name="name" title="Name" required decorator={[FieldItem]} component={[TextInput, { placeholder: 'Ada' }]} />
      <F.Field name="email" title="Email" required rules={{ format: 'email', message: "Invalid email address" }}
        decorator={[FieldItem]} component={[TextInput]} />
      <F.Field name="age" title="Age" rules={{ min: 0, max: 150 }} decorator={[FieldItem]} component={[NumberInput]} />
      <div style={{ display: 'flex', gap: 8, marginTop: 8 }}>
        <Submit>Submit</Submit>
        <Reset afterReset={() => setResult("No submission yet")}>Reset</Reset>
      </div>
      <pre aria-live="polite">{result}</pre>
    </Form>
  );
}
```

## Bind an input with useField

Inside `<Field>`, `useField()` returns its binding. Pass the input value to `onChange` and forward `onBlur` so blur validation works. Use the field's `id` to associate the control with its label.

```tsx preview
import { useState } from 'react';
import { enUS, Field, Form, useField, useForm } from '@shikiform/react';

function TextInput({ label }: { label: string }) {
  const field = useField<string>();
  return (
    <p>
      <label htmlFor={field.id}>{label} </label>
      <input id={field.id} value={field.value ?? ''} onChange={(event) => field.onChange(event.target.value)} onBlur={field.onBlur} />
      {field.errors[0] && <small role="alert" style={{ color: 'crimson' }}> {field.errors[0]}</small>}
    </p>
  );
}

export default function App() {
  const [result, setResult] = useState("No submission yet");
  const form = useForm({ messages: enUS, initialValues: { username: '' } });
  return (
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))}>
      <Field name="username" required component={[TextInput, { label: "Name" }]} />
      <button type="submit">Submit</button>
      <pre aria-live="polite">{result}</pre>
    </Form>
  );
}
```

## Reuse a group of fields

Define an address fragment once with `typed<Address>()`. Mount it inside each `ObjectField` and use names relative to that object.

```tsx preview
import { enUS, Form, ObjectField, typed, useField, useForm } from '@shikiform/react';

type Address = { city: string; street: string };
const A = typed<Address>();

function LabeledInput() {
  const field = useField<string>();
  return (
    <div style={{ marginBottom: 6 }}>
      <label htmlFor={field.id} style={{ display: 'inline-block', width: 56 }}>{field.title}</label>
      <input id={field.id} value={field.value ?? ''} onChange={(e) => field.onChange(e.target.value)} onBlur={field.onBlur} />
    </div>
  );
}

function AddressFields() {
  return (
    <>
      <A.Field name="city" title="City" component={[LabeledInput]} />
      <A.Field name="street" title="Street" component={[LabeledInput]} />
    </>
  );
}

export default function App() {
  const form = useForm({ messages: enUS, initialValues: { home: { city: "Hangzhou", street: '' }, office: { city: "Shanghai", street: '' } } });
  return (
    <Form form={form}>
      <fieldset><legend>Home address</legend><ObjectField name="home"><AddressFields /></ObjectField></fieldset>
      <fieldset><legend>Office address</legend><ObjectField name="office"><AddressFields /></ObjectField></fieldset>
    </Form>
  );
}
```
