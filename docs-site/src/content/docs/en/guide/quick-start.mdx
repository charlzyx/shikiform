## Install

```bash
npm i @shikiform/react
```

`@shikiform/react` includes the core engine (`@shikiform/core`), headless React bindings, and pro hooks for query tables, editable arrays, column settings, and temporary draft editors. Install `@shikiform/core` directly only when using the engine without React.

Requires `react` and `react-dom` (`^16.8.0 || ^17 || ^18 || ^19`). On React 18+, omitting `<Form id>` auto-generates hydration-safe field IDs via `useId()`; React 16.8/17 SSR apps should pass an explicit `<Form id="…">` to avoid hydration mismatches. ShikiForm has zero UI-library dependencies and works with native HTML or any React UI library.

## Your first form

Create one engine with `useForm`. `<Form>` provides it to descendants and handles native submit events. Pass `kit={siteKit}` once to supply the default `FormItem` decorator so each field only needs its path, rules, and control. Fill in the fields and submit to see the resulting values.

> `@site/ui` is an internal docs-only demo kit built on `@ui-sdk/react`. To connect a public UI library in your app, see [Ant Design](/en/guide/headless#ant-design), [TDesign React](/en/guide/headless#tdesign-react), or [connecting a UI library](/en/guide/headless#adapt-a-control).

```tsx preview
import { useState } from 'react';
import { enUS, Form, Reset, Submit, typed, useForm } from '@shikiform/react';
import { Button, FieldInput, FieldNumber, siteKit } from '@site/ui';

type Values = { name: string; email: string; age: number | null };

export default function App() {
  const [result, setResult] = useState("No submission yet");
  const form = useForm<Values>({ messages: enUS, initialValues: { name: '', email: '', age: 18 } });
  const F = typed(form);
  return (
    <Form form={form} kit={siteKit} onSubmit={(values) => setResult(JSON.stringify(values))} style={{ maxWidth: 420 }}>
      <F.Field name="name" title="Name" required component={[FieldInput, { placeholder: 'Ada' }]} />
      <F.Field
        name="email"
        title="Email"
        required
        rules={{ format: 'email', message: "Invalid email address" }}
        component={[FieldInput, { placeholder: 'ada@example.com' }]}
      />
      <F.Field name="age" title="Age" rules={{ min: 0, max: 150 }} component={[FieldNumber, { min: 0, max: 150 }]} />
      <div style={{ display: 'flex', gap: 8 }}>
        <Submit asChild><Button variant="primary">Submit</Button></Submit>
        <Reset asChild afterReset={() => setResult("No submission yet")}><Button variant="outline">Reset</Button></Reset>
      </div>
      <pre aria-live="polite">{result}</pre>
    </Form>
  );
}
```

## Bind an input with useField

Inside `<Field>`, `useField()` returns its binding. Pass the input value to `onChange` and forward `onBlur` so blur validation works. Use the field's `id` to associate the control with its label.

```tsx preview
import { useState } from 'react';
import { enUS, Field, Form, useField, useForm } from '@shikiform/react';
import { Button, Input } from '@site/ui';

function TextInput({ label }: { label: string }) {
  const field = useField<string>();
  return (
    <div className="sf-ui-form-item">
      <label htmlFor={field.id} className="sf-ui-form-label">{label}</label>
      <Input
        id={field.id}
        value={field.value ?? ''}
        invalid={field.errors.length > 0}
        onChange={(event) => field.onChange(event.target.value)}
        onBlur={field.onBlur}
      />
      {field.errors[0] && <p role="alert" className="sf-ui-form-error">{field.errors[0]}</p>}
    </div>
  );
}

export default function App() {
  const [result, setResult] = useState("No submission yet");
  const form = useForm({ messages: enUS, initialValues: { username: '' } });
  return (
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values))} style={{ maxWidth: 360 }}>
      <Field name="username" required component={[TextInput, { label: "Name" }]} />
      <div><Button variant="primary" type="submit">Submit</Button></div>
      <pre aria-live="polite">{result}</pre>
    </Form>
  );
}
```

## Reuse a group of fields

Define an address fragment once with `typed<Address>()`. Mount it inside each `ObjectField` and use names relative to that object.

```tsx preview
import { enUS, Form, ObjectField, typed, useForm } from '@shikiform/react';
import { Card, FieldInput, siteKit } from '@site/ui';

type Address = { city: string; street: string };
const A = typed<Address>();

function AddressFields() {
  return (
    <div style={{ display: 'grid', gap: 12 }}>
      <A.Field name="city" title="City" component={FieldInput} />
      <A.Field name="street" title="Street" component={[FieldInput, { placeholder: 'Street & building' }]} />
    </div>
  );
}

export default function App() {
  const form = useForm({
    messages: enUS,
    initialValues: { home: { city: "Hangzhou", street: '' }, office: { city: "Shanghai", street: '' } },
  });
  return (
    <Form form={form} kit={siteKit}>
      <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(220px, 1fr))', gap: 12 }}>
        <Card title="Home address" padding="sm">
          <ObjectField name="home"><AddressFields /></ObjectField>
        </Card>
        <Card title="Office address" padding="sm">
          <ObjectField name="office"><AddressFields /></ObjectField>
        </Card>
      </div>
    </Form>
  );
}
```
