## Object and array scopes

Use `ObjectField` for a group of related properties and `ArrayField` for repeated records. The render function receives components typed to that scope. Use stable row keys for moves and insertions.

```tsx preview
import { type ReactNode } from 'react';
import { enUS, ArrayField, Field, Form, ObjectField, Watch, useArrayField, useField, useIndex, useForm, type Draft } from '@shikiform/react';

type Contact = { id: number; name: string; phone: string };
type Values = { profile: { name: string; address: { city: string } }; contacts: Contact[] };

let nextId = 3;

function TextInput({ width }: { width?: number }) {
  const field = useField<string>();
  return (
    <label style={{ display: 'inline-flex', alignItems: 'center', gap: 4 }}>
      <span>{field.title}</span>
      <input id={field.id} style={width ? { width } : undefined} value={field.value ?? ''}
        onChange={(e) => field.onChange(e.target.value)} onBlur={field.onBlur} />
      {field.description && <small style={{ opacity: 0.7 }}>{field.description}</small>}
      {field.errors[0] && <small role="alert" style={{ color: 'crimson' }}>{field.errors[0]}</small>}
    </label>
  );
}

/** Read the array and row position from scope. */
function RowActions() {
  const array = useArrayField<Contact>();
  const index = useIndex();
  return (
    <span style={{ display: 'inline-flex', gap: 4 }}>
      <button type="button" disabled={index === 0} onClick={() => array.move(index, index - 1)}>Move up</button>
      <button type="button" disabled={index === array.keys.length - 1} onClick={() => array.move(index, index + 1)}>Move down</button>
      <button type="button" onClick={() => array.remove(index)}>Delete</button>
    </span>
  );
}

/** Put the add button in the ArrayField container to access its array handle. */
function ContactList({ children }: { children?: ReactNode }) {
  const array = useArrayField<Contact>();
  return (
    <div>
      {children}
      <button type="button" onClick={() => array.push({ id: nextId++, name: '', phone: '' })}>Add contact</button>
    </div>
  );
}

export default function App() {
  const form = useForm<Values>({
    messages: enUS,
    initialValues: {
      profile: { name: 'Ada', address: { city: "London" } },
      contacts: [{ id: 1, name: 'Charles', phone: '100' }, { id: 2, name: 'Mary', phone: '200' }],
    },
    fields: { 'contacts.*.name': { required: true } },
  });
  return (
    <Form form={form}>
      <ObjectField name="profile">
        <div style={{ marginBottom: 6 }}><Field name="name" title="Name" component={[TextInput]} /></div>
        <ObjectField name="address">
          <div style={{ marginBottom: 10 }}>
            <Field name="city" title="City" description="Path: profile.address.city" component={[TextInput]} />
          </div>
        </ObjectField>
      </ObjectField>
      <ArrayField name="contacts" rowKey="id" component={ContactList}>
        {() => (
          <div style={{ display: 'flex', gap: 8, alignItems: 'center', marginBottom: 8, flexWrap: 'wrap' }}>
            <Field name="name" title="Contact" component={[TextInput, { width: 110 }]} />
            <Field name="phone" title="Phone" component={[TextInput, { width: 110 }]} />
            <RowActions />
          </div>
        )}
      </ArrayField>
      <Watch select={(d: Draft<Values>) => d.$value}>{(values) => <pre>{JSON.stringify(values, null, 2)}</pre>}</Watch>
    </Form>
  );
}
```

### Use relative field names

A child of `address` uses `city`. Including the parent prefix again produces the wrong path and throws in development.

```tsx
const F = typed(form)

// ✗ Resolves to group.group.inner and throws at runtime
<F.ObjectField name="group">
  <F.Field name="group.inner" />
</F.ObjectField>

// ✓ obj.Field checks relative paths against this object
<F.ObjectField name="group">
  {(obj) => <obj.Field name="inner" />}
</F.ObjectField>
```

## Update a nested value

Assign to the field's own `$value` to notify its readers. Replacing an entire object updates all readers of that object. Outside effects, use `form.setValue(path, value)`.

```ts
d.user.name.$value = 'x'                              // ✓ Only update user.name
d.user.$value.name = 'x'                              // ✓ The same path write
d.user.$value = { ...d.user.$value, name: 'x' }       // Replacing user notifies every reader of that record.
d.items.$value.push(row)                              // ✓ Same row operation as useArrayField().push
d.items.$value = [...d.items.$value, row]             // Replace the whole array.
```

See [dynamic arrays and tables](/en/guide/arrays) for row operations, drag handles and array-level rules.
