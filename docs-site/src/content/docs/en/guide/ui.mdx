## A second control for one field

Set `ui` on an additional field control. It shares the value, rules, options and errors, while keeping its own DOM ID (`<form id>-<path>--<ui>`; on React 18+, omitting `<Form id>` uses `useId()` for a hydration-safe prefix, while React 16.8/17 SSR apps should pass an explicit `<Form id="…">`) and presentation state. Put validation and options on the primary field. Mounting two unnamed `<Field>` components for the same form and path warns once in development, prompting you to add a distinct `ui` name to the second `<Field>`. Whole-object assignments to `$componentProps` or `$decoratorProps` on `$ui(name)` accept only a plain object or `undefined` (other values throw `ShikiUsageError` in development and are ignored in production).

```tsx preview
import type { CSSProperties } from 'react';
import { enUS, defineEffects, Form, typed, useField, useForm } from '@shikiform/react';
import { FieldInput, FormItem, siteKit } from '@site/ui';

type Values = { nickname: string };

// A secondary UI receives the shared binding and its own presentation state.
function Echo({ style }: { style?: CSSProperties }) {
  const field = useField<string>();
  return <span id={field.id} data-echo="" className="sf-ui-readonly" style={style}>{field.value || "(empty)"}</span>;
}

export default function App() {
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { nickname: '' },
    effects: [
      defineEffects<Values>(({ reaction }) => {
        reaction((d) => {
          d.nickname.$ui('echo').$componentProps.style =
            d.nickname.$value.length > 8 ? { color: 'red' } : undefined;
        });
      }),
    ],
  });
  const F = typed(form);
  return (
    <Form form={form} kit={siteKit} id="echo" data-ui-echo-demo="" style={{ maxWidth: 420 }}>
      <F.Field name="nickname" title="Nickname" decorator={[FormItem, { layout: 'horizontal' }]} component={[FieldInput, { placeholder: "The echo turns red after eight characters" }]} />
      <F.Field name="nickname" ui="echo" title="Echo" decorator={[FormItem, { layout: 'horizontal' }]} component={Echo} />
    </Form>
  );
}
```

## Control one instance

Use `d.x.$ui('name')` to change one instance's title, props, visibility, disabled or read-only state. Inside it, `useField()` returns the shared value and that instance's UI state.

```tsx preview
import { enUS, defineEffects, Form, typed, useForm } from '@shikiform/react';
import { FieldNumber, FieldSlider, FieldSwitch, FormItem, siteKit } from '@site/ui';

type Values = { budget: number; locked: boolean };

export default function App() {
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { budget: 30, locked: false },
    effects: [
      defineEffects<Values>(({ reaction }) => {
        reaction((d) => {
          d.budget.$ui('slider').$disabled = d.locked.$value ? true : undefined;
        });
      }),
    ],
  });
  const F = typed(form);
  return (
    <Form form={form} kit={siteKit} data-ui-slider-demo="" style={{ maxWidth: 420 }}>
      <F.Field name="locked" title="Lock slider" decorator={[FormItem, { layout: 'horizontal' }]} component={FieldSwitch} />
      <F.Field name="budget" title="Budget" required decorator={[FormItem, { layout: 'horizontal' }]} component={[FieldNumber, { min: 0, max: 100 }]} />
      <F.Field name="budget" ui="slider" title="Slider" decorator={[FormItem, { layout: 'horizontal' }]} component={[FieldSlider, { min: 0, max: 100 }]} />
    </Form>
  );
}
```

## A node without a value

Use `<UI>` for a tip or group box and address it as `d.$ui('name')`. It has no entry in the submitted data.

```tsx preview
import { useState } from 'react';
import { enUS, defineEffects, Form, Submit, typed, UI, useField, useForm } from '@shikiform/react';
import { Alert, Button, Input, siteKit } from '@site/ui';

type Values = { amount: number | null };

function AmountInput() {
  const field = useField<number | null>();
  return (
    <Input
      id={field.id}
      type="number"
      prefix="¥"
      placeholder="Try entering 50000 or more"
      value={field.value ?? ''}
      onChange={(e) => field.onChange(e.target.value === '' ? null : Number(e.target.value))}
      onBlur={field.onBlur}
    />
  );
}

export default function App() {
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { amount: null },
    effects: [
      defineEffects<Values>(({ reaction }) => {
        reaction((d) => {
          d.$ui('approvalTip').$visible = (d.amount.$value ?? 0) >= 50000;
        });
      }),
    ],
  });
  const F = typed(form);
  const [saved, setSaved] = useState<Values>();
  return (
    <Form form={form} kit={siteKit} onSubmit={setSaved} data-ui-alert-demo="" style={{ maxWidth: 420 }}>
      <F.Field name="amount" title="Expense amount" component={AmountInput} />
      <UI name="approvalTip" component={[Alert, { variant: 'warning' }]}>Amounts of 50000 or more need manager approval</UI>
      <div><Submit asChild><Button variant="primary">Submit</Button></Submit></div>
      {saved && <pre>{JSON.stringify(saved)}</pre>}
    </Form>
  );
}
```

## Hiding a container

Hiding a UI container unmounts its children. Their field-level visibility, validation and submit settings still apply. Set the field's own flags when those should change too.

```tsx preview
import { useState } from 'react';
import { enUS, defineEffects, Form, Submit, typed, UI, useForm } from '@shikiform/react';
import { Button, Card, FieldInput, FieldSwitch, siteKit } from '@site/ui';

type Values = { needInvoice: boolean; invoiceTitle: string; remark: string };

export default function App() {
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { needInvoice: false, invoiceTitle: '', remark: "Keep" },
    effects: [
      defineEffects<Values>(({ reaction }) => {
        reaction((d) => {
          const need = d.needInvoice.$value;
          d.$ui('invoiceBox').$visible = need;   // Only controls whether the card is rendered.
          d.invoiceTitle.$visible = need;        // Hide and skip validation. Visibility alone preserves submission.
          d.invoiceTitle.$omit = !need;          // Exclude the invoice title when an invoice is unnecessary.
        });
      }),
    ],
  });
  const F = typed(form);
  const [saved, setSaved] = useState<Values>();
  return (
    <Form form={form} kit={siteKit} onSubmit={setSaved} data-ui-box-demo="" style={{ maxWidth: 420 }}>
      <F.Field name="needInvoice" title="Request invoice" component={FieldSwitch} />
      <UI name="invoiceBox" component={[Card, { title: "Invoice details", padding: 'sm' }]}>
        <div style={{ display: 'grid', gap: 12 }}>
          <F.Field name="invoiceTitle" title="Bill to" required component={FieldInput} />
          <F.Field name="remark" title="Notes" component={FieldInput} />
        </div>
      </UI>
      <div><Submit asChild><Button variant="primary">Submit</Button></Submit></div>
      {saved && <pre>{JSON.stringify(saved)}</pre>}
    </Form>
  );
}
```

## Fallbacks

A reaction overrides JSX. Unset visibility, disabled and read-only fall back to the underlying field or scope. Component props are specific to the instance. Assign `undefined` to release an override.

## Check UI names

Extend `UiNames` once to catch misspellings at compile time.

```ts
// src/shikiform.d.ts
import '@shikiform/core'
declare module '@shikiform/core' {
  interface UiNames { echo: true; slider: true; approvalTip: true }
}
```
