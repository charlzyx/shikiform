`<ArrayField>` creates a scope for each row and tracks its identity. A change to the array structure updates the list. Editing one cell only updates its subscribers.

## Reorder a list

Drag a contact or use the move buttons. `ContactRow` and `ContactList` both call `useWatch(selectContactsEditable)` so their actions follow the effective state of the contacts array. Toggle read-only mode to try it.

```tsx preview
import type { ReactNode } from 'react';
import { enUS, Form, typed, useArrayField, useField, useForm, useIndex, useWatch, type Draft } from '@shikiform/react';

type Contact = { name: string; phone: string };
type Values = { contacts: Contact[] };

function TextInput({ placeholder }: { placeholder?: string }) {
  const field = useField<string>();
  if (field.readOnly) return <span>{field.value || '-'}</span>;
  return (
    <span>
      <input id={field.id} value={field.value ?? ''} placeholder={placeholder}
        disabled={field.disabled} onChange={(e) => field.onChange(e.target.value)} onBlur={field.onBlur} />
      {field.errors[0] && <small role="alert" style={{ color: 'crimson', marginLeft: 4 }}>{field.errors[0]}</small>}
    </span>
  );
}

const selectContactsEditable = (d: Draft<Values>) => !d.contacts.$readOnly && !d.contacts.$disabled;

function ContactRow({ children }: { children?: ReactNode }) {
  const array = useArrayField<Contact>();
  const index = useIndex();
  const editable = useWatch(selectContactsEditable);
  return (
    <div {...(editable ? array.dragProps(index) : {})}
      style={{ display: 'flex', gap: 8, alignItems: 'center', marginBottom: 8 }}>
      {editable && <span style={{ cursor: 'grab' }} title="Drag to reorder">⋮⋮</span>}
      <span>#{index + 1}</span>
      {children}
      {editable && (
        <>
          <button type="button" disabled={index === 0} onClick={() => array.move(index, index - 1)}>Move up</button>
          <button type="button" disabled={index === array.keys.length - 1} onClick={() => array.move(index, index + 1)}>Move down</button>
          <button type="button" onClick={() => array.remove(index)}>Delete</button>
        </>
      )}
    </div>
  );
}

function ContactList({ children }: { children?: ReactNode }) {
  const array = useArrayField<Contact>();
  const editable = useWatch(selectContactsEditable);
  return (
    <div>
      {children}
      {editable && <button type="button" onClick={() => array.push({ name: '', phone: '' })}>Add contact</button>}
    </div>
  );
}

export default function App() {
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { contacts: [{ name: 'Ada', phone: '100' }, { name: 'Charles', phone: '200' }] },
    fields: { 'contacts.*.name': { required: true } },
  });
  const F = typed(form);
  return (
    <Form form={form}>
      <p>
        <label>
          <input type="checkbox" onChange={(e) => form.batch((d) => { d.$readOnly = e.target.checked ? true : undefined; })} />
          {' '}Read only
        </label>
      </p>
      <F.ArrayField name="contacts" component={ContactList}>
        {(row) => (
          <ContactRow>
            <row.Field name="name" component={[TextInput, { placeholder: "Name" }]} />
            <row.Field name="phone" component={[TextInput, { placeholder: "Phone" }]} />
          </ContactRow>
        )}
      </F.ArrayField>
      <F.Watch select={(d) => d.contacts.$value}>{(rows) => <pre>{JSON.stringify(rows, null, 2)}</pre>}</F.Watch>
    </Form>
  );
}
```

`typed(form).ArrayField` supplies a `row.Field` typed to one contact. Inside a row, `useIndex()` reads the position and `useRecord<Contact>()` reads its value. Object rows without IDs use object identity, so moving a row preserves its state and input focus; inside a `batch`, pending writes to nested child arrays also follow row identity across moves/swaps (and are dropped if the row is removed). Pass `rowKey` when your data supplies a stable ID. Dropping a row invokes `move` within the same array. In `virtualScroll` mode, virtual rows reconcile with stable row keys so reordering rows inside the visible window preserves row component state. Calling row operations (`push`, `insert`, `remove`, `move`, `swap`, `replace`) on a stale array handle whose current value is no longer an array initializes `null`/`undefined` to `[]`, and for any other non-array value throws `ShikiUsageError` in development and is ignored in production to protect the existing value.

## Read-only and disabled arrays

Rows inherit these states from the form or array. Apply them to editing buttons and drag handles as well as inputs.

```ts
useForm({ initialValues, readOnly: true })          // Start the whole form in read-only mode.
form.batch((d) => { d.$disabled = true })           // Disable the whole form. Assign undefined to restore the lower layer.
form.batch((d) => { d.contacts.$readOnly = true })  // Make just this array read-only.
```

## Array-level errors

A rule declared on `members.*.phones` validates each phone list as a whole. Render its messages once in the array container, where they remain visible even when the list is empty. `useArrayField().errors` and `warnings` belong to that array. The draft's `$errors` also includes descendant fields.

```tsx preview
import type { ReactNode } from 'react';
import { enUS, Form, Submit, typed, useArrayField, useField, useForm, useIndex } from '@shikiform/react';

type Phone = { no: string };
type Member = { name: string; phones: Phone[] };
type Values = { members: Member[] };

function TextInput({ placeholder }: { placeholder?: string }) {
  const field = useField<string>();
  return <input value={field.value ?? ''} placeholder={placeholder}
    onChange={(e) => field.onChange(e.target.value)} onBlur={field.onBlur} />;
}

function PhoneList({ children }: { children?: ReactNode }) {
  const array = useArrayField<Phone>();
  return (
    <div>
      {children}
      {array.errors.length > 0 && <p role="alert" style={{ color: 'crimson', margin: '4px 0' }}>{array.errors.join(', ')}</p>}
      <button type="button" onClick={() => array.push({ no: '' })}>Add phone</button>
    </div>
  );
}

function RemovePhone() {
  const array = useArrayField<Phone>();
  const index = useIndex();
  return <button type="button" onClick={() => array.remove(index)}>Delete</button>;
}

export default function App() {
  const form = useForm<Values>({
    messages: enUS,
    initialValues: { members: [{ name: 'Ada', phones: [] }] },
    fields: {
      'members.*.phones': {
        rules: { validator: (phones) => (Array.isArray(phones) && phones.length ? null : "Add at least one phone") },
      },
    },
  });
  const F = typed(form);
  return (
    <Form form={form}>
      <F.ArrayField name="members">
        {(member) => (
          <div style={{ border: '1px solid #ddd', padding: 10, marginBottom: 10 }}>
            <div style={{ marginBottom: 8 }}>
              <member.Field name="name" component={[TextInput, { placeholder: "Name" }]} />
            </div>
            <member.ArrayField name="phones" component={PhoneList}>
              {(phone) => (
                <div style={{ display: 'flex', gap: 6, marginBottom: 6 }}>
                  <phone.Field name="no" component={[TextInput, { placeholder: "Phone" }]} />
                  <RemovePhone />
                </div>
              )}
            </member.ArrayField>
          </div>
        )}
      </F.ArrayField>
      <Submit>Submit</Submit>
    </Form>
  );
}
```

Submit the form with an empty list, then add a phone. Once a structural error is visible, an insert, removal, move or replacement revalidates the array. Mount the container inside `<ArrayField>` to provide its context.

## Editable tables with useArrayTable

Use `useArrayTable` for pagination, selection, expansion and row operations. `<ArrayField.Row>` establishes the correct scope for each visible row. Table presentation state stays outside submitted values.

```tsx preview play=array-table
import { enUS, ArrayField, Field, Form, useField, useForm, useRecord } from '@shikiform/react';
import { useArrayTable } from '@shikiform/react';

type Row = { id: string; name: string; price: number; qty: number };
let nextId = 4;
const createRow = (): Row => ({ id: String(nextId++), name: '', price: 0, qty: 1 });

function CellText() {
  const field = useField<string>();
  return <input value={field.value ?? ''} onChange={(e) => field.onChange(e.target.value)} onBlur={field.onBlur} />;
}

function CellNumber() {
  const field = useField<number>();
  return <input type="number" style={{ width: 72 }} value={field.value ?? 0}
    onChange={(e) => field.onChange(Number(e.target.value))} onBlur={field.onBlur} />;
}

function Subtotal() {
  const row = useRecord<Row>();
  return <span>{((row.price ?? 0) * (row.qty ?? 0)).toFixed(2)}</span>;
}

function PurchaseTable() {
  const table = useArrayTable<Row>('list', {
    rowKey: 'id',
    createRowKey: () => String(nextId++),
    pagination: { pageSize: 5 },
  });
  return (
    <div>
      <div style={{ marginBottom: 8 }}>
        <button type="button" onClick={() => table.add(createRow())}>Add row</button>
      </div>
      <table>
        <thead>
          <tr><th>#</th><th>Name</th><th>Unit price</th><th>Quantity</th><th>Subtotal</th><th>Actions</th></tr>
        </thead>
        <tbody>
          {table.pageRows.map(({ key, index, position }) => (
            <ArrayField.Row key={key} path={`list.${index}`} index={index} itemKey={key}>
              <tr>
                <td>{position}</td>
                <td><Field name="name" required component={[CellText]} /></td>
                <td><Field name="price" component={[CellNumber]} /></td>
                <td><Field name="qty" component={[CellNumber]} /></td>
                <td><Subtotal /></td>
                <td>
                  <button type="button" onClick={() => table.copy(index)}>Copy</button>{' '}
                  <button type="button" onClick={() => void table.remove(index)}>Delete</button>
                </td>
              </tr>
            </ArrayField.Row>
          ))}
        </tbody>
      </table>
    </div>
  );
}

export default function App() {
  const form = useForm<{ list: Row[] }>({
    messages: enUS,
    initialValues: { list: [
      { id: '1', name: "Pen", price: 12, qty: 2 },
      { id: '2', name: "Notebook", price: 8.5, qty: 3 },
      { id: '3', name: "Ink", price: 20, qty: 1 },
    ] },
  });
  return <Form form={form}><PurchaseTable /></Form>;
}
```
