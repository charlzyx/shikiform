不用 antd、或者要自己做表格和树组件时，直接用 Hook。`<Field>`、`<ArrayField>` 都是它们的薄封装；下面的例子只用原生 HTML。

- `useField(name)` 绑定一个字段，`useWatch(select)` / `<Watch>` 订阅任意 `$` 状态。
- 行内：`useArray()` 拿最近一层数组的操作，`useIndex()` 拿行号，`useRecord()` 拿这一行的值；都不需要写数组名。
- 树节点内：`useTreeNode()` 返回 `{ path, pos, index, record }`。
- 拖拽：数组句柄（`useArray()` / `useArrayField(name)`）的 `dragProps(index)`、树句柄（`useTreeField(name)`）的 `dragProps(pos)`，给出 HTML5 拖拽属性；移动本身是同一个句柄的 `move`。

`useArray` / `useIndex` / `useRecord` 在 `<ArrayField>` 的行里都能用，`ArrayItems` / `ArrayCards` / `ArrayTable` 容器里也一样。

## 数组：行组件不需要名字

```tsx preview
import { ArrayField, Form, Watch, useArray, useField, useForm, useIndex, useRecord, type Draft } from 'shikiform';

type Todo = { id: number; title: string; done: boolean };
type Values = { todos: Todo[] };
let nextId = 4;

function TitleInput() {
  const field = useField<Todo, 'title'>('title', { required: true });
  return (
    <span>
      <input aria-label="标题" value={field.value ?? ''} onChange={(event) => field.onChange(event.target.value)} onBlur={field.onBlur} />
      {field.errors[0] && <small style={{ color: 'crimson' }}> {field.errors[0]}</small>}
    </span>
  );
}

function DoneBox() {
  const field = useField<Todo, 'done'>('done');
  return <input type="checkbox" aria-label="完成" checked={field.value} onChange={(event) => field.onChange(event.target.checked)} />;
}

function Row() {
  const array = useArray<Todo>();
  const index = useIndex();
  const record = useRecord<Todo>();
  return (
    <li {...array.dragProps(index)} style={{ cursor: 'grab', opacity: record.done ? 0.5 : 1 }}>
      ⠿ <DoneBox /> <TitleInput /> <small>#{index} · id {record.id}</small>{' '}
      <button type="button" onClick={() => array.remove(index)}>删除</button>
    </li>
  );
}

function TodoList() {
  return (
    <ul style={{ paddingLeft: 0, listStyle: 'none' }}>
      <ArrayField name="todos" rowKey="id">{() => <Row />}</ArrayField>
    </ul>
  );
}

export default function App() {
  const form = useForm<Values>({
    initialValues: { todos: [{ id: 1, title: '读文档', done: true }, { id: 2, title: '写表单', done: false }, { id: 3, title: '拖动排序', done: false }] },
  });
  return (
    <Form form={form}>
      <TodoList />
      <button type="button" onClick={() => form.pushArray('todos', { id: nextId++, title: '', done: false })}>添加</button>
      <Watch select={(d: Draft<Values>) => d.todos.$value.filter((todo) => todo.done).length}>
        {(done) => <p>已完成 {done} 项</p>}
      </Watch>
    </Form>
  );
}
```

行里的 `useArray()` 就是外层数组的句柄：`remove`、`move`、`dragProps` 都在上面，不用写数组名。拖动一行，`#` 后的行号变了，`id` 不变。

## 树：每个节点一层 RowScope

树没有现成的行组件，节点由你递归渲染。每个节点包一层 `<RowScope path pos>`，里面的 `useField('name')` 就指向这个节点，`useTreeNode()` 给出它的坐标和值：

```tsx
function NodeLabel({ dragProps }: { dragProps: TreeHandle['dragProps'] }) {   // TreeHandle = ReturnType<typeof useTreeField>
  const { pos, record } = useTreeNode<Node>()
  const field = useField<Node, 'name'>('name')
  return <div {...dragProps(pos)}>⠿ <input value={field.value} onChange={(e) => field.onChange(e.target.value)} /> {record.children?.length ?? 0}</div>
}

<RowScope path={`${path}.${index}`} pos={[...pos, index]}>
  <NodeLabel dragProps={tree.dragProps} />
  {node.children?.length ? <Branch path={`${path}.${index}.children`} nodes={node.children} pos={[...pos, index]} /> : null}
</RowScope>
```

`useTreeField('menu')` 返回的 `tree.dragProps(pos)` 给节点加上拖拽：拖到另一个节点上就移动到那个位置，不能拖进自己的子孙。可运行的完整版见 [示例库 · 树：跨层移动](/examples/arrays#树跨层移动)。

## 接自己的组件库

`connect` / `mapProps` / `mapBlur` / `mapFieldProps` 在 `shikiform` 里，不需要 `shikiform/antd`，也不需要安装 antd。默认不传 antd 的 `status`。控件类型里没有 `onBlur` 时（弹出层 Select），用 `blurOn: 'onOpenChange'`（关闭时失焦）或 `blurOn: 'onChange'`，否则失焦校验不会跑，类型会报错。

标签和错误用 [`useFieldDecorator`](/reference/react-hooks#usefielddecorator)，按钮用 [`Submit` / `Reset`](/reference/react-hooks#usesubmit)（`asChild` 或 render prop）。`Reset` 的 `afterReset` 在重置之后拿到新值。

下面是一套 shadcn 风格的最小适配（原生 input / select 代替 shadcn 组件，不增加依赖）。`FormItem` 只展开 hook 的数据；Select 没有 `onBlur`，关闭（这里是选中后）走 `blurOn`。

```tsx preview
import { useState, type ReactNode } from 'react';
import { Field, Form, Reset, Submit, connect, mapProps, mapReadPretty, useFieldDecorator, useForm } from 'shikiform';

function TextControl(props: {
  id?: string;
  value?: string;
  placeholder?: string;
  disabled?: boolean;
  readOnly?: boolean;
  onChange?: (event: { target: { value: string } }) => void;
  onBlur?: () => void;
}) {
  return (
    <input
      id={props.id}
      placeholder={props.placeholder}
      disabled={props.disabled}
      readOnly={props.readOnly}
      value={props.value ?? ''}
      onChange={props.onChange}
      onBlur={props.onBlur}
    />
  );
}

function PopoverSelect(props: {
  id?: string;
  value?: string;
  disabled?: boolean;
  options?: { label: string; value: string; disabled?: boolean }[];
  onChange?: (value: string) => void;
  onOpenChange?: (open: boolean) => void;
}) {
  return (
    <select
      id={props.id}
      disabled={props.disabled}
      value={props.value ?? ''}
      onChange={(event) => {
        props.onChange?.(event.target.value);
        props.onOpenChange?.(false);
      }}
    >
      <option value="">请选择</option>
      {(props.options ?? []).map((option) => (
        <option key={String(option.value)} value={String(option.value)} disabled={option.disabled}>{option.label}</option>
      ))}
    </select>
  );
}

const Input = connect(TextControl, mapReadPretty(({ value }: { value?: string }) => <span>{value || '—'}</span>));
const Select = connect(PopoverSelect, mapProps({ options: 'options', blurOn: 'onOpenChange' }));

function FormItem({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return (
    <div style={{ display: 'grid', gap: 4, marginBottom: 12 }}>
      {d.label ? <label {...d.labelProps}>{d.label}{d.required ? <span aria-hidden="true"> *</span> : null}</label> : null}
      {children}
      {d.descriptionProps ? <p {...d.descriptionProps} style={{ margin: 0, color: '#666' }}>{d.description}</p> : null}
      {d.errorProps ? <p {...d.errorProps} style={{ margin: 0, color: 'crimson' }}>{d.errors.join('、')}</p> : null}
    </div>
  );
}

export default function App() {
  const [saved, setSaved] = useState('还没有提交');
  const form = useForm({ initialValues: { name: '', city: '' } });
  return (
    <Form form={form} onSubmit={(values) => setSaved(JSON.stringify(values))}>
      <Field name="name" title="姓名" description="对外显示" required decorator={[FormItem]} component={[Input, { placeholder: 'Ada' }]} />
      <Field name="city" title="城市" required enum={[{ label: '杭州', value: 'hz' }, { label: '南京', value: 'nj' }]} decorator={[FormItem]} component={[Select]} />
      <Submit asChild><button>提交</button></Submit>{' '}
      <Reset afterReset={() => setSaved('已重置')} asChild><button>重置</button></Reset>
      <pre aria-live="polite">{saved}</pre>
    </Form>
  );
}
```

控件自己带 label / error（Mantine）时不再套 decorator：`control` 只给标签、错误、`id` 和 `onBlur`，**不含** `value` / `onChange`。值用 `useField()` 接上，不要把 `control` 摊到 `connect()` 过的控件上（两边都会写 `id` / `onBlur`）。

```tsx
import { Field, useField, useFieldDecorator } from 'shikiform'

function MantineInput() {
  const field = useField<string>()
  const { control } = useFieldDecorator()
  return (
    <TextInput
      {...control}
      value={field.value ?? ''}
      onChange={(event) => field.onChange(event.currentTarget.value)}
    />
  )
}

<Field name="name" title="姓名" required component={[MantineInput]} />
```

签名见 [React Hooks](/reference/react-hooks#usefielddecorator)。组件注册表（默认 decorator、按名字登记控件）和按控件的值转换（日期 format / parse）不在这一层。

签名、订阅边界和更多例子见 [React Hooks 参考](/reference/react-hooks)。
