## 试试费用报销单

切换费用类型，每行独立联动；增删明细，合计与审批方式随之变化。

```tsx preview
import { useEffect, useRef, useState, type ReactNode } from 'react';
import { Select, SelectContent, SelectItem, SelectItemIndicator, SelectTrigger, SelectValue } from '@ui-sdk/react/select';
import { defineEffects, Form, Submit, typed, useArrayField, useField, useFieldDecorator, useForm, useIndex, useWatch, type Draft } from '@shikiform/react';

type Expense = { category: 'travel' | 'meal'; amount: number | null; city: string; attendees: string };
type Values = { expenses: Expense[]; summary: { total: number; approval: 'standard' | 'manager' } };

const effects = defineEffects<Values>(({ reaction, draft }) => {
  reaction.each(draft.expenses, (row) => {
    const { category, city, attendees } = row;
    const travel = category.$value === 'travel';
    city.$visible = travel;
    city.$required = travel;
    city.$omit = !travel;
    attendees.$visible = !travel;
    attendees.$required = !travel;
    attendees.$omit = travel;
  });
  reaction(({ expenses, summary }) => {
    const total = expenses.$value.reduce((sum, item) => sum + (item.amount ?? 0), 0);
    summary.$value.total = Math.round(total * 100) / 100;
    summary.$value.approval = total > 1000 ? 'manager' : 'standard';
  });
});

function FieldItem({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return <div className="claim-field">
    <label {...d.labelProps}>{d.label}{d.required && <span aria-hidden="true"> *</span>}</label>
    {children}
    {d.errorProps && <small {...d.errorProps} className="claim-error">{d.errors[0]}</small>}
  </div>;
}
function TextInput() {
  const field = useField<string>();
  const d = useFieldDecorator();
  return <input aria-required={field.required} aria-invalid={d.invalid || undefined} aria-describedby={d.describedBy} id={field.id} value={field.value ?? ''} onChange={(e) => field.onChange(e.target.value)} onBlur={field.onBlur} />;
}
function AmountInput() {
  const field = useField<number | null>();
  const d = useFieldDecorator();
  return <input aria-required={field.required} aria-invalid={d.invalid || undefined} aria-describedby={d.describedBy} id={field.id} type="number" min="0.01" step="0.01" inputMode="decimal" value={field.value ?? ''}
    onChange={(e) => field.onChange(e.target.value === '' ? null : Number(e.target.value))} onBlur={field.onBlur} />;
}
function CategorySelect() {
  const field = useField<string>();
  const d = useFieldDecorator();
  const items = field.options.map((o) => ({ value: String(o.value), label: o.label, disabled: o.disabled }));
  return (
    <Select items={items} value={field.value || null} disabled={field.disabled} onValueChange={(next) => next && field.onChange(next)} onOpenChange={(open) => { if (!open) field.onBlur(); }}>
      <SelectTrigger id={field.id} aria-required={field.required} aria-invalid={d.invalid || undefined} aria-describedby={d.describedBy}>
        <SelectValue />
      </SelectTrigger>
      <SelectContent>
        {items.map((option) => (
          <SelectItem key={option.value} value={option.value} disabled={option.disabled}>
            {option.label}
            <SelectItemIndicator aria-hidden="true">✓</SelectItemIndicator>
          </SelectItem>
        ))}
      </SelectContent>
    </Select>
  );
}
function Expenses({ children }: { children?: ReactNode }) {
  const expenses = useArrayField<Expense>();
  return <div className="claim-lines">{children}
    <button className="claim-add" type="button" onClick={() => expenses.push({ category: 'travel', amount: null, city: '', attendees: '' })}>+ 添加费用</button>
  </div>;
}
function RowHeader() {
  const expenses = useArrayField<Expense>();
  const index = useIndex();
  return <div className="claim-row-head"><span>明细 {String(index + 1).padStart(2, '0')}</span>
    <button type="button" aria-label={`移除 ${index + 1}`} disabled={expenses.keys.length <= 1} onClick={() => expenses.remove(index)}>移除</button>
  </div>;
}
function Summary() {
  const { total, approval } = useWatch((d: Draft<Values>) => d.summary.$value);
  return <div className="claim-summary" aria-live="polite">
    <div><span>报销合计</span><strong>¥ {total.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</strong></div>
    <div className="claim-approval"><b>{approval === 'manager' ? '主管审批' : '常规审批'}</b><small>超过 ¥1,000 转主管审批</small></div>
  </div>;
}
function ForgetResult({ clear }: { clear: () => void }) {
  const values = useWatch((d: Draft<Values>) => d.$value);
  const clearRef = useRef(clear);
  clearRef.current = clear;
  useEffect(() => { clearRef.current(); }, [values]);
  return null;
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({
    initialValues: {
      expenses: [
        { category: 'travel', amount: 680, city: '上海', attendees: '' },
        { category: 'meal', amount: 240, city: '', attendees: '产品组 · 3 人' },
      ],
      summary: { total: 0, approval: 'standard' },
    },
    effects: [effects],
  });
  const F = typed(form);
  return <section className="claim-demo">
    <div className="claim-heading"><span className="claim-seal" aria-hidden="true">支</span><div><small>EXPENSE CLAIM · LIVE</small><h3>一次出差，两笔费用。</h3></div><span className="claim-local">本地演示</span></div>
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values, null, 2))} onInvalid={() => setResult('')}>
      <ForgetResult clear={() => setResult('')} />
      <F.ArrayField name="expenses" component={Expenses}>
        {(row) => <div className="claim-row">
          <RowHeader />
          <div className="claim-fields">
            <row.Field name="category" title="费用类型" enum={[{ label: '差旅交通', value: 'travel' }, { label: '团队用餐', value: 'meal' }]} decorator={[FieldItem]} component={[CategorySelect]} />
            <row.Field name="amount" title="金额（元）" required rules={{ min: 0.01 }} decorator={[FieldItem]} component={[AmountInput]} />
            <row.Field name="city" title="出差城市" decorator={[FieldItem]} component={[TextInput]} />
            <row.Field name="attendees" title="用餐人员" decorator={[FieldItem]} component={[TextInput]} />
          </div>
        </div>}
      </F.ArrayField>
      <Summary />
      <div className="claim-actions"><span>试着切换费用类型，或把合计调到 1,000 元以上。</span><Submit>校验并预览 JSON →</Submit></div>
      {result && <div className="claim-result"><strong>提交数据 · 未发送任何请求</strong><pre aria-live="polite">{result}</pre></div>}
    </Form>
  </section>;
}
```

## 与常见表单方案的差异

| 维度 | ShikiForm | react-hook-form | Formily |
|---|---|---|---|
| 状态与联动 | 集中在引擎内，在 `reaction` 中对 `$` 状态直接赋值 | 多在组件内调用 `watch` / `setValue` | 使用 `reactions`，常与 Schema 配合 |
| UI 绑定 | 引擎零 UI 依赖，通过 `useField` / `connect` / `useFieldDecorator` 接入任意组件库 | 无 UI，受控组件通过 `Controller` 桥接 | 提供独立 UI 适配包 |
| 显隐与提交 | `visible: false` 不渲染、跳过校验，值仍提交；是否提交只看 `omit`。`disabled` 置灰但仍提交 | 由组件挂载状态决定；`disabled` 默认不提交 | `hidden` 默认跳过校验并保留值，`none` 不提交 |

## 安装与模块划分

React 项目安装 `@shikiform/react` 即可，已包含表单引擎、Headless React 绑定以及查询列表、草稿表单、数组表格等无 UI 业务 Hooks：

```bash
npm i @shikiform/react
```

| 包名 | 职责 |
|---|---|
| `@shikiform/react` | `Form`、`Field`、`useField`、`connect`、`useFieldDecorator` 等 React 绑定，以及 `useQueryTable`、`useShadowForm`、`useArrayTable`、`useColumnSettings` 等无 UI 业务 Hooks，并重导出 `@shikiform/core` 常用 API |
| `@shikiform/core` | 纯 TypeScript 表单引擎：状态存储、draft、`reaction` / `watch` 与校验，不依赖 React（仅非 React 环境单独安装） |

完整导出清单见 [API 总览](/api)。使用 AI 编程工具时可将 [/llms.txt](/llms.txt) 作为上下文传入。
