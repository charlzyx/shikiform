## 试试开票表单

切换发票类型、修改或增删明细，提交查看 JSON；展开代码查看 effects 中的嵌套对象更新与数组逐行计算。

```tsx preview
import { useEffect, useRef, useState, type ReactNode } from 'react';
import { defineEffects, Form, Submit, typed, useArrayField, useField, useFieldDecorator, useForm, useIndex, useWatch, type Draft } from '@shikiform/react';

type Item = { price: number | null; qty: number | null; subtotal: number };
type Values = {
  invoice: { kind: 'personal' | 'company'; taxId: string };
  items: Item[];
  summary: { total: number };
};

const effects = defineEffects<Values>(({ reaction }) => {
  reaction((d) => {
    const company = d.invoice.kind.$value === 'company';
    d.invoice.taxId.$visible = company;
    d.invoice.taxId.$required = company;
    d.invoice.taxId.$omit = !company;
  });
  reaction.each((d) => d.items, (row) => {
    row.subtotal.$value = (row.price.$value ?? 0) * (row.qty.$value ?? 0);
    row.subtotal.$readOnly = true;
  });
  reaction((d) => {
    d.summary.$value.total = d.items.$value.reduce((sum, item) => sum + item.subtotal, 0);
  });
});

function FieldItem({ children }: { children?: ReactNode }) {
  const d = useFieldDecorator();
  return (
    <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', gap: 8 }}>
      <label {...d.labelProps}>
        {d.required && <span style={{ color: 'crimson' }}>* </span>}
        {d.label}
      </label>
      {children}
      {d.errorProps && <small {...d.errorProps} style={{ color: 'crimson' }}>{d.errors[0]}</small>}
    </div>
  );
}

function TextInput() {
  const field = useField<string>();
  return <input id={field.id} value={field.value ?? ''} style={{ maxWidth: '100%' }}
    onChange={(e) => field.onChange(e.target.value)} onBlur={field.onBlur} />;
}

function NumberInput() {
  const field = useField<number | null>();
  return <input id={field.id} type="number" min={0} step="any" readOnly={field.readOnly}
    value={field.value ?? ''} style={{ width: 100, maxWidth: '100%' }}
    onChange={(e) => field.onChange(e.target.value === '' ? null : Number(e.target.value))}
    onBlur={field.onBlur} />;
}

function RadioGroup() {
  const field = useField<string>();
  return (
    <span style={{ display: 'inline-flex', flexWrap: 'wrap', gap: 12 }}>
      {field.options.map((o) => (
        <label key={String(o.value)}>
          <input type="radio" name={field.id} checked={field.value === o.value}
            onChange={() => field.onChange(String(o.value))} />
          {' '}{o.label}
        </label>
      ))}
    </span>
  );
}

function Items({ children }: { children?: ReactNode }) {
  const items = useArrayField<Item>();
  return <div>
    {children}
    <button type="button" onClick={() => items.push({ price: 0, qty: 1, subtotal: 0 })}>添加明细</button>
  </div>;
}

function RemoveItem() {
  const items = useArrayField<Item>();
  const index = useIndex();
  return <button type="button" onClick={() => items.remove(index)}>删除</button>;
}

function ForgetResult({ clear }: { clear: () => void }) {
  const values = useWatch((d: Draft<Values>) => d.$value);
  const clearRef = useRef(clear);
  clearRef.current = clear;
  useEffect(() => { clearRef.current(); }, [values]);
  return null;
}

export default function App() {
  const [result, setResult] = useState('');
  const form = useForm<Values>({
    initialValues: {
      invoice: { kind: 'personal', taxId: '' },
      items: [{ price: 10, qty: 2, subtotal: 0 }],
      summary: { total: 0 },
    },
    effects: [effects],
  });
  const F = typed(form);
  return (
    <Form form={form} onSubmit={(values) => setResult(JSON.stringify(values, null, 2))} onInvalid={() => setResult('')}>
      <ForgetResult clear={() => setResult('')} />
      <F.Field name="invoice.kind" title="发票类型" enum={[{ label: '个人', value: 'personal' }, { label: '企业', value: 'company' }]}
        decorator={[FieldItem]} component={[RadioGroup]} />
      <F.Field name="invoice.taxId" title="税号" decorator={[FieldItem]} component={[TextInput]} />
      <F.ArrayField name="items" component={Items}>
        {(row) => (
          <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', gap: 12, marginBottom: 12 }}>
            <row.Field name="price" title="单价" required rules={{ min: 0 }} decorator={[FieldItem]} component={[NumberInput]} />
            <row.Field name="qty" title="数量" required rules={{ min: 1 }} decorator={[FieldItem]} component={[NumberInput]} />
            <row.Field name="subtotal" title="小计" decorator={[FieldItem]} component={[NumberInput]} />
            <RemoveItem />
          </div>
        )}
      </F.ArrayField>
      <F.Field name="summary.total" title="合计" readOnly decorator={[FieldItem]} component={[NumberInput]} />
      <Submit>提交</Submit>
      <pre aria-live="polite">{result}</pre>
    </Form>
  );
}
```

## 与常见表单方案的差异

| 维度 | ShikiForm | react-hook-form | Formily |
|---|---|---|---|
| 状态与联动 | 集中在引擎内，在 `reaction` 中对 `$` 状态直接赋值 | 多在组件内调用 `watch` / `setValue` | 使用 `reactions`，常与 Schema 配合 |
| UI 绑定 | 引擎零 UI 依赖，通过 `useField` / `connect` / `useFieldDecorator` 接入任意组件库 | 无 UI，受控组件通过 `Controller` 桥接 | 提供独立 UI 适配包 |
| 显隐与提交 | `visible: false` 不渲染、跳过校验，值仍提交；是否提交只看 `omit`。`disabled` 置灰但仍提交 | 由组件挂载状态决定；`disabled` 默认不提交 | `hidden` 默认跳过校验并保留值，`none` 不提交 |

## 安装与模块划分

React 项目默认安装 `@shikiform/react`，再按需安装无 UI 业务 Hooks 包：

```bash
npm i @shikiform/react
# 查询列表、草稿表单、数组表格等无 UI Hooks
npm i @shikiform/hooks
```

| 包名 | 职责 |
|---|---|
| `@shikiform/core` | 纯 TypeScript 表单引擎：状态存储、draft、`reaction` / `watch` 与校验，不依赖 React |
| `@shikiform/react` | `Form`、`Field`、`useField`、`connect`、`useFieldDecorator` 等 React 绑定，并重导出 `@shikiform/core` |
| `@shikiform/hooks` | `useQueryTable`、`useShadowForm`、`useArrayTable`、`useColumnSettings` 等无 UI 业务 Hooks |

完整导出清单见 [API 总览](/api)。使用 AI 编程工具时可将 [/llms.txt](/llms.txt) 作为上下文传入。
