```bash
npm run build        # 文档站，并把 playground 放进 dist/play/
```

## 一个字段

`useForm` 持有引擎。`<Form>` 把引擎放进上下文。`<Field>` 只订阅自己的 path。

```tsx
import { Field, Form, useForm } from 'shikiform'

interface Values {
  email: string
  name: string
}

export function Profile() {
  const form = useForm<Values>({
    initialValues: { email: '', name: '' },
  })
  return (
    <Form form={form} onSubmit={(values) => console.log(values)}>
      <Field name="email" rules={{ format: 'email', message: '邮箱格式不对' }}>
        {(field) => (
          <label>
            <input
              value={field.value ?? ''}
              onChange={(event) => field.onChange(event.target.value)}
              onBlur={field.onBlur}
            />
            <span>{field.errors[0]}</span>
          </label>
        )}
      </Field>
      <Field name="name" rules={{ required: true, message: '请填写姓名' }}>
        {(field) => (
          <input
            value={field.value ?? ''}
            onChange={(event) => field.onChange(event.target.value)}
            onBlur={field.onBlur}
          />
        )}
      </Field>
      <button type="submit">提交</button>
    </Form>
  )
}
```

`name` 不在 `Values` 里时，类型检查会报错。默认在失焦和提交时校验。

## Ant Design

`shikiform/antd` 用 `createFormHook` 把输入绑好。`name` 同样对照值类型检查。

```tsx
import { useAntdForm } from 'shikiform/antd'

const form = useAntdForm<Values>({ initialValues: { email: '', name: '' } })

<form.AppForm>
  <form.AppField name="email">{(field) => <field.Input placeholder="邮箱" />}</form.AppField>
  <form.Submit>提交</form.Submit>
</form.AppForm>
```

`form.Submit` 订阅 `$canSubmit`。字段还有错误时按钮是禁用的；需要失败后仍能点击提交时传 `disabled={false}`。
